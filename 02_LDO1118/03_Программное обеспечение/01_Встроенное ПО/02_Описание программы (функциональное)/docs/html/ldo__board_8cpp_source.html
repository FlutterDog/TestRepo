33 *
34 * @details

37 static const uint8_t STATUS_LED_MASK = _BV(PC0);

39 /**
40 * @brief Маска входов адресного переключателя.
41 *
42 * @details

45 static const uint8_t ADDRESS_INPUTS_MASK =

48 /**
49 * @brief Устанавливает или сбрасывает выбранные биты PORT-регистра.
50 *
51 * @param port Регистр PORTx.
52 * @param mask Маска изменяемых битов.
53 * @param state true — установить биты; false — сбросить биты.
54 */
55 void writePortMask(volatile uint8_t& port, uint8_t mask, bool state)
56 {
57 if (state)
58 {
59 port |= mask;
60 }
61 else
62 {
63 port &= static_cast<uint8_t>(~mask);
64 }
65 }

88 /**
89 * @brief Инициализирует аппаратные линии платы LDO.
90 */
91 void initIo(void)
92 {
93 /*
94 * Светодиод статуса: PC0, выход, начальное состояние — выключен.
95 */
96 DDRC |= STATUS_LED_MASK;
97 PORTC &= static_cast<uint8_t>(~STATUS_LED_MASK);
98
99 /*
100 * Адресный переключатель: PC1...PC5, входы без внутренней подтяжки.
101 * Стабильные уровни должны формироваться внешней схемой платы.
102 */
103 DDRC &= static_cast<uint8_t>(~ADDRESS_INPUTS_MASK);
104 PORTC &= static_cast<uint8_t>(~ADDRESS_INPUTS_MASK);
105
106 /*
107 * Линия управления RS-485 DE/RE.
108 */
109 rs485ConfigureOutput();
110 rs485ReceiveMode();
111
112 /*
113 * Релейные выходы.
114 *
115 * Ревизия BOARD_OLD_RS485_PB2:
116 * - RS-485 DE/RE: PB2;
117 * - реле: PD2...PD7, PB0, PB1.
118 *
119 * Ревизия BOARD_NEW_RS485_PD2:
120 * - RS-485 DE/RE: PD2;
121 * - реле: PD3...PD7, PB0, PB1, PB2.
122 */
123 if (LDO_BOARD_REVISION == BOARD_OLD_RS485_PB2)
124 {
125 DDRD |= static_cast<uint8_t>(
126 _BV(PD2) | _BV(PD3) | _BV(PD4) | _BV(PD5) | _BV(PD6) | _BV(PD7)
127 );
128
129 DDRB |= static_cast<uint8_t>(
130 _BV(PB0) | _BV(PB1)
131 );
132 }
133 else
134 {
135 DDRD |= static_cast<uint8_t>(
136 _BV(PD3) | _BV(PD4) | _BV(PD5) | _BV(PD6) | _BV(PD7)
137 );
138
139 DDRB |= static_cast<uint8_t>(
140 _BV(PB0) | _BV(PB1) | _BV(PB2)
141 );
142 }
143
144 relaysWrite(0x00);
145 }
146
147 /**
148 * @brief Считывает Modbus-адрес с адресного переключателя.
149 *
150 * @return Modbus-адрес устройства.
151 *
152 * @details
153 * Соответствие входов и битов адреса:
154 * - PC5 -> bit 0, 0b00000001;
155 * - PC4 -> bit 1, 0b00000010;
156 * - PC3 -> bit 2, 0b00000100;
157 * - PC2 -> bit 3, 0b00001000;
158 * - PC1 -> bit 4, 0b00010000.
159 *
160 * Если все входы равны нулю, возвращается адрес 1.
161 */
162 uint8_t readSlaveAddress(void)
163 {
164 uint8_t addr = 0x00;
165
166 if ((PINC & _BV(PC5)) != 0u) { addr |= 0x01u; }
167 if ((PINC & _BV(PC4)) != 0u) { addr |= 0x02u; }
168 if ((PINC & _BV(PC3)) != 0u) { addr |= 0x04u; }
169 if ((PINC & _BV(PC2)) != 0u) { addr |= 0x08u; }
170 if ((PINC & _BV(PC1)) != 0u) { addr |= 0x10u; }
171
172 if (addr == 0x00u)
173 {
174 addr = 0x01u;
175 }
176
177 return addr;
178 }

244 /**
245 * @brief Записывает состояние релейных выходов.
246 *
247 * @param state Битовое состояние релейных выходов.
248 *
249 * @details
250 * Ревизия BOARD_OLD_RS485_PB2:
251 * - bit 0 -> PB1;
252 * - bit 1 -> PB0;
253 * - bit 2 -> PD7;
254 * - bit 3 -> PD6;
255 * - bit 4 -> PD5;
256 * - bit 5 -> PD4;
257 * - bit 6 -> PD3;
258 * - bit 7 -> PD2.
259 *
260 * Ревизия BOARD_NEW_RS485_PD2:
261 * - bit 0 -> PB2;
262 * - bit 1 -> PB1;
263 * - bit 2 -> PB0;
264 * - bit 3 -> PD7;
265 * - bit 4 -> PD6;
266 * - bit 5 -> PD5;
267 * - bit 6 -> PD4;
268 * - bit 7 -> PD3.
269 */
270 void relaysWrite(uint8_t state)
271 {
272 if (LDO_BOARD_REVISION == BOARD_OLD_RS485_PB2)
273 {
274 writePortMask(PORTB, _BV(PB1), (state & 0x01u) != 0u); // 0b00000001
275 writePortMask(PORTB, _BV(PB0), (state & 0x02u) != 0u); // 0b00000010
276
277 writePortMask(PORTD, _BV(PD7), (state & 0x04u) != 0u); // 0b00000100
278 writePortMask(PORTD, _BV(PD6), (state & 0x08u) != 0u); // 0b00001000
279 writePortMask(PORTD, _BV(PD5), (state & 0x10u) != 0u); // 0b00010000
280 writePortMask(PORTD, _BV(PD4), (state & 0x20u) != 0u); // 0b00100000
281 writePortMask(PORTD, _BV(PD3), (state & 0x40u) != 0u); // 0b01000000
282 writePortMask(PORTD, _BV(PD2), (state & 0x80u) != 0u); // 0b10000000
283 }
284 else
285 {
286 writePortMask(PORTB, _BV(PB2), (state & 0x01u) != 0u); // 0b00000001
287 writePortMask(PORTB, _BV(PB1), (state & 0x02u) != 0u); // 0b00000010
288 writePortMask(PORTB, _BV(PB0), (state & 0x04u) != 0u); // 0b00000100
289
290 writePortMask(PORTD, _BV(PD7), (state & 0x08u) != 0u); // 0b00001000
291 writePortMask(PORTD, _BV(PD6), (state & 0x10u) != 0u); // 0b00010000
292 writePortMask(PORTD, _BV(PD5), (state & 0x20u) != 0u); // 0b00100000
293 writePortMask(PORTD, _BV(PD4), (state & 0x40u) != 0u); // 0b01000000
294 writePortMask(PORTD, _BV(PD3), (state & 0x80u) != 0u); // 0b10000000
295 }
