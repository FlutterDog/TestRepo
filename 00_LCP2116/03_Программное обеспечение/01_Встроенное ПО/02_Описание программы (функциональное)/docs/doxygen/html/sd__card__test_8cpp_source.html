3 * @brief Диагностика microSD и проверка базовых файловых операций.
4 *
5 * Обычный sd_card_test_poll() не блокирует: он выполняет debounce card-detect
6 * и не чаще одного раза в секунду повторяет mount. Команда `sd test` является

21constexpr uint32_t SD_DETECT_DEBOUNCE_MS = 50U;
22constexpr uint32_t SD_INIT_RETRY_MS = 1000U;
23
24uint8_t g_detect_stable = 0U;
25uint8_t g_detect_last_raw = 0U;
26uint8_t g_detect_debounced = 0U;
27uint32_t g_detect_last_change_ms = 0U;
28uint32_t g_last_init_attempt_ms = 0U;
29uint8_t g_begin_attempted = 0U;
30LcpSdStorageResult g_last_storage_result = LCP_SD_STORAGE_NOT_READY;
31
32void update_detect(uint32_t now_ms)
33{
34 const uint8_t raw_inserted = lcp_sd_card_inserted();
35
36 if (raw_inserted != g_detect_last_raw)
37 {
38 g_detect_last_raw = raw_inserted;
39 g_detect_last_change_ms = now_ms;
40 g_detect_stable = 0U;
41 return;
42 }
43
44 if ((g_detect_stable == 0U) &&
45 ((uint32_t)(now_ms - g_detect_last_change_ms) >=
46 SD_DETECT_DEBOUNCE_MS))
47 {
48 g_detect_stable = 1U;
49 g_detect_debounced = raw_inserted;
50
51 if (g_detect_debounced == 0U)
52 {
53 lcp_sd_storage_reset();
54 lcp_sd_set_ok_led(0U);
55 g_last_storage_result = LCP_SD_STORAGE_NO_CARD;
56 }
57 }
58}

61void sd_card_test_init(void)
62{
63 lcp_sd_init_pins();
64 lcp_sd_deselect();
65 lcp_sd_set_ok_led(0U);
66
67 g_detect_last_raw = lcp_sd_card_inserted();
68 g_detect_debounced = g_detect_last_raw;
69 g_detect_last_change_ms = millis();
70 g_detect_stable = 0U;

78void sd_card_test_poll(void)
79{
80 const uint32_t now_ms = millis();
81
82 update_detect(now_ms);
83
84 if (g_detect_debounced == 0U)
85 {
86 return;
87 }
88
89 if (lcp_sd_storage_ready() != 0U)
90 {
91 lcp_sd_set_ok_led(1U);
92 return;
93 }
94
95 if ((g_begin_attempted != 0U) &&
96 ((uint32_t)(now_ms - g_last_init_attempt_ms) < SD_INIT_RETRY_MS))
97 {
98 return;
99 }
100
101 g_begin_attempted = 1U;
102 g_last_init_attempt_ms = now_ms;
103 g_last_storage_result = lcp_sd_storage_begin();
104
105 lcp_sd_set_ok_led((g_last_storage_result == LCP_SD_STORAGE_OK) ? 1U : 0U);
106}

147void sd_card_test_run_file_test(void)
148{
149 int16_t values[4] = { 3, 111, -222, 333 };
150 int16_t loaded[4] = { 0, 0, 0, 0 };
151 uint8_t loaded_count = 0U;
152
153 diagnostic_print_banner("MICROSD FILE TEST");
154 SerialUSB.print("test_file=SDTEST.TXT, action=overwrite then read back\r\n");
155
156 sd_card_test_poll();
157
158 if (lcp_sd_storage_ready() == 0U)
159 {
160 SerialUSB.print("result=failed, reason=card not ready\r\n");
161 return;
162 }
163
164 const SdConfigResult save_result =
165 sd_config_save_int16("SDTEST.TXT", values, 3U);
166
167 SerialUSB.print("write_result=");
168 SerialUSB.print(sd_config_result_text(save_result));
169 SerialUSB.print("\r\n");
170
171 if (save_result != SD_CONFIG_OK)
172 {
173 SerialUSB.print("result=failed\r\n");
174 return;
175 }
176
177 const SdConfigResult load_result = sd_config_load_int16(
178 "SDTEST.TXT",
179 loaded,
180 sizeof(loaded) / sizeof(loaded[0]),
181 &loaded_count);
182
183 SerialUSB.print("read_result=");
184 SerialUSB.print(sd_config_result_text(load_result));
185 SerialUSB.print(", loaded_count=");
186 SerialUSB.print(static_cast<int>(loaded_count));
187 SerialUSB.print("\r\n");
188
189 const uint8_t data_match =
190 ((load_result == SD_CONFIG_OK) &&
191 (loaded_count == 3U) &&
192 (loaded[1] == 111) &&
193 (loaded[2] == -222) &&
194 (loaded[3] == 333)) ? 1U : 0U;
195
196 SerialUSB.print("data_match=");
197 SerialUSB.print((data_match != 0U) ? "yes" : "no");
198 SerialUSB.print("\r\nresult=");
199 SerialUSB.print((data_match != 0U) ? "OK" : "FAILED");
200 SerialUSB.print("\r\n");
201}
