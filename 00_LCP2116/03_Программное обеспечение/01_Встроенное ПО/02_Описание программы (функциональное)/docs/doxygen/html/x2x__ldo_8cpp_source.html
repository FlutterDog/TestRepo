18X2XModulePollResult x2x_module_poll_ldo1118(X2XDeviceHeader* device,
19 X2XModuleContext& context)
20{
21 if ((device == 0) ||
22 (device->type != X2X_DEVICE_LDO1118) ||
23 (context.master == 0) ||
24 (context.runtime == 0))
25 {
26 return X2X_MODULE_POLL_CYCLE_COMPLETE;
27 }
28
29 X2XLdo1118* ldo = static_cast<X2XLdo1118*>(device);
30
31 switch (context.runtime->state)
32 {
33 case LDO_POLL_START_WRITE:
34 {
35 if (modbus_rtu_master_ready(*context.master) == 0U)
36 {
37 return X2X_MODULE_POLL_IN_PROGRESS;
38 }
39
40 const uint16_t output_value =
41 static_cast<uint16_t>(ldo->output_value);
42
43 if (modbus_rtu_master_start_write_multiple(*context.master,
44 device->slave_address,
45 0U,
46 &output_value,
47 1U,
48 X2X_MODBUS_TRANSACTION_TIMEOUT_MS) == 0U)
49 {
50 x2x_module_mark_failure(*device,
51 modbus_rtu_master_result(*context.master),
52 modbus_rtu_master_exception_code(*context.master));
53 modbus_rtu_master_reset(*context.master);
54 context.runtime->state = LDO_POLL_START_WRITE;
55 return X2X_MODULE_POLL_CYCLE_COMPLETE;
56 }
57
58 context.runtime->state = LDO_POLL_WAIT_WRITE;
59 return X2X_MODULE_POLL_IN_PROGRESS;
60 }
61
62 case LDO_POLL_WAIT_WRITE:
63 {
64 const ModbusRtuResult result =
65 modbus_rtu_master_result(*context.master);
66
67 if (result == MODBUS_RTU_RESULT_BUSY)
68 {
69 return X2X_MODULE_POLL_IN_PROGRESS;
70 }
71
72 if (result == MODBUS_RTU_RESULT_OK)
73 {
74 x2x_module_mark_success(*device, context.now_ms);
75 }
76 else
77 {
78 x2x_module_mark_failure(*device,
79 result,
80 modbus_rtu_master_exception_code(*context.master));
81 }
82
83 modbus_rtu_master_reset(*context.master);
84 context.runtime->state = LDO_POLL_START_WRITE;
85 return X2X_MODULE_POLL_CYCLE_COMPLETE;
86 }
87
88 default:
89 modbus_rtu_master_reset(*context.master);
90 context.runtime->state = LDO_POLL_START_WRITE;
91 return X2X_MODULE_POLL_CYCLE_COMPLETE;
92 }
93}
