1/**
2 * @file ethernet_modbus_service.hpp
3 * @brief Read-only Modbus TCP server на двух W5500 контроллера LCP2116.
4 *
5 * Слой связывает три независимые части:
6 *
7 * 1. `ethernet_network_config.*` читает MAC/IP/subnet/gateway;
8 * 2. `w5500_lite.*` передаёт и принимает цельные TCP chunks через socket 0;
9 * 3. `modbus_tcp_server.*` разбирает поток и формирует protocol response.
10 *
11 * Прикладная holding map строится только в ethernet_modbus_service.cpp. Для
12 * добавления новых регистров измените размер карты, `update_holding_map()` и
13 * диагностическую расшифровку. W5500 HAL при этом менять не нужно.
14 *
15 * Если требуется новая Modbus-функция, сначала добавьте отдельный callback в
16 * protocol/modbus_tcp, затем подключите его здесь. Не размещайте прикладные
17 * FieldSensor/X2X-зависимости непосредственно в W5500 HAL.
18 */
19

45/**
46 * @brief Младшие биты quality-регистра одного FieldSensor-порта.
47 *
48 * Старший байт содержит числовой ModbusRtuResult последней завершённой или
49 * текущей транзакции. Последние значения сохраняются при ошибке, но потребитель
50 * обязан проверять VALID перед их использованием.
51 */
52enum FieldSensorTcpQuality : uint16_t
53{
54 FIELD_SENSOR_TCP_QUALITY_VALID = 0x0001U, /**< Последние values допустимы. */
55 FIELD_SENSOR_TCP_QUALITY_CONNECTION_LOST = 0x0002U, /**< Достигнут порог ошибок. */
56 FIELD_SENSOR_TCP_QUALITY_PORT_PRESENT = 0x0004U, /**< Физический UART существует. */
57 FIELD_SENSOR_TCP_QUALITY_REQUEST_ACTIVE = 0x0008U, /**< RTU master сейчас занят. */
58 FIELD_SENSOR_TCP_QUALITY_SERVICE_PAUSED = 0x0010U /**< Новые RTU-запросы запрещены. */
59};

61/**
62 * @brief Runtime одного независимого Ethernet-интерфейса.
63 *
64 * ETH1 и ETH2 имеют отдельные W5500, network config, socket state, потоковый
65 * parser и счётчики, но публикуют одинаковую прикладную holding map.
66 */
67struct EthernetModbusInterfaceState
68{
69 W5500NetworkConfig config; /**< Фактически применённые MAC/IP/subnet/gateway. */
70 EthernetNetworkConfigReport config_report; /**< Результаты четырёх файлов этого ETH. */
71 ModbusTcpServer server; /**< Независимый stream/protocol engine соединения. */
72 uint8_t initialized; /**< Выполнена хотя бы одна аппаратная попытка begin. */
73 uint8_t init_ok; /**< VERSIONR W5500 равен ожидаемому 0x04. */
74 uint8_t last_socket_status; /**< Последнее считанное аппаратное SN_SR socket 0. */
75 uint32_t transport_error_count; /**< RX append overflow или неполная постановка ответа. */
76};
