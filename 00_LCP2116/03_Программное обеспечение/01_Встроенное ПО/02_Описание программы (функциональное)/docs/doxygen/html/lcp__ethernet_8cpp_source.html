29static void lcp_ethernet_enable_peripheral_clock(uint32_t peripheral_id)
30{
31 if (peripheral_id < 32U)
32 {
33 PMC->PMC_PCER0 = (1UL << peripheral_id);
34 }
35 else
36 {
37 PMC->PMC_PCER1 = (1UL << (peripheral_id - 32U));
38 }
39}

60static void lcp_ethernet_write_line(const LcpEthernetLine& line, uint8_t value)
61{
62 if (value != 0U)

98void lcp_ethernet_reset(LcpEthernetId ethernet_id)
