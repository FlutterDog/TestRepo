71void initialize_runtime(void)
72{
73 LcpFieldPortConfig serial[LCP_FIELD_PORT_COUNT];
74
75 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
76 {
77 serial[index] = g_configs[index].serial;
78 }
79
80 lcp_field_ports_init(serial);
81 const uint32_t now_ms = millis();
82
83 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
84 {
85 FieldSensorPortState& port = g_ports[index];
86 memset(&port, 0, sizeof(port));
87 port.config = g_configs[index];
88 port.port_present = lcp_field_port_present(
89 static_cast<LcpFieldPortId>(index));
90 port.connection_lost = 1U;
91 port.valid = 0U;
92 port.last_result = (port.port_present != 0U) ?
93 MODBUS_RTU_RESULT_IDLE : MODBUS_RTU_RESULT_TRANSPORT_ERROR;
94 port.next_poll_ms = now_ms;
95
96 modbus_rtu_master_init(
97 port.master,
98 lcp_field_port_transport(static_cast<LcpFieldPortId>(index)),
99 FIELD_SENSOR_MODBUS_INTERFRAME_GAP_MS);

121void schedule_next_request(FieldSensorPortState& port, uint32_t now_ms)
122{
123 port.next_poll_ms += port.config.poll_period_ms;
124
125 if (deadline_reached(now_ms, port.next_poll_ms) != 0U)

131void mark_success(FieldSensorPortState& port, uint32_t now_ms)
132{
133 port.valid = 1U;
134 port.connection_lost = 0U;
135 port.consecutive_failures = 0U;
136 port.last_exception_code = 0U;
137 port.last_result = MODBUS_RTU_RESULT_OK;
138 port.last_update_ms = now_ms;
139 ++port.successful_poll_count;
140}

142void mark_failure(FieldSensorPortState& port,
143 ModbusRtuResult result,
144 uint8_t exception_code)
145{
146 if (port.consecutive_failures < 0xFFU)
147 {
148 ++port.consecutive_failures;
149 }
150
151 ++port.failed_poll_count;
152 port.last_result = result;
153 port.last_exception_code = exception_code;
154
155 if (port.consecutive_failures >= FIELD_SENSOR_CONNECTION_LOSS_THRESHOLD)
156 {
157 port.connection_lost = 1U;
158 port.valid = 0U;
159 }
160}

162void finish_request(FieldSensorPortState& port, uint32_t now_ms)
163{
164 const ModbusRtuResult result = modbus_rtu_master_result(port.master);
165
166 if (result == MODBUS_RTU_RESULT_OK)
167 {
168 mark_success(port, now_ms);
169 }
170 else
171 {
172 mark_failure(port,
173 result,
174 modbus_rtu_master_exception_code(port.master));
175 }
176
177 modbus_rtu_master_reset(port.master);
178 port.request_active = 0U;
179}

181void poll_port(LcpFieldPortId port_id, uint32_t now_ms)
182{
183 FieldSensorPortState& port = g_ports[port_id];
184
185 if ((port.config.role != FIELD_PORT_MASTER) ||
186 (port.port_present == 0U))
187 {
188 return;
189 }
190
191 modbus_rtu_master_poll(port.master);
192
193 if (port.request_active != 0U)
194 {
195 if (modbus_rtu_master_busy(port.master) != 0U)
196 {
197 return;
198 }
199
200 finish_request(port, now_ms);
201 }
202
203 if ((g_paused != 0U) ||
204 (g_config_reload_pending != 0U) ||
205 (deadline_reached(now_ms, port.next_poll_ms) == 0U) ||
206 (modbus_rtu_master_ready(port.master) == 0U))
207 {
208 return;
209 }
210
211 schedule_next_request(port, now_ms);
212
213 if (modbus_rtu_master_start_read_holding(
214 port.master,
215 port.config.slave_address,
216 port.config.start_register,
217 port.config.register_count,
218 port.registers,
219 port.config.timeout_ms) == 0U)
220 {
221 mark_failure(port,
222 modbus_rtu_master_result(port.master),
223 modbus_rtu_master_exception_code(port.master));
224 modbus_rtu_master_reset(port.master);
225 return;
226 }
227
228 port.request_active = 1U;
229}

241void field_sensor_service_poll(void)
242{
243 const uint32_t now_ms = millis();
244
245 if (g_applied_config_generation != lcp_config_service_generation())
246 {
247 g_config_reload_pending = 1U;
248 }
249
250 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
251 {
252 poll_port(static_cast<LcpFieldPortId>(index), now_ms);
253 }
254
255 if ((g_config_reload_pending != 0U) &&
256 (all_requests_idle() != 0U))
257 {
258 apply_active_serial_config();
259 }
260}
