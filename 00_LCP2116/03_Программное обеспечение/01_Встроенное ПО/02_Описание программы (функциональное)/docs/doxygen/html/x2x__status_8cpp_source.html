123uint8_t parse_unsigned(const char*& cursor, uint32_t* value)
124{
125 constexpr uint32_t MAX_VALUE = 65535U;
126 uint32_t result = 0U;
127 uint8_t digit_found = 0U;
128
129 if ((cursor == 0) || (value == 0))
130 {
131 return 0U;
132 }
133
134 while ((*cursor == ' ') || (*cursor == '\t'))
135 {
136 ++cursor;
137 }
138
139 while ((*cursor >= '0') && (*cursor <= '9'))
140 {
141 const uint32_t digit = static_cast<uint32_t>(*cursor - '0');
142
143 if (result > ((MAX_VALUE - digit) / 10U))
144 {
145 return 0U;
146 }
147
148 result = (result * 10U) + digit;
149 digit_found = 1U;
150 ++cursor;
151 }
152
153 if (digit_found == 0U)
154 {
155 return 0U;
156 }
157
158 *value = result;
159 return 1U;
160}

162uint8_t parse_signed(const char*& cursor, int32_t* value)
163{
164 int32_t sign = 1L;
165 uint32_t magnitude = 0U;
166
167 if ((cursor == 0) || (value == 0))
168 {
169 return 0U;
170 }
171
172 while ((*cursor == ' ') || (*cursor == '\t'))
173 {
174 ++cursor;
175 }
176
177 if (*cursor == '-')
178 {
179 sign = -1L;
180 ++cursor;
181 }
182 else if (*cursor == '+')
183 {
184 ++cursor;
185 }
186
187 if (parse_unsigned(cursor, &magnitude) == 0U)
188 {
189 return 0U;
190 }
191
192 const int32_t signed_value = sign * static_cast<int32_t>(magnitude);
193
194 if ((signed_value < -32768L) || (signed_value > 32767L))
195 {
196 return 0U;
197 }
198
199 while ((*cursor == ' ') || (*cursor == '\t'))
200 {
201 ++cursor;
202 }
203
204 if (*cursor != '\0')
205 {
206 return 0U;
207 }
208
209 *value = signed_value;
210 return 1U;
211}

213void handle_ldo_command(const char* arguments)
214{
215 const char* cursor = arguments;
216 uint32_t slave = 0U;
217 int32_t value = 0L;
218
219 if ((parse_unsigned(cursor, &slave) == 0U) ||
220 (parse_signed(cursor, &value) == 0U) ||
221 (slave == 0U) ||
222 (slave > X2X_MAX_MODULES))
223 {
224 SerialUSB.print("usage: x2x ldo SLAVE VALUE\r\n");
225 SerialUSB.print("SLAVE = 1..6 and must reference LDO1118\r\n");
226 SerialUSB.print("VALUE must fit int16\r\n");
227 return;
228 }
229
230 if (x2x_service_set_ldo_output(static_cast<uint8_t>(slave),
231 static_cast<int16_t>(value)) == 0U)
232 {
233 SerialUSB.print("x2x ldo failed\r\n");
234 SerialUSB.print("selected slave is absent or not LDO1118\r\n");
235 return;
236 }
237
238 SerialUSB.print("x2x ldo updated: ");
239 diagnostic_print_assignment("slave");
240 SerialUSB.print(static_cast<unsigned long>(slave));
241 SerialUSB.print(", ");
242 diagnostic_print_assignment("value");
243 SerialUSB.print(static_cast<long>(value));
244 SerialUSB.print("\r\n");
245}

370uint8_t x2x_status_handle_command(const char* command)
371{
372 if (command == 0)
373 {
374 return 0U;
375 }
376
377 if ((strcmp(command, "x2x") == 0) ||
378 (strcmp(command, "x2x status") == 0))
379 {
380 x2x_status_print_report();
381 return 1U;
382 }
383
384 if (strcmp(command, "x2x reload") == 0)
385 {
386 const X2XConfigResult result = x2x_service_reload_config();
387 SerialUSB.print("x2x reload: ");
388 SerialUSB.print(x2x_config_result_text(result));
389 SerialUSB.print("\r\n");
390 x2x_status_print_report();
391 return 1U;
392 }
393
394 if (strcmp(command, "x2x pause") == 0)
395 {
396 x2x_service_pause();
397 SerialUSB.print((x2x_service_paused() != 0U) ?
398 "x2x polling: paused\r\n" :
399 "x2x pause pending until active cycle completes\r\n");
400 return 1U;
401 }
402
403 if (strcmp(command, "x2x resume") == 0)
404 {
405 x2x_service_resume();
406 SerialUSB.print("x2x polling: running\r\n");
407 return 1U;
408 }
409
410 static const char LDO_PREFIX[] = "x2x ldo ";
411
412 if (strncmp(command, LDO_PREFIX, sizeof(LDO_PREFIX) - 1U) == 0)
413 {
414 handle_ldo_command(command + sizeof(LDO_PREFIX) - 1U);
415 return 1U;
416 }
417
418 return 0U;
419}
