11static uint8_t platform_reverse_bits(uint8_t value)
12{
13 value = static_cast<uint8_t>(((value & 0xF0U) >> 4) | ((value & 0x0FU) << 4));
14 value = static_cast<uint8_t>(((value & 0xCCU) >> 2) | ((value & 0x33U) << 2));
15 value = static_cast<uint8_t>(((value & 0xAAU) >> 1) | ((value & 0x55U) << 1));
16 return value;
17}

81uint8_t SPIClass::transfer(uint8_t value)
82{
83 if (initialized_ == 0U)
84 {
85 return 0U;
86 }
87
88 if (settings_.bit_order() == LSBFIRST)
89 {
90 const uint8_t tx_value = platform_reverse_bits(value);
91 const uint8_t rx_value = hal_spi_transfer(tx_value);
92 return platform_reverse_bits(rx_value);
93 }

98uint16_t SPIClass::transfer16(uint16_t value)
99{
100 uint16_t result = 0U;
101
102 if (settings_.bit_order() == LSBFIRST)
103 {
104 result = static_cast<uint16_t>(transfer(static_cast<uint8_t>(value & 0xFFU)));
105 result |= static_cast<uint16_t>(transfer(static_cast<uint8_t>((value >> 8) & 0xFFU))) << 8;
106 }
107 else
108 {
109 result = static_cast<uint16_t>(transfer(static_cast<uint8_t>((value >> 8) & 0xFFU))) << 8;
110 result |= static_cast<uint16_t>(transfer(static_cast<uint8_t>(value & 0xFFU)));
111 }
112
113 return result;
114}
