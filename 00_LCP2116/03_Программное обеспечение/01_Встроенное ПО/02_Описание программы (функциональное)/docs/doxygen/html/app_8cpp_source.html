61uint32_t linker_span_bytes(const uint8_t* begin, const uint8_t* end)
62{
63 const uintptr_t begin_address = reinterpret_cast<uintptr_t>(begin);
64 const uintptr_t end_address = reinterpret_cast<uintptr_t>(end);
65
66 return (end_address >= begin_address) ?
67 static_cast<uint32_t>(end_address - begin_address) : 0U;
68}

70uint8_t usb_console_startup_guard(uint32_t now_ms)
71{
72 const uint8_t usb_open = SerialUSB ? 1U : 0U;
73
74 if ((usb_open != 0U) && (g_usb_was_open == 0U))
75 {
76 g_usb_opened_ms = now_ms;
77 }
78
79 if (usb_open == 0U)
80 {
81 g_usb_opened_ms = now_ms;
82 }
83
84 g_usb_was_open = usb_open;
85
86 if (usb_open == 0U)
87 {
88 return 0U;
89 }
90
91 return ((uint32_t)(now_ms - g_usb_opened_ms) <
92 USB_CONSOLE_STARTUP_GUARD_MS) ? 1U : 0U;
93}

95void initialize_runtime_services(void)
96{
97 if (g_runtime_services_initialized != 0U)
98 {
99 return;
100 }
101
102 field_sensor_service_init();
103 ethernet_modbus_service_init();
104 x2x_service_init();
105 g_runtime_services_initialized = 1U;
106}
107
108void initialize_runtime_services_when_ready(uint32_t now_ms)
109{
110 if (g_runtime_services_initialized != 0U)
111 {
112 return;
113 }
114
115 if (lcp_config_service_source() == LCP_CONFIG_SOURCE_INTERNAL_FLASH)
116 {
117 initialize_runtime_services();
118 return;
119 }
120
121 if (((uint32_t)(now_ms - g_runtime_services_wait_start_ms) >=
122 INITIAL_CONFIG_WAIT_MS) &&
123 (lcp_config_service_busy() == 0U))
124 {
125 /*
126 * При отсутствии корректной Flash и SD контроллер переходит на
127 * безопасные defaults. Если карта появится позже, automatic import
128 * увеличит generation, и сервисы применят новый bundle штатно.
129 */
130 initialize_runtime_services();
131 }
132}

159void setup(void)
160{
161 sam3x_watchdog_boot_recovery_from_watchdog_reset();
162
163 lcp_board_init_gpio();
164 lcp_rs485_init_builtin_ports();
165
166 pinMode(PLC_OK_PIN, OUTPUT);
167 digitalWrite(PLC_OK_PIN, LOW);
168
169 SerialUSB.begin(115200U, SERIAL_8N1);
170 SPI.begin();
171
172 /*
173 * Echo остаётся только на свободных линиях: UART0/X2X до появления модулей
174 * и универсальный HMI. S1–S4 после запуска принадлежат FieldSensor service.
175 */
176 rs485_echo_test_init();
177 sc16is_echo_test_init();
178
179 /*
180 * Внутренняя Flash является рабочим источником конфигурации. microSD
181 * монтируется асинхронно и используется для первоначального импорта,
182 * явного обновления и сервисной резервной копии.
183 */
184 sd_card_test_init();
185 lcp_config_service_init();
186 lcp_config_usb::init();
187 g_runtime_services_wait_start_ms = millis();
188 g_runtime_services_initialized = 0U;
189 g_usb_was_open = 0U;
190 g_usb_opened_ms = 0U;
191
192 /* При наличии валидного Flash-слота интерфейсы запускаются немедленно. */
193 initialize_runtime_services_when_ready(millis());
194
195 battery_status_init();
196 rtc_status_init();
197 diagnostic_console_init();
198 watchdog_status_init();
199}

201void loop(void)
202{
203 static uint32_t last_toggle_ms = 0U;
204 static uint8_t led_state = LOW;
205
206 const uint32_t now_ms = millis();
207 const uint8_t usb_console_guard = usb_console_startup_guard(now_ms);
208
209 if ((uint32_t)(now_ms - last_toggle_ms) >= OK_LED_PERIOD_MS)
210 {
211 last_toggle_ms = now_ms;
212 led_state = (led_state == LOW) ? HIGH : LOW;
213 digitalWrite(PLC_OK_PIN, led_state);
214 }
215
216 /*
217 * USB transport всегда проверяет RX первым. После открытия CDC текстовая
218 * консоль кратковременно удерживается, чтобы первый бинарный кадр Studio не
219 * мог быть принят ею как обычная строка. После guard обе подсистемы
220 * вызываются подряд, без окна, создаваемого опросом остальных сервисов.
221 */
222 lcp_config_usb::poll();
223
224 if ((lcp_config_usb::active() == 0U) && (usb_console_guard == 0U))
225 {
226 diagnostic_console_poll();
227 }
228
229 /* Если USB пишет неактивный слот, SD service не запускает Flash-команды. */
230 sd_card_test_poll();
231
232 if (lcp_config_usb::flash_busy() == 0U)
233 {
234 lcp_config_service_poll();
235 }
236
237 initialize_runtime_services_when_ready(now_ms);
238
239 if (g_runtime_services_initialized != 0U)
240 {
241 x2x_service_poll();
242 field_sensor_service_poll();
243 ethernet_modbus_service_poll();
244 }
245
246 /* UART0 принадлежит либо X2X master, либо fallback echo, но не обоим. */
247 rs485_echo_test_set_x2x_enabled(
248 ((g_runtime_services_initialized == 0U) ||
249 (x2x_service_owns_port() == 0U)) ? 1U : 0U);
250 rs485_echo_test_poll();
251
252 /* SC16IS echo обслуживает свободный диагностический интерфейс HMI. */
253 sc16is_echo_test_poll();
254
255 battery_status_poll();
256 rtc_status_poll();
257 watchdog_status_poll();
258}

260void app_rtos_start(void)
261{
262 const BaseType_t task_result = xTaskCreate(lcp_task,
263 "LCP",
264 LCP_TASK_STACK_WORDS,
265 nullptr,
266 LCP_TASK_PRIORITY,
267 &g_lcp_task_handle);
268
269 configASSERT(task_result == pdPASS);
270
271 if (task_result != pdPASS)
272 {
273 fatal_stop();
274 }
275
276 vTaskStartScheduler();
277
278 /* Scheduler возвращается только при ошибке запуска, например нехватке heap. */
279 fatal_stop();
280}
