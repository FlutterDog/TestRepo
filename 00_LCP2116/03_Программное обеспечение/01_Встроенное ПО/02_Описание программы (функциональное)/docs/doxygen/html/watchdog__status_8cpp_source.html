28void watchdog_status_init(void)
29{
30 sam3x_watchdog_capture_boot_status();
31 sam3x_watchdog_increment_boot_count();
32
33 const Sam3xWatchdogConfig config =
34 {
35 WATCHDOG_TIMEOUT_TICKS,
36 1U,
37 1U,
38 0U,
39 0U
40 };
41
42 sam3x_watchdog_enable(&config);
43 g_last_feed_ms = millis();
44 g_test_reset_armed = 0U;
45}

47void watchdog_status_poll(void)
48{
49 if (g_test_reset_armed != 0U)
50 {
51 return;
52 }
53
54 const uint32_t now_ms = millis();
55
56 if ((uint32_t)(now_ms - g_last_feed_ms) >= WATCHDOG_FEED_PERIOD_MS)
57 {
58 g_last_feed_ms = now_ms;
59 sam3x_watchdog_restart();
60 }
61}

83 diagnostic_print_assignment("mode");

118uint8_t watchdog_status_handle_command(const char* command)
119{
120 if (command == 0)
121 {
122 return 0U;
123 }
124
125 if ((strcmp(command, "watchdog") == 0) ||
126 (strcmp(command, "watchdog status") == 0))
127 {
128 watchdog_status_print_report();
129 return 1U;
130 }
131
132 if (strcmp(command, "watchdog feed") == 0)
133 {
134 sam3x_watchdog_restart();
135 g_last_feed_ms = millis();
136 SerialUSB.print("watchdog feed: ok\r\n");
137 return 1U;
138 }
139
140 if (strcmp(command, "watchdog test reset") == 0)
141 {
142 g_test_reset_armed = 1U;
143 SerialUSB.print("watchdog reset test armed\r\n");
144 SerialUSB.print("automatic feed stopped\r\n");
145 SerialUSB.print("hardware reset is expected after timeout\r\n");
146 SerialUSB.print("next boot may perform USB-recovery reset\r\n");
147 return 1U;
148 }
149
150 return 0U;
151}
