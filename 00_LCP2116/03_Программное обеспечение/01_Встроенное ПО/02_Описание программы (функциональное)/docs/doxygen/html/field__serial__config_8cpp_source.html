19SdConfigResult load_baud(
20 LcpFieldPortConfig configs[LCP_FIELD_PORT_COUNT])
21{
22 uint32_t values[LCP_FIELD_PORT_COUNT + 1U];
23 uint8_t loaded_count = 0U;
24 const SdConfigResult result = sd_config_load_uint32(
25 BAUD_FILE_NAME,
26 values,
27 sizeof(values) / sizeof(values[0]),
28 &loaded_count);
29
30 if (result != SD_CONFIG_OK)
31 {
32 return result;
33 }
34
35 if ((loaded_count != EXPECTED_VALUE_COUNT) ||
36 (values[0] != EXPECTED_VALUE_COUNT))
37 {
38 return SD_CONFIG_INVALID_COUNT;
39 }
40
41 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
42 {
43 const uint32_t baudrate = values[index + 1U];
44
45 if ((baudrate < MINIMUM_BAUDRATE) ||
46 (baudrate > MAXIMUM_BAUDRATE))
47 {
48 return SD_CONFIG_INVALID_VALUE;
49 }
50 }
51
52 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
53 {
54 configs[index].baudrate = values[index + 1U];
55 }
56
57 return SD_CONFIG_OK;
58}

60uint8_t parity_frame(int16_t code, HalUartFrame* frame)
61{
62 if (frame == 0)
63 {
64 return 0U;
65 }
66
67 switch (code)
68 {
69 case 0:
70 *frame = HAL_UART_FRAME_8N1;
71 return 1U;
72 case 1:
73 *frame = HAL_UART_FRAME_8O1;
74 return 1U;
75 case 2:
76 *frame = HAL_UART_FRAME_8E1;
77 return 1U;
78 default:
79 return 0U;
80 }

138void field_serial_config_load(
139 LcpFieldPortConfig configs[LCP_FIELD_PORT_COUNT],
140 FieldSerialConfigReport* report)
141{
142 if ((configs == 0) || (report == 0))
143 {
144 return;
145 }
146
147 report->baud_result = SD_CONFIG_NOT_ATTEMPTED;
148 report->parity_result = SD_CONFIG_NOT_ATTEMPTED;
149 report->loaded_from_sd = 0U;
150
151 LcpFieldPortConfig staged[LCP_FIELD_PORT_COUNT];
152
153 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
154 {
155 staged[index] = configs[index];
156 }
157
158 report->baud_result = load_baud(staged);
159
160 if (report->baud_result == SD_CONFIG_OK)
161 {
162 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
163 {
164 configs[index].baudrate = staged[index].baudrate;
165 }
166
167 report->loaded_from_sd = 1U;
168 }
169
170 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
171 {
172 staged[index] = configs[index];
173 }
174
175 report->parity_result = load_parity(staged);
176
177 if (report->parity_result == SD_CONFIG_OK)
178 {
179 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
180 {
181 configs[index].frame = staged[index].frame;
182 }
183
184 report->loaded_from_sd = 1U;
185 }
