64static Sam3xResetType decode_reset_type(uint32_t reset_status_register)
65{
66 const uint32_t raw_type = (reset_status_register & RSTC_SR_RSTTYP_Msk) >> RSTC_SR_RSTTYP_Pos;
67
68 switch (raw_type)
69 {
70 case 0U:
71 return SAM3X_RESET_GENERAL;
72
73 case 1U:
74 return SAM3X_RESET_BACKUP;
75
76 case 2U:
77 return SAM3X_RESET_WATCHDOG;
78
79 case 3U:
80 return SAM3X_RESET_SOFTWARE;
81
82 case 4U:
83 return SAM3X_RESET_USER;
84
85 default:
86 return SAM3X_RESET_UNKNOWN;
87 }
88}

90void sam3x_watchdog_boot_recovery_from_watchdog_reset(void)
91{
92 const uint32_t reset_status_register = RSTC->RSTC_SR;
93 const Sam3xResetType reset_type = decode_reset_type(reset_status_register);
94
95 if (reset_type == SAM3X_RESET_WATCHDOG)
96 {
97 if (GPBR->SYS_GPBR[1] != WATCHDOG_RECOVERY_IN_PROGRESS_MAGIC)
98 {
99 GPBR->SYS_GPBR[1] = WATCHDOG_RECOVERY_IN_PROGRESS_MAGIC;
100 GPBR->SYS_GPBR[2] = WATCHDOG_RECOVERY_PERFORMED_MAGIC;
101 sam3x_watchdog_software_reset_processor_and_peripherals();
102 }
103
104 return;
105 }
106
107 if ((reset_type == SAM3X_RESET_SOFTWARE) &&
108 (GPBR->SYS_GPBR[1] == WATCHDOG_RECOVERY_IN_PROGRESS_MAGIC))
109 {
110 GPBR->SYS_GPBR[1] = 0UL;
111 return;
112 }
113
114 GPBR->SYS_GPBR[1] = 0UL;
115 GPBR->SYS_GPBR[2] = 0UL;
116}

177void sam3x_watchdog_enable(const Sam3xWatchdogConfig* config)
178{
179 uint32_t mode;
180 uint16_t timeout_ticks;
181
182 if (config == 0)
183 {
184 return;
185 }
186
187 timeout_ticks = config->timeout_ticks_256hz;
188
189 if (timeout_ticks == 0U)
190 {
191 timeout_ticks = 1U;
192 }
193
194 if (timeout_ticks > 0x0FFFU)
195 {
196 timeout_ticks = 0x0FFFU;
197 }
198
199 mode = WDT_MR_WDV(timeout_ticks) | WDT_MR_WDD(0x0FFFU);
200
201 if (config->reset_processor != 0U)
202 {
203 mode |= WDT_MR_WDRPROC;
204 }
205
206 if (config->reset_all != 0U)
207 {
208 mode |= WDT_MR_WDRSTEN;
209 }
210
211 if (config->halt_in_debug != 0U)
212 {
213 mode |= WDT_MR_WDDBGHLT;
214 }
215
216 if (config->halt_in_idle != 0U)
217 {
218 mode |= WDT_MR_WDIDLEHLT;
219 }
220
221 WDT->WDT_MR = mode;
222 sam3x_watchdog_restart();
223}
