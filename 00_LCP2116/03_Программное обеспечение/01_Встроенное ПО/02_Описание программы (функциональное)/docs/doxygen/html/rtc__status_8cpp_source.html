3 * @brief Неблокирующая диагностика и настройка локального RTC ATSAM3X8E.

5 * Команда `rtc set` только запускает аппаратную последовательность обновления.

43uint8_t parse_digits(const char*& cursor,
44 uint8_t digit_count,
45 uint16_t* output)
46{
47 if ((cursor == 0) || (output == 0) || (digit_count == 0U))
48 {
49 return 0U;
50 }
51
52 uint16_t value = 0U;
53
54 for (uint8_t index = 0U; index < digit_count; ++index)
55 {
56 const char character = cursor[index];
57
58 if ((character < '0') || (character > '9'))
59 {
60 return 0U;
61 }
62
63 value = static_cast<uint16_t>(
64 (value * 10U) + static_cast<uint16_t>(character - '0'));
65 }
66
67 cursor += digit_count;
68 *output = value;
69 return 1U;
70}

72uint8_t consume_separator(const char*& cursor, char first, char second)
73{
74 if ((cursor == 0) || ((*cursor != first) && (*cursor != second)))
75 {
76 return 0U;
77 }
78
79 ++cursor;
80 return 1U;
81}

83uint8_t parse_year(const char*& cursor, uint16_t* year)
84{
85 if ((cursor == 0) || (year == 0))
86 {
87 return 0U;
88 }
89
90 const char* probe = cursor;
91 uint16_t parsed = 0U;
92
93 if ((parse_digits(probe, 4U, &parsed) != 0U) &&
94 ((*probe == '-') || (*probe == '.')))
95 {
96 cursor = probe;
97 *year = sam3x_rtc_normalize_year(parsed);
98 return 1U;
99 }
100
101 probe = cursor;
102
103 if ((parse_digits(probe, 2U, &parsed) != 0U) &&
104 ((*probe == '-') || (*probe == '.')))
105 {
106 cursor = probe;
107 *year = static_cast<uint16_t>(2000U + parsed);
108 return 1U;
109 }
110
111 return 0U;
112}

114uint8_t parse_colon_value(const char*& cursor, uint16_t* output)

116 if ((cursor == 0) || (output == 0) || (*cursor != ':'))

118 return 0U;

121 ++cursor;

125uint8_t parse_datetime_value(const char* text,
126 Sam3xRtcDateTime* date_time)
127{
128 if ((text == 0) || (date_time == 0))
129 {
130 return 0U;
131 }
132
133 const char* cursor = skip_spaces(text);
134 uint16_t value = 0U;
135
136 if ((parse_year(cursor, &date_time->year) == 0U) ||
137 (consume_separator(cursor, '-', '.') == 0U) ||
138 (parse_digits(cursor, 2U, &value) == 0U))
139 {
140 return 0U;
141 }
142 date_time->month = static_cast<uint8_t>(value);
143
144 if ((consume_separator(cursor, '-', '.') == 0U) ||
145 (parse_digits(cursor, 2U, &value) == 0U))
146 {
147 return 0U;
148 }
149 date_time->day = static_cast<uint8_t>(value);
150
151 if ((*cursor != ' ') && (*cursor != 't'))
152 {
153 return 0U;
154 }
155
156 ++cursor;
157 cursor = skip_spaces(cursor);
158
159 if (parse_digits(cursor, 2U, &value) == 0U)
160 {
161 return 0U;
162 }
163 date_time->hour = static_cast<uint8_t>(value);
164
165 if (parse_colon_value(cursor, &value) == 0U)
166 {
167 return 0U;
168 }
169 date_time->minute = static_cast<uint8_t>(value);
170
171 if (parse_colon_value(cursor, &value) == 0U)
172 {
173 return 0U;
174 }
175 date_time->second = static_cast<uint8_t>(value);
176
177 cursor = skip_spaces(cursor);
178
179 if (*cursor != '\0')
180 {
181 return 0U;
182 }
183
184 date_time->day_of_week = sam3x_rtc_calculate_day_of_week(
185 date_time->year,
186 date_time->month,
187 date_time->day);
188
189 return (sam3x_rtc_datetime_range_valid(date_time) != 0U) ? 1U : 0U;
190}

237void service_rtc_update_timeout(void)
238{
239 if (sam3x_rtc_update_busy() == 0U)
240 {
241 return;
242 }
243
244 const uint32_t now_ms = millis();
245
246 if ((uint32_t)(now_ms - g_update_start_ms) >= RTC_UPDATE_TIMEOUT_MS)
247 {
248 sam3x_rtc_cancel_datetime_update(SAM3X_RTC_ACK_TIMEOUT);
249 }
250}

292{

296

298 const uint8_t register_valid = sam3x_rtc_registers_valid();

312 SerialUSB.print(sam3x_rtc_result_text(sam3x_rtc_update_result()));
313 SerialUSB.print("\r\nupdate_timeout_ms=");

315 SerialUSB.print("\r\n");
316

320 SerialUSB.print("\r\ndatetime=");
321

359uint8_t rtc_status_handle_command(const char* command)
360{
361 if (command == 0)
362 {
363 return 0U;
364 }
365
366 if ((strcmp(command, "time") == 0) ||
367 (strcmp(command, "rtc time") == 0))
368 {
369 rtc_status_print_time();
370 return 1U;
371 }
372
373 if ((strcmp(command, "rtc") == 0) ||
374 (strcmp(command, "rtc status") == 0))
375 {
376 rtc_status_print_report();
377 return 1U;
378 }
379
380 static const char RTC_SET_PREFIX[] = "rtc set ";
381
382 if (strncmp(command,
383 RTC_SET_PREFIX,
384 sizeof(RTC_SET_PREFIX) - 1U) == 0)
385 {
386 Sam3xRtcDateTime date_time = {};
387 const char* value = command + sizeof(RTC_SET_PREFIX) - 1U;
388
389 if (parse_datetime_value(value, &date_time) == 0U)
390 {
391 SerialUSB.print("rtc set failed: invalid date/time\r\n");
392 SerialUSB.print("usage: rtc set YYYY-MM-DD HH:MM:SS\r\n");
393 SerialUSB.print("accepted year: YYYY or YY\r\n");
394 SerialUSB.print("accepted date separator: '-' or '.'\r\n");
395 SerialUSB.print("accepted date/time separator: space or T\r\n");
396 return 1U;
397 }
398
399 const Sam3xRtcResult start_result =
400 sam3x_rtc_start_datetime_update(&date_time);
401
402 SerialUSB.print("rtc requested_datetime=");
403 print_datetime(date_time);
404 SerialUSB.print("\r\nrtc update_start_result=");
405 SerialUSB.print(sam3x_rtc_result_text(start_result));
406 SerialUSB.print("\r\n");
407
408 if (start_result == SAM3X_RTC_OK)
409 {
410 g_update_start_ms = millis();
411 SerialUSB.print("rtc update is nonblocking\r\n");
412 SerialUSB.print("use 'time' or 'rtc' to inspect completion\r\n");
413 }
414
415 return 1U;
416 }
417
418 return 0U;
419}
