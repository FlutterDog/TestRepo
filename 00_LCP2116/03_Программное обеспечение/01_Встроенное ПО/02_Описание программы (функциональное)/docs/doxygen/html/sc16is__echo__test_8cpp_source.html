77void accept_rx(Sc16isEchoPort& port, uint32_t now_ms)
78{
79 if (port.tx_offset != 0U)
80 {
81 return;
82 }
83
84 uint8_t processed = 0U;
85
86 while ((sc16is_available(port.port.chip_select,
87 port.port.channel) > 0U) &&
88 (processed < SC16IS_MAX_RX_BYTES_PER_POLL))
89 {
90 const int value = sc16is_read(port.port.chip_select,
91 port.port.channel);
92
93 if (value < 0)
94 {
95 break;
96 }
97
98 if (port.length >= SC16IS_ECHO_BUFFER_SIZE)
99 {
100 drop_frame(port, now_ms);
101 return;
102 }
103
104 port.buffer[port.length++] = static_cast<uint8_t>(value);
105 port.last_rx_ms = now_ms;
106 port.response_pending = 0U;
107 port.response_due_ms = 0U;
108 port.tx_deadline_ms = 0U;
109 ++processed;
110 }
111}
112
113void arm_response_if_frame_complete(Sc16isEchoPort& port,
114 uint32_t now_ms)
115{
116 if ((port.length == 0U) || (port.response_pending != 0U))
117 {
118 return;
119 }
120
121 if ((uint32_t)(now_ms - port.last_rx_ms) < SC16IS_INTERFRAME_GAP_MS)
122 {
123 return;
124 }
125
126 port.response_due_ms = now_ms + SC16IS_RESPONSE_DELAY_MS;
127 port.tx_deadline_ms = now_ms + SC16IS_TX_TIMEOUT_MS;
128 port.response_pending = 1U;
129 port.tx_offset = 0U;
130}
131
132void send_if_response_due(Sc16isEchoPort& port, uint32_t now_ms)
133{
134 if (port.response_pending == 0U)
135 {
136 return;
137 }
138
139 if (deadline_reached(now_ms, port.tx_deadline_ms) != 0U)
140 {
141 drop_frame(port, now_ms);
142 return;
143 }
144
145 if (static_cast<int32_t>(now_ms - port.response_due_ms) < 0)
146 {
147 return;
148 }
149
150 const size_t remaining =
151 static_cast<size_t>(port.length - port.tx_offset);
152 const size_t written = sc16is_write_buffer(
153 port.port.chip_select,
154 port.port.channel,
155 &port.buffer[port.tx_offset],
156 remaining);
157
158 port.tx_offset = static_cast<uint16_t>(port.tx_offset + written);
159
160 if (port.tx_offset >= port.length)
161 {
162 drop_frame(port, now_ms);
163 }
164}

194void sc16is_echo_test_poll(void)
195{
196 const uint32_t now_ms = millis();
197
198 for (uint8_t index = 0U; index < g_port_count; ++index)
199 {
200 accept_rx(g_ports[index], now_ms);
201 arm_response_if_frame_complete(g_ports[index], now_ms);
202 send_if_response_due(g_ports[index], now_ms);
203 }
