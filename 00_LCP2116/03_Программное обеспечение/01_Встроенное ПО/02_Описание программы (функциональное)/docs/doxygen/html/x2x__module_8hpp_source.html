30/**
31 * @brief Состояние драйвера отдельного экземпляра модуля.
32 *
33 * Runtime хранится отдельно от данных устройства, чтобы общая структура
34 * X2XDeviceHeader отражала состояние связи, а многошаговая машина драйвера не
35 * загрязняла пользовательские данные конкретного типа.
36 */
37struct X2XModuleRuntime
38{
39 uint8_t state; /**< Текущее состояние драйвера типа. */
40 uint8_t waveform_phase; /**< Текущая фаза A/B/C при чтении LCT. */
41 uint16_t transfer_offset; /**< Смещение очередного блока осциллограммы. */
42 uint32_t next_waveform_check_ms;/**< Срок следующей проверки готовности LCT. */
43};

45/** @brief Общие ресурсы одного вызова драйвера текущего модуля. */
46struct X2XModuleContext
47{
48 ModbusRtuMaster* master; /**< Единственный master внутренней шины X2X. */
49 X2XModuleRuntime* runtime; /**< Runtime текущего slave. */
50 X2XWaveformBuffer* waveform; /**< Общий буфер последней осциллограммы LCT. */
51 uint16_t* register_buffer; /**< Общий временный буфер регистров. */
52 size_t register_capacity; /**< Размер register_buffer в регистрах. */
53 uint32_t now_ms; /**< Один согласованный снимок millis(). */
54};
