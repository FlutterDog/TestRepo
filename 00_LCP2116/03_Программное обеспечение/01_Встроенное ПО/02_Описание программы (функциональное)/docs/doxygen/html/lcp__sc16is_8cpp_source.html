1/**
2 * @file lcp_sc16is.cpp
3 * @brief Карта и одноразовое автоопределение внешних UART SC16IS7xx.
4 *
5 * Микросхемы являются частью стационарной платы и не рассматриваются как
6 * hot-plug устройства. Инициализация SPI/CS и probe поэтому идемпотентны:
7 * echo-test, FieldSensor и повторная загрузка serial-конфигурации используют
8 * одну сохранённую карту и не повторяют scratchpad-тесты без необходимости.
9 * Настройку baud/frame выполняет владелец каждого логического порта.
10 */

36{
37 switch (layout)

89void lcp_sc16is_probe(void)
90{
91 if (g_probe_complete != 0U)
92 {
93 return;
94 }
95
96 lcp_sc16is_init_pins();
97
98 g_sc16is_map.layout = LCP_SC16IS_LAYOUT_UNKNOWN;
99 g_sc16is_map.uart1_ch_a_present =
100 channel_present(LCP_SC16IS_UART1_CS, SC16IS_CHANNEL_A);
101 g_sc16is_map.uart2_ch_a_present =
102 channel_present(LCP_SC16IS_UART2_CS, SC16IS_CHANNEL_A);
103
104 const uint8_t uart2_dual =
105 sc16is_probe_dual_channel(LCP_SC16IS_UART2_CS);
106 g_sc16is_map.uart2_ch_b_present = uart2_dual;
107 g_sc16is_map.uart3_ch_a_present =
108 channel_present(LCP_SC16IS_UART3_CS, SC16IS_CHANNEL_A);
109
110 if (uart2_dual != 0U)
111 {
112 g_sc16is_map.layout = LCP_SC16IS_LAYOUT_DUAL_ON_UART2;
113 g_sc16is_map.hmi =
114 { LCP_SC16IS_UART2_CS, SC16IS_CHANNEL_A, 1U };
115 g_sc16is_map.s1 =
116 { LCP_SC16IS_UART2_CS, SC16IS_CHANNEL_B, 1U };
117 }
118 else
119 {
120 if ((g_sc16is_map.uart2_ch_a_present != 0U) &&
121 (g_sc16is_map.uart3_ch_a_present != 0U))
122 {
123 g_sc16is_map.layout =
124 LCP_SC16IS_LAYOUT_TWO_SINGLE_UART2_UART3;
125 }
126
127 g_sc16is_map.hmi =
128 { LCP_SC16IS_UART2_CS,
129 SC16IS_CHANNEL_A,
130 g_sc16is_map.uart2_ch_a_present };
131 g_sc16is_map.s1 =
132 { LCP_SC16IS_UART3_CS,
133 SC16IS_CHANNEL_A,
134 g_sc16is_map.uart3_ch_a_present };
135 }
136
137 g_sc16is_map.pc =
138 { LCP_SC16IS_UART1_CS,
139 SC16IS_CHANNEL_A,
140 g_sc16is_map.uart1_ch_a_present };
141
142 g_probe_complete = 1U;
143}

151void lcp_sc16is_print_probe_report(void)
152{
153 if (!SerialUSB)
154 {
155 return;
156 }
157
158 lcp_sc16is_probe();
159
160 SerialUSB.print("\r\n[ SC16IS EXTERNAL UARTS ]\r\n");
161 SerialUSB.print("probe_complete=");
162 SerialUSB.print((g_probe_complete != 0U) ? "yes" : "no");
163 SerialUSB.print(", layout=");
164 SerialUSB.print(layout_text(g_sc16is_map.layout));
165 SerialUSB.print("\r\n\r\n-- Physical channels --\r\n");
166 print_present("UART1_CS channel A", g_sc16is_map.uart1_ch_a_present);
167 print_present("UART2_CS channel A", g_sc16is_map.uart2_ch_a_present);
168 print_present("UART2_CS channel B", g_sc16is_map.uart2_ch_b_present);
169 print_present("UART3_CS channel A", g_sc16is_map.uart3_ch_a_present);
170
171 SerialUSB.print("\r\n-- Logical assignment --\r\n");
172 print_assignment("PC", g_sc16is_map.pc);
173 print_assignment("HMI", g_sc16is_map.hmi);
174 print_assignment("S1", g_sc16is_map.s1);
175}
