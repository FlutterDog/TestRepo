26X2XRegistryResult construct_slot(uint8_t index,
27 X2XDeviceType type,
28 uint8_t slave_address,
29 uint16_t asdu)
30{
31 const X2XModuleDescriptor* descriptor = x2x_catalog_find(type);
32
33 if (descriptor == 0)
34 {
35 return X2X_REGISTRY_UNKNOWN_TYPE;
36 }
37
38 if (descriptor->object_size > X2X_DEVICE_SLOT_BYTES)
39 {
40 return X2X_REGISTRY_STORAGE_TOO_SMALL;
41 }
42
43 X2XDeviceHeader* device = descriptor->construct(
44 g_device_storage[index].bytes,
45 sizeof(g_device_storage[index].bytes),
46 slave_address,
47 asdu);
48
49 if (device == 0)
50 {
51 return X2X_REGISTRY_CONSTRUCTION_FAILED;
52 }
53
54 g_devices[index] = device;
55 return X2X_REGISTRY_OK;
56}
57}

66X2XRegistryResult x2x_registry_build(const X2XConfig& config,
67 uint16_t default_asdu)
68{
69 if (config.module_count > X2X_MAX_MODULES)
70 {
71 return X2X_REGISTRY_INVALID_CONFIG;
72 }
73
74 x2x_registry_reset();
75
76 X2XRegistryResult result = construct_slot(0U,
77 X2X_DEVICE_LCP,
78 0U,
79 default_asdu);
80
81 if (result != X2X_REGISTRY_OK)
82 {
83 x2x_registry_reset();
84 return result;
85 }
86
87 for (uint8_t module_index = 0U;
88 module_index < config.module_count;
89 ++module_index)
90 {
91 const uint8_t slave_address = module_index + 1U;
92 const X2XModuleDescriptor* descriptor =
93 x2x_catalog_find(config.module_types[module_index]);
94
95 if ((descriptor == 0) || (descriptor->allowed_in_x2x_config == 0U))
96 {
97 x2x_registry_reset();
98 return X2X_REGISTRY_UNKNOWN_TYPE;
99 }
100
101 result = construct_slot(slave_address,
102 descriptor->type,
103 slave_address,
104 default_asdu);
105
106 if (result != X2X_REGISTRY_OK)
107 {
108 x2x_registry_reset();
109 return result;
110 }
111 }
112
113 g_module_count = config.module_count;
114 return X2X_REGISTRY_OK;
115}
