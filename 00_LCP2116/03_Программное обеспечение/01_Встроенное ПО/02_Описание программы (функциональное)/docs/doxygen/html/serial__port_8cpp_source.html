63size_t SerialPort::write(const uint8_t* buffer, size_t size)
64{
65 if ((initialized_ == 0U) || (buffer == 0))
66 {
67 return 0U;
68 }
69
70 size_t written = 0U;
71
72 for (size_t index = 0U; index < size; ++index)
73 {
74 if (hal_uart_write(port_id_, buffer[index]) == 0U)
75 {
76 break;
77 }

79 ++written;
80 }
