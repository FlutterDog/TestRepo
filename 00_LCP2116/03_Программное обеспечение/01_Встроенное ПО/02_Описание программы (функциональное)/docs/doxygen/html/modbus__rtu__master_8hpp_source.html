1/**
2 * @file modbus_rtu_master.hpp
3 * @brief Неблокирующий Modbus RTU master для одного half-duplex порта.
4 *
5 * Один объект обслуживает только одну транзакцию одновременно. Для нескольких
6 * физических UART создаётся отдельный ModbusRtuMaster на каждый transport.
7 * Объект не владеет transport и пользовательским FC03-буфером: оба должны жить
8 * до завершения транзакции.
9 */
10
11#pragma once
12
13#include <stddef.h>
14#include <stdint.h>
15
16/** Максимальный обычный адрес Modbus RTU slave. Адрес 0 зарезервирован broadcast. */
17static const uint8_t MODBUS_RTU_MAX_SLAVE_ADDRESS = 247U;
18
19/** Максимальное количество регистров FC03 в одной транзакции baseline-master. */
20static const uint16_t MODBUS_RTU_MASTER_MAX_READ_REGISTERS = 100U;
21
22/** Максимальное количество регистров FC10 в одной транзакции baseline-master. */
23static const uint16_t MODBUS_RTU_MASTER_MAX_WRITE_REGISTERS = 16U;
24
25/** Максимальный размер принятого RTU-кадра. */
26static const uint16_t MODBUS_RTU_MASTER_RX_CAPACITY = 256U;
27
28/** Максимальный размер передаваемого RTU-кадра. */
29static const uint16_t MODBUS_RTU_MASTER_TX_CAPACITY = 64U;
30
31static_assert(MODBUS_RTU_MASTER_RX_CAPACITY >=
32 (5U + (2U * MODBUS_RTU_MASTER_MAX_READ_REGISTERS)),
33 "Modbus RTU RX buffer is too small for configured read limit");
34static_assert(MODBUS_RTU_MASTER_TX_CAPACITY >=
35 (9U + (2U * MODBUS_RTU_MASTER_MAX_WRITE_REGISTERS)),
36 "Modbus RTU TX buffer is too small for configured write limit");
37
38/**
39 * @brief Абстрактный байтовый transport одного Modbus RTU-порта.
40 *
41 * Все callback обязаны относиться к одному физическому half-duplex UART.
42 * `write` должен принять весь запрос за один вызов либо вернуть меньше длины;
43 * частичный запрос считается transport error. `tx_idle` означает физическое
44 * окончание передачи последнего stop-bit, а не только пустой software buffer.
45 */
46struct ModbusRtuTransport
47{
48 size_t (*write)(const uint8_t* data, size_t length); /**< Передаёт полный RTU-запрос. */
49 size_t (*available)(void); /**< Возвращает число доступных RX-байтов. */
50 int (*read)(void); /**< Возвращает байт 0..255 либо отрицательное значение. */
51 uint8_t (*tx_idle)(void); /**< Возвращает 1 после физического окончания TX. */
52 void (*clear_rx)(void); /**< Очищает накопленный RX перед новой транзакцией. */
53};
54
55/** @brief Результат последней Modbus RTU-транзакции. */
56enum ModbusRtuResult : uint8_t
57{
58 MODBUS_RTU_RESULT_IDLE = 0U, /**< Транзакция отсутствует. */
59 MODBUS_RTU_RESULT_BUSY = 1U, /**< Выполняется передача или ожидание ответа. */
60 MODBUS_RTU_RESULT_OK = 2U, /**< Ответ полностью проверен. */
61 MODBUS_RTU_RESULT_TIMEOUT = 3U, /**< Ответ не получен за timeout. */
62 MODBUS_RTU_RESULT_CRC_ERROR = 4U, /**< CRC ответа не совпал. */
63 MODBUS_RTU_RESULT_INVALID_RESPONSE = 5U,/**< Адрес, функция, длина или данные ответа неверны. */
64 MODBUS_RTU_RESULT_EXCEPTION = 6U, /**< Slave вернул Modbus exception. */
65 MODBUS_RTU_RESULT_TRANSPORT_ERROR = 7U, /**< UART не принял запрос или не завершил TX. */
66 MODBUS_RTU_RESULT_INVALID_ARGUMENT = 8U /**< Параметры запуска не прошли проверку. */
67};
68
69/** @brief Внутреннее состояние автомата Modbus RTU master. */
70enum ModbusRtuMasterState : uint8_t
71{
72 MODBUS_RTU_MASTER_STATE_IDLE = 0U, /**< Автомат свободен. */
73 MODBUS_RTU_MASTER_STATE_WAIT_TX = 1U, /**< Ожидается физическое окончание передачи. */
74 MODBUS_RTU_MASTER_STATE_WAIT_RESPONSE = 2U /**< Ожидаются байты ответа. */
75};
76
77/** @brief Тип активной транзакции. */
78enum ModbusRtuTransactionType : uint8_t
79{
80 MODBUS_RTU_TRANSACTION_NONE = 0U, /**< Транзакция не выбрана. */
81 MODBUS_RTU_TRANSACTION_READ_HOLDING = 1U, /**< FC03 Read Holding Registers. */
82 MODBUS_RTU_TRANSACTION_WRITE_MULTIPLE = 2U /**< FC10 Write Multiple Registers. */
83};

85/**
86 * @brief Полное состояние одного независимого Modbus RTU master.
87 *
88 * Структура содержит state machine и собственные RX/TX-кадры, но не выделяет
89 * динамическую память. `read_output` устанавливается только на время FC03 и
90 * должен указывать на массив не меньше `register_count` элементов.
91 */
92struct ModbusRtuMaster
93{
94 const ModbusRtuTransport* transport; /**< Transport, живущий не меньше master. */
95 ModbusRtuMasterState state; /**< Текущее состояние автомата. */
96 ModbusRtuTransactionType transaction_type; /**< Тип текущей транзакции. */
97 ModbusRtuResult result; /**< Текущий или итоговый результат. */
98 uint8_t slave_address; /**< Адрес текущего slave, 1..247. */
99 uint8_t function_code; /**< FC03 или FC10. */
100 uint8_t exception_code; /**< Последний exception code. */
101 uint16_t start_address; /**< Начальный адрес регистров. */
102 uint16_t register_count; /**< Количество регистров. */
103 uint16_t* read_output; /**< Внешний буфер результата FC03. */
104 uint32_t deadline_ms; /**< Абсолютный deadline текущей стадии. */
105 uint32_t timeout_ms; /**< Timeout ответа после окончания TX. */
106 uint32_t next_request_ms; /**< Конец обязательной межкадровой паузы. */
107 uint32_t interframe_gap_ms; /**< Минимальная пауза между транзакциями. */
108 uint16_t expected_response_length; /**< Ожидаемая длина обычного ответа. */
109 uint16_t rx_length; /**< Число принятых байтов. */
110 uint16_t tx_length; /**< Длина сформированного запроса. */
111 uint8_t rx_buffer[MODBUS_RTU_MASTER_RX_CAPACITY]; /**< Внутренний RX-кадр. */
112 uint8_t tx_buffer[MODBUS_RTU_MASTER_TX_CAPACITY]; /**< Внутренний TX-кадр. */
113};
