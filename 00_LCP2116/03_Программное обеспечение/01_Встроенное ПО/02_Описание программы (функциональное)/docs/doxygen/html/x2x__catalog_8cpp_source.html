16template<typename DeviceType, X2XDeviceType TypeId>
17X2XDeviceHeader* construct_device(void* storage,
18 size_t storage_size,
19 uint8_t slave_address,
20 uint16_t asdu)
21{
22 static_assert(sizeof(DeviceType) <= X2X_DEVICE_SLOT_BYTES,
23 "X2X device type does not fit static registry slot");
24
25 if ((storage == 0) || (storage_size < sizeof(DeviceType)))
26 {
27 return 0;
28 }
29
30 memset(storage, 0, sizeof(DeviceType));
31 DeviceType* device = new (storage) DeviceType;
32
33 device->type = TypeId;
34 device->asdu = asdu;
35 device->slave_address = slave_address;
36 device->connection_lost = (TypeId == X2X_DEVICE_LCP) ? 0U : 1U;
37 device->device_fault = 0U;
38 device->last_communication_error = X2X_COMMUNICATION_OK;
39
40 return device;
41}

43void print_float_values(const float* values,
44 uint8_t value_count,
45 uint8_t maximum_to_print)
46{
47 const uint8_t print_count =
48 (value_count < maximum_to_print) ? value_count : maximum_to_print;
49
50 SerialUSB.print(" float_values:");
51
52 for (uint8_t index = 0U; index < print_count; ++index)
53 {
54 if ((index % 4U) == 0U)
55 {
56 SerialUSB.print("\r\n ");
57 }
58 else
59 {
60 SerialUSB.print(", ");
61 }
62
63 diagnostic_print_index(index);
64 SerialUSB.print(values[index]);
65 }
66
67 SerialUSB.print("\r\n");
68
69 if (print_count < value_count)
70 {
71 SerialUSB.print(" ");
72 diagnostic_print_assignment("float_values_shown");
73 SerialUSB.print(static_cast<int>(print_count));
74 SerialUSB.print(", ");
75 diagnostic_print_assignment("total");
76 SerialUSB.print(static_cast<int>(value_count));
77 SerialUSB.print("\r\n");
78 }
79}
