1/**
2 * @file field_sensor_service.hpp
3 * @brief Демонстрационный Modbus RTU master на физических портах S1..S4.
4 *
5 * Baseline намеренно жёстко задаёт один пример прибора и запускает его на
6 * каждом порту. Три уровня конфигурации разделены:
7 *
8 * 1. параметры прибора (`slave_address`, FC03 start/count, period, timeout)
9 * задаются в `set_default_configs()` файла field_sensor_service.cpp;
10 * 2. физические baud/parity S1..S4 загружаются в field_serial_config.cpp из
11 * `baud.TXT` и `Parity.TXT`;
12 * 3. привязка S1..S4 к UART/SC16IS находится в board/lcp_field_ports.cpp.
13 *
14 * При добавлении другого прибора не следует изменять ModbusRtuMaster. Создайте
15 * отдельный прикладной service или замените значения `set_default_configs()` и
16 * обработку результата. Если прибор возвращает больше данных, увеличьте
17 * FIELD_SENSOR_REGISTER_COUNT, runtime buffer и опубликованную Ethernet map.
18 *
19 * Один half-duplex-порт имеет ровно одного владельца. S1–S4 с момента setup
20 * принадлежат этому service; диагностический echo на них не запускается.
21 */
22

31/**
32 * Количество holding-регистров фиксированного baseline-прибора.
33 *
34 * При изменении значения одновременно проверьте:
35 * - `set_default_configs()`;
36 * - `FieldSensorPortState::registers`;
37 * - карту `ethernet_modbus_service.cpp`;
38 * - диагностику `field_status.cpp` и `ethernet_status.cpp`.
39 */
40static const uint8_t FIELD_SENSOR_REGISTER_COUNT = 2U;
41
42/**
43 * @brief Роль физического пользовательского порта.
44 *
45 * Один half-duplex-порт в каждый момент времени имеет только одну роль.
46 * Baseline использует MASTER. SLAVE оставлен для отключённого reusable-примера
47 * `field_sensor_slave_example.cpp` и не включается одновременно с master.
48 */
49enum FieldPortRole : uint8_t
50{
51 FIELD_PORT_DISABLED = 0U, /**< Порт не обслуживается FieldSensor. */
52 FIELD_PORT_MASTER = 1U, /**< Активный Modbus RTU master. */
53 FIELD_PORT_SLAVE = 2U /**< Альтернативный slave-режим, не baseline. */
54};
55
56/**
57 * @brief Прикладные параметры одного экземпляра демонстрационного прибора.
58 *
59 * Структура копируется в runtime при initialize_runtime(). Изменение исходного
60 * массива g_configs во время активной транзакции запрещено; reload сначала
61 * ждёт, пока все четыре master станут idle.
62 */
63struct FieldSensorConfig
64{
65 FieldPortRole role; /**< Владелец/роль физического порта. */
66 uint8_t slave_address; /**< Обычный Modbus RTU адрес 1..247. */
67 uint16_t start_register; /**< Первый holding register запроса FC03. */
68 uint16_t register_count; /**< Число регистров; не больше master limit и runtime buffer. */
69 uint32_t poll_period_ms; /**< Период между запусками запросов; 300 мс в baseline. */
70 uint32_t timeout_ms; /**< Срок физического TX и ответа; 500 мс в baseline. */
71 LcpFieldPortConfig serial; /**< Baud и frame физического S1..S4. */
72};
73
74/**
75 * @brief Runtime одного физического FieldSensor-порта.
76 *
77 * Каждый S1..S4 имеет собственный ModbusRtuMaster, буфер, таймер и качество.
78 * Последние корректные `registers` не обнуляются при ошибке. Использовать их
79 * разрешено только при `valid != 0`; после пяти последовательных ошибок
80 * устанавливаются `connection_lost=1` и `valid=0`.
81 */
82struct FieldSensorPortState
83{
84 FieldSensorConfig config; /**< Применённая конфигурация прибора и UART. */
85 ModbusRtuMaster master; /**< Независимый protocol engine этого порта. */
86 uint16_t registers[FIELD_SENSOR_REGISTER_COUNT]; /**< Последние корректные FC03 values. */
87 uint8_t port_present; /**< Физический transport обнаружен и доступен. */
88 uint8_t request_active; /**< Запрос запущен и ещё не завершён. */
89 uint8_t valid; /**< 1, когда последние values имеют допустимое качество. */
90 uint8_t connection_lost; /**< 1 после порога последовательных ошибок. */
91 uint8_t consecutive_failures; /**< Ошибки после последнего успешного ответа. */
92 uint8_t last_exception_code; /**< Последний Modbus exception code. */
93 ModbusRtuResult last_result; /**< Итог последней завершённой транзакции. */
94 uint32_t successful_poll_count; /**< Накопительный счётчик успешных ответов. */
95 uint32_t failed_poll_count; /**< Накопительный счётчик ошибок/start failures. */
96 uint32_t last_update_ms; /**< millis() последнего корректного ответа. */
97 uint32_t next_poll_ms; /**< Абсолютный wrap-safe срок следующего запуска. */
98};
