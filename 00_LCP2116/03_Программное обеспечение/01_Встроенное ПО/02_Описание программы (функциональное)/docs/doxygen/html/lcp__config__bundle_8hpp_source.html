18/**
19 * @brief Версия 1 бинарного payload внутреннего хранилища.
20 *
21 * Поля имеют фиксированную ширину. Reserved заполняется нулями и оставляет
22 * пространство для совместимых дополнений без немедленного увеличения записи.
23 * Выравнивание 4 гарантирует безопасный доступ к uint32_t на Cortex-M3.
24 */
25struct __attribute__((packed, aligned(4))) LcpConfigBundle
26{
27 uint32_t field_baudrate[LCP_FIELD_PORT_COUNT];
28 uint8_t field_parity[LCP_FIELD_PORT_COUNT]; /**< 0=8N1, 1=8O1, 2=8E1. */
29
30 uint8_t ethernet_mac[LCP_ETHERNET_COUNT][6];
31 uint8_t ethernet_ip[LCP_ETHERNET_COUNT][4];
32 uint8_t ethernet_subnet[LCP_ETHERNET_COUNT][4];
33 uint8_t ethernet_gateway[LCP_ETHERNET_COUNT][4];
34
35 uint8_t x2x_module_count;
36 uint8_t x2x_module_id[X2X_MAX_MODULES];
37
38 uint8_t eia_count;
39 int16_t eia_value[LCP_CONFIG_EIA_VALUE_COUNT];
40
41 uint8_t reserved[LCP_CONFIG_RESERVED_BYTES];
42};
