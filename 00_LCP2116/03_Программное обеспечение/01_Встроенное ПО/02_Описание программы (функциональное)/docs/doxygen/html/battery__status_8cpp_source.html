26uint8_t battery_raw_to_ok(uint8_t raw)
27{
28 if (LCP_BACKUP_BATTERY_OK_ACTIVE_HIGH != 0U)

47void battery_status_poll(void)
48{
49 const uint32_t now_ms = millis();
50 const uint8_t raw = lcp_battery_raw();
51
52 if (raw != g_last_raw)
53 {
54 g_last_raw = raw;
55 g_last_change_ms = now_ms;
56 g_stable = 0U;
57 return;
58 }
59
60 if ((g_stable == 0U) &&
61 ((uint32_t)(now_ms - g_last_change_ms) >= BATTERY_DEBOUNCE_MS))
62 {
63 g_stable = 1U;
64 g_debounced_raw = raw;
65 }
66}
