14static const uint32_t X2X_BITS_PER_CHARACTER_WITH_MARGIN = 11U;
15static uint32_t g_tx_not_before_ms = 0U;
16
17size_t x2x_write(const uint8_t* data, size_t length)
18{
19 if (data == 0)
20 {
21 return 0U;
22 }
23
24 size_t written = 0U;
25
26 while (written < length)
27 {
28 if (hal_uart_write(HAL_UART_PORT_0, data[written]) == 0U)
29 {
30 break;
31 }
32
33 ++written;
34 }
35
36 const uint32_t duration_ms = static_cast<uint32_t>(
37 ((written * X2X_BITS_PER_CHARACTER_WITH_MARGIN * 1000U) +
38 (LCP_X2X_BAUDRATE - 1U)) /
39 LCP_X2X_BAUDRATE);
40 g_tx_not_before_ms = millis() + duration_ms + 1U;
41
42 return written;
43}
