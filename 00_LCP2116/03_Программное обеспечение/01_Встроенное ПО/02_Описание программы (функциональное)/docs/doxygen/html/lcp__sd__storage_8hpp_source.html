15/** @brief Результат низкоуровневой операции с microSD. */
16enum LcpSdStorageResult
17{
18 LCP_SD_STORAGE_OK = 0, /**< Операция завершена успешно. */
19 LCP_SD_STORAGE_NO_CARD = 1, /**< Карта не отвечает на SPI. */
20 LCP_SD_STORAGE_CARD_INIT_FAILED = 2, /**< SD initialization sequence не завершена. */
21 LCP_SD_STORAGE_VOLUME_UNSUPPORTED = 3, /**< Раздел или FAT не поддерживается. */
22 LCP_SD_STORAGE_FILE_NOT_FOUND = 4, /**< Файл отсутствует в корневом каталоге. */
23 LCP_SD_STORAGE_FILE_OPEN_FAILED = 5, /**< Directory entry найден, но файл не открыт. */
24 LCP_SD_STORAGE_FILE_EXISTS_ERROR = 6, /**< Не удалось корректно заменить существующий файл. */
25 LCP_SD_STORAGE_DIRECTORY_FULL = 7, /**< В корневом каталоге нет свободной записи. */
26 LCP_SD_STORAGE_NO_FREE_CLUSTER = 8, /**< FAT не содержит свободного кластера. */
27 LCP_SD_STORAGE_READ_FAILED = 9, /**< Ошибка чтения блока или цепочки FAT. */
28 LCP_SD_STORAGE_WRITE_FAILED = 10, /**< Ошибка записи данных, FAT или directory entry. */
29 LCP_SD_STORAGE_INVALID_NAME = 11, /**< Имя не соответствует поддерживаемому FAT 8.3. */
30 LCP_SD_STORAGE_NOT_READY = 12 /**< Файловая система не смонтирована. */
31};
32
33/** @brief Контекст последовательного чтения одного открытого файла. */
34struct LcpSdReadFile
35{
36 uint32_t first_cluster; /**< Первый кластер файла. */
37 uint32_t current_cluster; /**< Текущий кластер цепочки FAT. */
38 uint32_t current_cluster_index; /**< Номер текущего кластера относительно начала файла. */
39 uint32_t file_size; /**< Размер файла в байтах. */
40 uint32_t position; /**< Текущая позиция чтения. */
41 uint32_t loaded_sector_lba; /**< LBA сектора, находящегося в sector[]. */
42 uint8_t sector_loaded; /**< Ненулевое значение, если sector[] валиден. */
43 uint8_t sector[512]; /**< Внутренний секторный кэш открытого файла. */
44};
