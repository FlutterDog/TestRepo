28void x2x_module_mark_failure(X2XDeviceHeader& device,
29 ModbusRtuResult result,
30 uint8_t exception_code)
31{
32 if (device.consecutive_failures < 0xFFU)
33 {
34 ++device.consecutive_failures;
35 }
36
37 ++device.failed_poll_count;
38 device.last_exception_code = exception_code;
39
40 switch (result)
41 {
42 case MODBUS_RTU_RESULT_TIMEOUT:
43 device.last_communication_error = X2X_COMMUNICATION_TIMEOUT;
44 break;
45
46 case MODBUS_RTU_RESULT_CRC_ERROR:
47 device.last_communication_error = X2X_COMMUNICATION_CRC_ERROR;
48 break;
49
50 case MODBUS_RTU_RESULT_EXCEPTION:
51 device.last_communication_error = X2X_COMMUNICATION_EXCEPTION;
52 break;
53
54 case MODBUS_RTU_RESULT_TRANSPORT_ERROR:
55 device.last_communication_error = X2X_COMMUNICATION_TRANSPORT_ERROR;
56 break;
57
58 case MODBUS_RTU_RESULT_INVALID_RESPONSE:
59 case MODBUS_RTU_RESULT_INVALID_ARGUMENT:
60 default:
61 device.last_communication_error = X2X_COMMUNICATION_PROTOCOL_ERROR;
62 break;
63 }
64
65 if (device.consecutive_failures >= X2X_CONNECTION_LOSS_THRESHOLD)
66 {
67 device.connection_lost = 1U;
68 }
69}

71float x2x_module_registers_to_float(uint16_t low_register,
72 uint16_t high_register)
73{
74 const uint32_t raw_value =
75 (static_cast<uint32_t>(high_register) << 16U) |
76 static_cast<uint32_t>(low_register);
77 float result = 0.0F;
78
79 memcpy(&result, &raw_value, sizeof(result));
80 return result;
81}

83void x2x_module_decode_float_array(float* destination,
84 size_t float_count,
85 const uint16_t* registers,
86 size_t start_register,
87 size_t register_count)
88{
89 if ((destination == 0) || (registers == 0))
90 {
91 return;
92 }
93
94 memset(destination, 0, float_count * sizeof(float));
95
96 for (size_t float_index = 0U; float_index < float_count; ++float_index)
97 {
98 const size_t low_index = start_register + (float_index * 2U);
99 const size_t high_index = low_index + 1U;
100
101 if (high_index >= register_count)
102 {
103 break;
104 }
105
106 destination[float_index] =
107 x2x_module_registers_to_float(registers[low_index],
108 registers[high_index]);
109 }
