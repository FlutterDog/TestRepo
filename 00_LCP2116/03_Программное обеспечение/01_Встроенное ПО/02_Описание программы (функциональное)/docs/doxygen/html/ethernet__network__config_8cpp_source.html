27template <size_t OctetCount>
28SdConfigResult load_octets(const char* file_name,
29 uint8_t (&output)[OctetCount])
30{
31 int16_t values[OctetCount + 1U];
32 uint8_t loaded_count = 0U;
33 const SdConfigResult result = sd_config_load_int16(
34 file_name,
35 values,
36 sizeof(values) / sizeof(values[0]),
37 &loaded_count);
38
39 if (result != SD_CONFIG_OK)
40 {
41 return result;
42 }
43
44 if ((loaded_count != OctetCount) ||
45 (values[0] != static_cast<int16_t>(OctetCount)))
46 {
47 return SD_CONFIG_INVALID_COUNT;
48 }
49
50 for (size_t index = 0U; index < OctetCount; ++index)
51 {
52 const int16_t value = values[index + 1U];
53
54 if ((value < 0) || (value > 255))
55 {
56 return SD_CONFIG_INVALID_VALUE;
57 }
58 }
59
60 for (size_t index = 0U; index < OctetCount; ++index)
61 {
62 output[index] = static_cast<uint8_t>(values[index + 1U]);
63 }
64
65 return SD_CONFIG_OK;
66}

92void load_interface(W5500NetworkConfig& config,
93 EthernetNetworkConfigReport& report,
94 const EthernetFileSet& files)
95{
96 W5500MacAddress mac = config.mac;
97 W5500IpAddress ip = config.ip;
98 W5500IpAddress subnet = config.subnet;
99 W5500IpAddress gateway = config.gateway;
100
101 report.mac_result = load_octets(files.mac, mac.octet);
102 report.ip_result = load_octets(files.ip, ip.octet);
103 report.subnet_result = load_octets(files.subnet, subnet.octet);
104 report.gateway_result = load_octets(files.gateway, gateway.octet);
105
106 if (report.mac_result == SD_CONFIG_OK)
107 {
108 config.mac = mac;
109 }
110
111 if (report.ip_result == SD_CONFIG_OK)
112 {
113 config.ip = ip;
114 }
115
116 if (report.subnet_result == SD_CONFIG_OK)
117 {
118 config.subnet = subnet;
119 }
120
121 if (report.gateway_result == SD_CONFIG_OK)
122 {
123 config.gateway = gateway;
124 }
125
126 note_loaded(report, report.mac_result);
127 note_loaded(report, report.ip_result);
128 note_loaded(report, report.subnet_result);
129 note_loaded(report, report.gateway_result);
130}
