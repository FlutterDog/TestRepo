114uint8_t parse_uint32(const char* text, uint32_t* output)
115{
116 uint32_t value = 0U;
117 uint16_t index = 0U;
118 uint8_t digit_found = 0U;
119
120 if ((text == 0) || (output == 0))
121 {
122 return 0U;
123 }
124
125 if (text[index] == '+')
126 {
127 ++index;
128 }
129
130 while (text[index] != '\0')
131 {
132 if ((text[index] < '0') || (text[index] > '9'))
133 {
134 return 0U;
135 }
136
137 const uint32_t digit = static_cast<uint32_t>(text[index] - '0');
138
139 if (value > ((0xFFFFFFFFUL - digit) / 10UL))
140 {
141 return 0U;
142 }
143
144 value = (value * 10UL) + digit;
145 digit_found = 1U;
146 ++index;
147 }
148
149 if (digit_found == 0U)
150 {
151 return 0U;
152 }
153
154 *output = value;
155 return 1U;
156}

158uint8_t parse_int16(const char* text, int16_t* output)
159{
160 uint16_t index = 0U;
161 uint8_t negative = 0U;
162 uint8_t digit_found = 0U;
163 uint32_t magnitude = 0U;
164 uint32_t limit = 32767U;
165
166 if ((text == 0) || (output == 0))
167 {
168 return 0U;
169 }
170
171 if (text[index] == '-')
172 {
173 negative = 1U;
174 limit = 32768U;
175 ++index;
176 }
177 else if (text[index] == '+')
178 {
179 ++index;
180 }
181
182 while (text[index] != '\0')
183 {
184 if ((text[index] < '0') || (text[index] > '9'))
185 {
186 return 0U;
187 }
188
189 const uint32_t digit = static_cast<uint32_t>(text[index] - '0');
190
191 if (magnitude > ((limit - digit) / 10U))
192 {
193 return 0U;
194 }
195
196 magnitude = (magnitude * 10U) + digit;
197 digit_found = 1U;
198 ++index;
199 }
200
201 if (digit_found == 0U)
202 {
203 return 0U;
204 }
205
206 const int32_t signed_value = (negative != 0U) ?
207 -static_cast<int32_t>(magnitude) : static_cast<int32_t>(magnitude);
208 *output = static_cast<int16_t>(signed_value);
209 return 1U;
210}

260template <typename ValueType>
261SdConfigResult load_numeric_file(const char* file_name,
262 ValueType* output,
263 size_t capacity,
264 uint8_t* loaded_count,
265 uint8_t (*parse_value)(const char*, ValueType*))
266{
267 if ((file_name == 0) || (output == 0) || (loaded_count == 0) ||
268 (parse_value == 0) || (capacity < 2U))
269 {
270 return SD_CONFIG_INVALID_COUNT;
271 }
272
273 *loaded_count = 0U;
274
275 for (size_t index = 0U; index < capacity; ++index)
276 {
277 output[index] = static_cast<ValueType>(0);
278 }
279
280 if (lcp_sd_storage_ready() == 0U)
281 {
282 return SD_CONFIG_CARD_NOT_READY;
283 }
284
285 LcpSdReadFile file;
286 const SdConfigResult open_result = translate_open_result(
287 lcp_sd_storage_open_read(file_name, &file));
288
289 if (open_result != SD_CONFIG_OK)
290 {
291 return open_result;
292 }
293
294 uint8_t expected_count = 0U;
295 uint8_t value_index = 0U;
296 uint8_t count_read = 0U;
297 uint8_t line_too_long = 0U;
298
299 while (read_physical_line(&file,
300 g_line_buffer,
301 SD_CONFIG_LINE_BUFFER_SIZE,
302 &line_too_long) != 0U)
303 {
304 if (line_too_long != 0U)
305 {
306 lcp_sd_storage_close_read(&file);
307 return SD_CONFIG_LINE_TOO_LONG;
308 }
309
310 strip_comment_and_trim(g_line_buffer);
311
312 if (g_line_buffer[0] == '\0')
313 {
314 continue;
315 }
316
317 if (is_fin_marker(g_line_buffer) != 0U)
318 {
319 break;
320 }
321
322 if (count_read == 0U)
323 {
324 uint32_t parsed_count = 0U;
325
326 if ((parse_uint32(g_line_buffer, &parsed_count) == 0U) ||
327 (parsed_count == 0U) ||
328 (parsed_count > 255U) ||
329 (parsed_count >= capacity))
330 {
331 lcp_sd_storage_close_read(&file);
332 return SD_CONFIG_INVALID_COUNT;
333 }
334
335 expected_count = static_cast<uint8_t>(parsed_count);
336 output[0] = static_cast<ValueType>(parsed_count);
337 count_read = 1U;
338 continue;
339 }
340
341 if (value_index >= expected_count)
342 {
343 lcp_sd_storage_close_read(&file);
344 return SD_CONFIG_INVALID_COUNT;
345 }
346
347 ValueType parsed_value = static_cast<ValueType>(0);
348
349 if (parse_value(g_line_buffer, &parsed_value) == 0U)
350 {
351 lcp_sd_storage_close_read(&file);
352 return SD_CONFIG_INVALID_VALUE;
353 }
354
355 ++value_index;
356 output[value_index] = parsed_value;
357 }
358
359 lcp_sd_storage_close_read(&file);
360
361 if (count_read == 0U)
362 {
363 return SD_CONFIG_EMPTY_FILE;
364 }
365
366 if (value_index != expected_count)
367 {
368 return SD_CONFIG_INCOMPLETE_FILE;
369 }
370
371 *loaded_count = value_index;
372 return SD_CONFIG_OK;
373}

390uint8_t append_uint(uint8_t* buffer,
391 uint16_t* position,
392 uint16_t capacity,
393 uint16_t value)
394{
395 char temporary[6];
396 uint8_t length = 0U;
397
398 if (value == 0U)
399 {
400 return append_char(buffer, position, capacity, '0');
401 }
402
403 while (value != 0U)
404 {
405 temporary[length++] = static_cast<char>('0' + (value % 10U));
406 value = static_cast<uint16_t>(value / 10U);
407 }
408
409 while (length != 0U)
410 {
411 if (append_char(buffer,
412 position,
413 capacity,
414 temporary[--length]) == 0U)
415 {
416 return 0U;
417 }
418 }
419
420 return 1U;
