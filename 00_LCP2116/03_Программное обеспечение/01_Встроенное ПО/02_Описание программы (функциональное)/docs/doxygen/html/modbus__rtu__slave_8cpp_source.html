26{
27 return ((transport != 0) &&
28 (transport->write != 0) &&

51void append_crc(ModbusRtuSlave& slave, uint16_t payload_length)
52{
53 const uint16_t crc = modbus_rtu_crc16(slave.tx_buffer, payload_length);
54 slave.tx_buffer[payload_length] = static_cast<uint8_t>(crc & 0x00FFU);
55 slave.tx_buffer[payload_length + 1U] = static_cast<uint8_t>(crc >> 8U);
56 slave.tx_length = payload_length + MODBUS_CRC_LENGTH;
57}

59uint8_t request_crc_valid(const ModbusRtuSlave& slave)
60{
61 const uint16_t calculated = modbus_rtu_crc16(
62 slave.rx_buffer,
63 MODBUS_READ_REQUEST_LENGTH - MODBUS_CRC_LENGTH);
64 const uint16_t received =
65 static_cast<uint16_t>(slave.rx_buffer[MODBUS_READ_REQUEST_LENGTH - 2U]) |
66 static_cast<uint16_t>(slave.rx_buffer[MODBUS_READ_REQUEST_LENGTH - 1U] << 8U);
67
68 return (calculated == received) ? 1U : 0U;
69}

71void begin_transmit(ModbusRtuSlave& slave)
72{
73 const size_t written = slave.transport->write(slave.tx_buffer,
74 slave.tx_length);
75
76 if (written != slave.tx_length)
77 {
78 ++slave.error_count;
79 modbus_rtu_slave_reset(slave);
80 return;
81 }
82
83 ++slave.response_count;
84 slave.tx_deadline_ms = millis() + MODBUS_RTU_SLAVE_TX_TIMEOUT_MS;
85 slave.state = MODBUS_RTU_SLAVE_WAIT_TX;
86}

88void send_exception(ModbusRtuSlave& slave,
89 uint8_t function_code,
90 uint8_t exception_code)
91{
92 slave.tx_buffer[0] = slave.slave_address;
93 slave.tx_buffer[1] = static_cast<uint8_t>(function_code | 0x80U);
94 slave.tx_buffer[2] = exception_code;
95 append_crc(slave, 3U);

99void process_request(ModbusRtuSlave& slave)
100{
101 ++slave.request_count;
102
103 if (request_crc_valid(slave) == 0U)
104 {
105 ++slave.error_count;
106 modbus_rtu_slave_reset(slave);
107 return;
108 }
109
110 if (slave.rx_buffer[0] != slave.slave_address)
111 {
112 modbus_rtu_slave_reset(slave);
113 return;
114 }
115
116 const uint8_t function_code = slave.rx_buffer[1];
117
118 if (function_code != MODBUS_FUNCTION_READ_HOLDING)
119 {
120 ++slave.error_count;
121 send_exception(slave,
122 function_code,
123 MODBUS_EXCEPTION_ILLEGAL_FUNCTION);
124 return;
125 }
126
127 const uint16_t start_address =
128 static_cast<uint16_t>(slave.rx_buffer[2] << 8U) |
129 static_cast<uint16_t>(slave.rx_buffer[3]);
130 const uint16_t register_count =
131 static_cast<uint16_t>(slave.rx_buffer[4] << 8U) |
132 static_cast<uint16_t>(slave.rx_buffer[5]);
133
134 if (register_count == 0U)
135 {
136 ++slave.error_count;
137 send_exception(slave,
138 function_code,
139 MODBUS_EXCEPTION_ILLEGAL_VALUE);
140 return;
141 }
142
143 if ((start_address >= slave.holding_register_count) ||
144 (register_count >
145 static_cast<uint16_t>(slave.holding_register_count - start_address)))
146 {
147 ++slave.error_count;
148 send_exception(slave,
149 function_code,
150 MODBUS_EXCEPTION_ILLEGAL_ADDRESS);
151 return;
152 }
153
154 if (register_count > MODBUS_MAX_RESPONSE_REGISTERS)
155 {
156 ++slave.error_count;
157 send_exception(slave,
158 function_code,
159 MODBUS_EXCEPTION_ILLEGAL_VALUE);
160 return;
161 }
162
163 slave.tx_buffer[0] = slave.slave_address;
164 slave.tx_buffer[1] = MODBUS_FUNCTION_READ_HOLDING;
165 slave.tx_buffer[2] = static_cast<uint8_t>(register_count * 2U);
166
167 for (uint16_t index = 0U; index < register_count; ++index)
168 {
169 const uint16_t value = slave.holding_registers[start_address + index];
170 const uint16_t byte_index = 3U + (index * 2U);
171 slave.tx_buffer[byte_index] = static_cast<uint8_t>(value >> 8U);
172 slave.tx_buffer[byte_index + 1U] =
173 static_cast<uint8_t>(value & 0x00FFU);
174 }
175
176 append_crc(slave,
177 static_cast<uint16_t>(3U + (register_count * 2U)));
178 begin_transmit(slave);
179}

218void modbus_rtu_slave_poll(ModbusRtuSlave& slave)
219{
220 if ((transport_valid(slave.transport) == 0U) ||
221 (slave_address_valid(slave.slave_address) == 0U) ||
222 (slave.holding_registers == 0) ||
223 (slave.holding_register_count == 0U) ||
224 (slave.interframe_gap_ms == 0U))
225 {
226 return;
227 }
228
229 const uint32_t now_ms = millis();
230
231 if (slave.state == MODBUS_RTU_SLAVE_WAIT_TX)
232 {
233 if (slave.transport->tx_idle() != 0U)
234 {
235 modbus_rtu_slave_reset(slave);
236 return;
237 }
238
239 if (deadline_reached(now_ms, slave.tx_deadline_ms) != 0U)
240 {
241 ++slave.error_count;
242 modbus_rtu_slave_reset(slave);
243 }
244
245 return;
246 }
247
248 /* Цикл читает только уже доступные байты и ограничен RX capacity. */
249 while (slave.transport->available() > 0U)
250 {
251 if (slave.rx_length >= MODBUS_RTU_SLAVE_RX_CAPACITY)
252 {
253 ++slave.error_count;
254 modbus_rtu_slave_reset(slave);
255 return;
256 }
257
258 const int value = slave.transport->read();
259
260 if (value < 0)
261 {
262 break;
263 }
264
265 slave.rx_buffer[slave.rx_length] = static_cast<uint8_t>(value);
266 ++slave.rx_length;
267 slave.last_rx_ms = now_ms;
268 }
269
270 if (slave.rx_length == MODBUS_READ_REQUEST_LENGTH)
271 {
272 process_request(slave);
273 return;
274 }
275
276 if (slave.rx_length > MODBUS_READ_REQUEST_LENGTH)
277 {
278 ++slave.error_count;
279 modbus_rtu_slave_reset(slave);
280 return;
281 }
282
283 if ((slave.rx_length != 0U) &&
284 (gap_elapsed(now_ms,
285 slave.last_rx_ms,
286 slave.interframe_gap_ms) != 0U))
287 {
288 ++slave.error_count;
289 modbus_rtu_slave_reset(slave);
290 }
