120void field_status_print_report(void)
121{
122 diagnostic_print_section("FIELDSENSOR S1..S4");
123
124 diagnostic_print_assignment("service");
125 SerialUSB.print((field_sensor_service_paused() != 0U) ?
126 "paused" : "running");
127 SerialUSB.print(", ");
128 diagnostic_print_assignment("config_reload");
129 SerialUSB.print((field_sensor_service_config_reload_pending() != 0U) ?
130 "pending" : "idle");
131 SerialUSB.print("\r\n");
132
133 diagnostic_print_assignment("device_model");
134 SerialUSB.print("hardcoded baseline example\r\n");
135 diagnostic_print_assignment("device_config_location");
136 SerialUSB.print("field_sensor_service.cpp\r\n");
137
138 diagnostic_print_assignment("serial_config_source");
139 SerialUSB.print(lcp_config_source_text(lcp_config_service_source()));
140 SerialUSB.print(", ");
141 diagnostic_print_assignment("config_generation");
142 SerialUSB.print(static_cast<unsigned long>(
143 lcp_config_service_generation()));
144 SerialUSB.print("\r\n");
145
146 for (uint8_t index = 0U;
147 index < field_sensor_service_port_count();
148 ++index)
149 {
150 print_port(static_cast<LcpFieldPortId>(index));
151 }
152}

154uint8_t field_status_handle_command(const char* command)
155{
156 if (command == 0)
157 {
158 return 0U;
159 }
160
161 if ((strcmp(command, "field") == 0) ||
162 (strcmp(command, "field status") == 0))
163 {
164 field_status_print_report();
165 return 1U;
166 }
167
168 if (strcmp(command, "field pause") == 0)
169 {
170 field_sensor_service_pause();
171 SerialUSB.print("FieldSensor pause requested\r\n");
172 SerialUSB.print("active requests may finish; no new requests start\r\n");
173 return 1U;
174 }
175
176 if (strcmp(command, "field resume") == 0)
177 {
178 field_sensor_service_resume();
179 SerialUSB.print("FieldSensor polling = running\r\n");
180 return 1U;
181 }
182
183 if (strcmp(command, "field reload") == 0)
184 {
185 field_sensor_service_request_config_reload();
186 SerialUSB.print("FieldSensor active configuration reload = pending\r\n");
187 return 1U;
188 }
189
190 return 0U;
191}
