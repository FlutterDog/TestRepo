44/*
45 * Both LCT firmware projects use IGAS_485 with MAX_READ_REGS = 16 and
46 * MAX_MESSAGE_LENGTH = 40. A 16-register response occupies 37 bytes
47 * including CRC and is the largest response that safely fits the slave
48 * packet buffer. Main data and waveforms must therefore be read in chunks
49 * of no more than 16 registers even though the LCP master supports 100.
50 */
51static const uint16_t LCT_SLAVE_MAX_READ_REGISTERS = 16U;
52
53static_assert(LCT_SLAVE_MAX_READ_REGISTERS <=
54 MODBUS_RTU_MASTER_MAX_READ_REGISTERS,
55 "LCT chunk size exceeds Modbus master read capacity");
56
57/*
58 * LCT1_r1_b/modbusProceed:
59 * 0..1 - contact/status words;
60 * 2..85 - 42 float values;
61 * 200..395, 400..595, 600..795 - 196 waveform samples per phase.
62 */
63const LctProfile LCT1114_PROFILE =
64{
65 X2X_DEVICE_LCT1114,
66 86U,
67 196U,
68 { 200U, 400U, 600U },
69 0U
70};
71
72/*
73 * LCT2_r8_p330/modbusProceed:
74 * 0..1 - contact/status words;
75 * 2..91 - 45 float values;
76 * 92..93 - raw and previous contact-state words;
77 * 3000..3399, 3400..3799, 3800..4199 - 400 samples per phase.
78 */
79const LctProfile LCT1114_2_PROFILE =
80{
81 X2X_DEVICE_LCT1114_2,
82 94U,
83 400U,
84 { 3000U, 3400U, 3800U },
85 1000U
86};

88uint16_t minimum_u16(uint16_t left, uint16_t right)
89{
90 return (left < right) ? left : right;

93uint16_t lct_chunk_count(uint16_t remaining)
94{
95 return minimum_u16(remaining, LCT_SLAVE_MAX_READ_REGISTERS);
96}

117void decode_lct_main(X2XDeviceHeader& device,
118 const LctProfile& profile,
119 const uint16_t* registers)
120{
121 const uint32_t digital_inputs =
122 static_cast<uint32_t>(registers[0] & 0x003FU) |
123 (static_cast<uint32_t>(registers[1] & 0x1FFFU) << 6U);
124
125 if (profile.type == X2X_DEVICE_LCT1114)
126 {
127 X2XLct1114& lct = static_cast<X2XLct1114&>(device);
128 lct.digital_inputs = digital_inputs;
129 x2x_module_decode_float_array(lct.tf_values,
130 X2XLct1114::TF_COUNT,
131 registers,
132 LCT_FLOAT_START_REGISTER,
133 profile.main_register_count);
134 }
135 else
136 {
137 X2XLct1114_2& lct = static_cast<X2XLct1114_2&>(device);
138 lct.digital_inputs = digital_inputs;
139
140 /*
141 * Registers 2..91 contain exactly 45 floats. Registers 92 and 93
142 * are independent uint16_t contact-state words and must not be
143 * interpreted as a synthetic 46th float.
144 */
145 x2x_module_decode_float_array(lct.tf_values,
146 X2XLct1114_2::TF_COUNT,
147 registers,
148 LCT_FLOAT_START_REGISTER,
149 LCT2_CONTACTS_STATE_RAW_REGISTER);
150 lct.contacts_state_raw =
151 registers[LCT2_CONTACTS_STATE_RAW_REGISTER];
152 lct.previous_contacts_state =
153 registers[LCT2_PREVIOUS_CONTACTS_STATE_REGISTER];
154 }
155}

157X2XModulePollResult poll_lct(X2XDeviceHeader* device,
158 X2XModuleContext& context,
159 const LctProfile& profile)
160{
161 if ((device == 0) ||
162 (device->type != profile.type) ||
163 (context.master == 0) ||
164 (context.runtime == 0) ||
165 (context.waveform == 0) ||
166 (context.register_buffer == 0) ||
167 (context.register_capacity < profile.main_register_count) ||
168 (profile.waveform_register_count > X2X_MAX_WAVEFORM_SAMPLES))
169 {
170 return X2X_MODULE_POLL_CYCLE_COMPLETE;
171 }
172
173 switch (context.runtime->state)
174 {
175 case LCT_POLL_START_MAIN:
176 {
177 if (modbus_rtu_master_ready(*context.master) == 0U)
178 {
179 return X2X_MODULE_POLL_IN_PROGRESS;
180 }
181
182 if (context.runtime->transfer_offset >=
183 profile.main_register_count)
184 {
185 context.runtime->transfer_offset = 0U;
186 }
187
188 const uint16_t remaining =
189 profile.main_register_count -
190 context.runtime->transfer_offset;
191 const uint16_t chunk_count = lct_chunk_count(remaining);
192 const uint16_t start_address =
193 static_cast<uint16_t>(LCT_MAIN_START_REGISTER +
194 context.runtime->transfer_offset);
195
196 if ((chunk_count == 0U) ||
197 (modbus_rtu_master_start_read_holding(
198 *context.master,
199 device->slave_address,
200 start_address,
201 chunk_count,
202 &context.register_buffer[context.runtime->transfer_offset],
203 X2X_MODBUS_TRANSACTION_TIMEOUT_MS) == 0U))
204 {
205 return fail_cycle(*device,
206 context,
207 modbus_rtu_master_result(*context.master));
208 }
209
210 context.runtime->state = LCT_POLL_WAIT_MAIN;
211 return X2X_MODULE_POLL_IN_PROGRESS;
212 }
213
214 case LCT_POLL_WAIT_MAIN:
215 {
216 const ModbusRtuResult result =
217 modbus_rtu_master_result(*context.master);
218
219 if (result == MODBUS_RTU_RESULT_BUSY)
220 {
221 return X2X_MODULE_POLL_IN_PROGRESS;
222 }
223
224 if (result != MODBUS_RTU_RESULT_OK)
225 {
226 return fail_cycle(*device, context, result);
227 }
228
229 const uint16_t remaining =
230 profile.main_register_count -
231 context.runtime->transfer_offset;
232 const uint16_t chunk_count = lct_chunk_count(remaining);
233
234 context.runtime->transfer_offset =
235 static_cast<uint16_t>(context.runtime->transfer_offset +
236 chunk_count);
237 modbus_rtu_master_reset(*context.master);
238
239 if (context.runtime->transfer_offset <
240 profile.main_register_count)
241 {
242 context.runtime->state = LCT_POLL_START_MAIN;
243 return X2X_MODULE_POLL_IN_PROGRESS;
244 }
245
246 context.runtime->transfer_offset = 0U;
247 decode_lct_main(*device, profile, context.register_buffer);
248 x2x_module_mark_success(*device, context.now_ms);
249
250 if ((profile.post_capture_delay_ms != 0U) &&
251 (static_cast<int32_t>(context.now_ms -
252 context.runtime->next_waveform_check_ms) < 0))
253 {
254 context.runtime->state = LCT_POLL_START_MAIN;
255 return X2X_MODULE_POLL_CYCLE_COMPLETE;
256 }
257
258 context.runtime->state = LCT_POLL_START_WAVEFORM_FLAG;
259 return X2X_MODULE_POLL_IN_PROGRESS;
260 }
261
262 case LCT_POLL_START_WAVEFORM_FLAG:
263 if (modbus_rtu_master_ready(*context.master) == 0U)
264 {
265 return X2X_MODULE_POLL_IN_PROGRESS;
266 }
267
268 if (modbus_rtu_master_start_read_holding(*context.master,
269 device->slave_address,
270 LCT_WAVEFORM_FLAG_REGISTER,
271 1U,
272 context.register_buffer,
273 X2X_MODBUS_TRANSACTION_TIMEOUT_MS) == 0U)
274 {
275 return fail_cycle(*device,
276 context,
277 modbus_rtu_master_result(*context.master));
278 }
279
280 context.runtime->state = LCT_POLL_WAIT_WAVEFORM_FLAG;
281 return X2X_MODULE_POLL_IN_PROGRESS;
282
283 case LCT_POLL_WAIT_WAVEFORM_FLAG:
284 {
285 const ModbusRtuResult result =
286 modbus_rtu_master_result(*context.master);
287
288 if (result == MODBUS_RTU_RESULT_BUSY)
289 {
290 return X2X_MODULE_POLL_IN_PROGRESS;
291 }
292
293 if (result != MODBUS_RTU_RESULT_OK)
294 {
295 return fail_cycle(*device, context, result);
296 }
297
298 const uint16_t waveform_flag = context.register_buffer[0];
299 modbus_rtu_master_reset(*context.master);
300
301 if (waveform_flag != LCT_WAVEFORM_FLAG_NEW_DATA)
302 {
303 context.runtime->state = LCT_POLL_START_MAIN;
304 return X2X_MODULE_POLL_CYCLE_COMPLETE;
305 }
306
307 memset(context.waveform->phase, 0, sizeof(context.waveform->phase));
308 context.waveform->valid = 0U;
309 context.waveform->owner_slave_address = device->slave_address;
310 context.waveform->samples_per_phase =
311 profile.waveform_register_count;
312 context.runtime->waveform_phase = 0U;
313 context.runtime->transfer_offset = 0U;
314 context.runtime->state = LCT_POLL_START_WAVEFORM_CHUNK;
315 return X2X_MODULE_POLL_IN_PROGRESS;
316 }
317
318 case LCT_POLL_START_WAVEFORM_CHUNK:
319 {
320 if (context.runtime->waveform_phase >= 3U)
321 {
322 context.waveform->valid = 1U;
323 ++context.waveform->sequence;
324 context.runtime->state = LCT_POLL_START_WAVEFORM_RESET;
325 return X2X_MODULE_POLL_IN_PROGRESS;
326 }
327
328 if (modbus_rtu_master_ready(*context.master) == 0U)
329 {
330 return X2X_MODULE_POLL_IN_PROGRESS;
331 }
332
333 const uint16_t remaining =
334 profile.waveform_register_count -
335 context.runtime->transfer_offset;
336 const uint16_t chunk_count = lct_chunk_count(remaining);
337 const uint16_t start_address =
338 static_cast<uint16_t>(
339 profile.waveform_start_register[
340 context.runtime->waveform_phase] +
341 context.runtime->transfer_offset);
342
343 if ((chunk_count == 0U) ||
344 (modbus_rtu_master_start_read_holding(
345 *context.master,
346 device->slave_address,
347 start_address,
348 chunk_count,
349 context.register_buffer,
350 X2X_MODBUS_TRANSACTION_TIMEOUT_MS) == 0U))
351 {
352 return fail_cycle(*device,
353 context,
354 modbus_rtu_master_result(*context.master));
355 }
356
357 context.runtime->state = LCT_POLL_WAIT_WAVEFORM_CHUNK;
358 return X2X_MODULE_POLL_IN_PROGRESS;
359 }
360
361 case LCT_POLL_WAIT_WAVEFORM_CHUNK:
362 {
363 const ModbusRtuResult result =
364 modbus_rtu_master_result(*context.master);
365
366 if (result == MODBUS_RTU_RESULT_BUSY)
367 {
368 return X2X_MODULE_POLL_IN_PROGRESS;
369 }
370
371 if (result != MODBUS_RTU_RESULT_OK)
372 {
373 context.waveform->valid = 0U;
374 return fail_cycle(*device, context, result);
375 }
376
377 const uint16_t remaining =
378 profile.waveform_register_count -
379 context.runtime->transfer_offset;
380 const uint16_t chunk_count = lct_chunk_count(remaining);
381
382 memcpy(&context.waveform->phase[
383 context.runtime->waveform_phase]
384 [context.runtime->transfer_offset],
385 context.register_buffer,
386 static_cast<size_t>(chunk_count) * sizeof(uint16_t));
387
388 context.runtime->transfer_offset =
389 static_cast<uint16_t>(context.runtime->transfer_offset +
390 chunk_count);
391 modbus_rtu_master_reset(*context.master);
392
393 if (context.runtime->transfer_offset >=
394 profile.waveform_register_count)
395 {
396 context.runtime->transfer_offset = 0U;
397 ++context.runtime->waveform_phase;
398 }
399
400 context.runtime->state = LCT_POLL_START_WAVEFORM_CHUNK;
401 return X2X_MODULE_POLL_IN_PROGRESS;
402 }
403
404 case LCT_POLL_START_WAVEFORM_RESET:
405 {
406 if (modbus_rtu_master_ready(*context.master) == 0U)
407 {
408 return X2X_MODULE_POLL_IN_PROGRESS;
409 }
410
411 const uint16_t reset_value = LCT_WAVEFORM_FLAG_RESET_VALUE;
412
413 if (modbus_rtu_master_start_write_multiple(
414 *context.master,
415 device->slave_address,
416 LCT_WAVEFORM_FLAG_REGISTER,
417 &reset_value,
418 1U,
419 X2X_MODBUS_TRANSACTION_TIMEOUT_MS) == 0U)
420 {
421 return fail_cycle(*device,
422 context,
423 modbus_rtu_master_result(*context.master));
424 }
425
426 context.runtime->state = LCT_POLL_WAIT_WAVEFORM_RESET;
427 return X2X_MODULE_POLL_IN_PROGRESS;
428 }
429
430 case LCT_POLL_WAIT_WAVEFORM_RESET:
431 {
432 const ModbusRtuResult result =
433 modbus_rtu_master_result(*context.master);
434
435 if (result == MODBUS_RTU_RESULT_BUSY)
436 {
437 return X2X_MODULE_POLL_IN_PROGRESS;
438 }
439
440 if (result != MODBUS_RTU_RESULT_OK)
441 {
442 context.waveform->valid = 0U;
443 return fail_cycle(*device, context, result);
444 }
445
446 if (profile.post_capture_delay_ms != 0U)
447 {
448 context.runtime->next_waveform_check_ms =
449 context.now_ms + profile.post_capture_delay_ms;
450 }
451
452 reset_cycle(*context.master, *context.runtime);
453 return X2X_MODULE_POLL_CYCLE_COMPLETE;
454 }
455
456 default:
457 context.waveform->valid = 0U;
458 reset_cycle(*context.master, *context.runtime);
459 return X2X_MODULE_POLL_CYCLE_COMPLETE;
460 }
461}
