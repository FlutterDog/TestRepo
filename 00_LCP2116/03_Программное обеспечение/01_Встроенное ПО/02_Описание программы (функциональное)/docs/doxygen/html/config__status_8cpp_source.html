27void print_hex32(uint32_t value)
28{
29 SerialUSB.print("0x");
30
31 for (int8_t shift = 28; shift >= 0; shift -= 4)
32 {
33 const uint8_t nibble = static_cast<uint8_t>((value >> shift) & 0x0FU);
34 SerialUSB.print(HEX_DIGITS[nibble]);
35 }
36}

240void print_candidate_difference(void)
241{
242 const LcpConfigBundle* candidate = lcp_config_service_candidate();
243
244 if (candidate == 0)
245 {
246 return;
247 }
248
249 const LcpConfigBundle& active = lcp_config_service_active();
250 uint8_t any = 0U;
251
252 SerialUSB.print("changed_files = ");
253
254 if (section_changed(active.field_baudrate,
255 candidate->field_baudrate,
256 sizeof(active.field_baudrate)) != 0U)
257 {
258 print_changed_name("baud.TXT", &any);
259 }
260
261 if (section_changed(active.field_parity,
262 candidate->field_parity,
263 sizeof(active.field_parity)) != 0U)
264 {
265 print_changed_name("Parity.TXT", &any);
266 }
267
268 static const char* names[LCP_ETHERNET_COUNT][4] =
269 {
270 { "MAC.txt", "IP.txt", "SUBNET.txt", "GATE.txt" },
271 { "MAC2.txt", "IP2.txt", "SUBNET2.txt", "GATE2.txt" }
272 };
273
274 for (uint8_t index = 0U; index < LCP_ETHERNET_COUNT; ++index)
275 {
276 if (section_changed(active.ethernet_mac[index],
277 candidate->ethernet_mac[index], 6U) != 0U)
278 {
279 print_changed_name(names[index][0], &any);
280 }
281
282 if (section_changed(active.ethernet_ip[index],
283 candidate->ethernet_ip[index], 4U) != 0U)
284 {
285 print_changed_name(names[index][1], &any);
286 }
287
288 if (section_changed(active.ethernet_subnet[index],
289 candidate->ethernet_subnet[index], 4U) != 0U)
290 {
291 print_changed_name(names[index][2], &any);
292 }
293
294 if (section_changed(active.ethernet_gateway[index],
295 candidate->ethernet_gateway[index], 4U) != 0U)
296 {
297 print_changed_name(names[index][3], &any);
298 }
299 }
300
301 if ((active.x2x_module_count != candidate->x2x_module_count) ||
302 (section_changed(active.x2x_module_id,
303 candidate->x2x_module_id,
304 sizeof(active.x2x_module_id)) != 0U))
305 {
306 print_changed_name("X2X.TXT", &any);
307 }
308
309 if ((active.eia_count != candidate->eia_count) ||
310 (section_changed(active.eia_value,
311 candidate->eia_value,
312 sizeof(active.eia_value)) != 0U))
313 {
314 print_changed_name("EIA.TXT", &any);
315 }
316
317 if (any == 0U)
318 {
319 SerialUSB.print("none");
320 }
321
322 SerialUSB.print("\r\n");
323}

420uint8_t config_status_handle_command(const char* command)
421{
422 if ((command == 0) ||
423 (strncmp(command, "config", 6U) != 0))
424 {
425 return 0U;
426 }
427
428 if ((text_equals(command, "config") != 0U) ||
429 (text_equals(command, "config status") != 0U))
430 {
431 diagnostic_print_banner("CONFIGURATION STATUS");
432 config_status_print_report();
433 return 1U;
434 }
435
436 if ((text_equals(command, "config show") != 0U) ||
437 (text_equals(command, "config show all") != 0U))
438 {
439 diagnostic_print_banner("ACTIVE CONFIGURATION");
440 print_all_files(lcp_config_service_active());
441 return 1U;
442 }
443
444 static const char SHOW_PREFIX[] = "config show ";
445
446 if (strncmp(command, SHOW_PREFIX, sizeof(SHOW_PREFIX) - 1U) == 0)
447 {
448 diagnostic_print_banner("ACTIVE CONFIGURATION FILE");
449
450 if (print_named_file(lcp_config_service_active(),
451 command + sizeof(SHOW_PREFIX) - 1U) == 0U)
452 {
453 SerialUSB.print("unknown configuration file\r\n");
454 }
455
456 return 1U;
457 }
458
459 if (text_equals(command, "config defaults") != 0U)
460 {
461 print_default_bundle();
462 return 1U;
463 }
464
465 if (text_equals(command, "config scan sd") != 0U)
466 {
467 print_operation_start(lcp_config_service_request_scan_sd());
468 return 1U;
469 }
470
471 if (text_equals(command, "config compare sd") != 0U)
472 {
473 print_operation_start(lcp_config_service_request_compare_sd());
474 return 1U;
475 }
476
477 if (text_equals(command, "config import sd") != 0U)
478 {
479 print_operation_start(lcp_config_service_request_import_sd());
480 return 1U;
481 }
482
483 if (text_equals(command, "config rollback") != 0U)
484 {
485 print_operation_start(lcp_config_service_request_rollback());
486 return 1U;
487 }
488
489 if (text_equals(command, "config erase confirm") != 0U)
490 {
491 print_operation_start(lcp_config_service_request_erase());
492 return 1U;
493 }
494
495 if (text_equals(command, "config erase") != 0U)
496 {
497 SerialUSB.print("configuration erase not started\r\n");
498 SerialUSB.print("use 'config erase confirm' to invalidate both slots\r\n");
499 return 1U;
500 }
501
502 SerialUSB.print("unknown config command; type 'help'\r\n");
503 return 1U;
504}
