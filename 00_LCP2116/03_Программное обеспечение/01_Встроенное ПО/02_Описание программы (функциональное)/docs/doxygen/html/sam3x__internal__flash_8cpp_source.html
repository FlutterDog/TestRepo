64/**
65 * Выполняется полностью из SRAM. Во время EWP запрещены interrupts, чтобы
66 * обработчик не попытался выполнить код или прочитать константу из Flash1.
67 */
68__attribute__((section(".ramfunc"), noinline))
69FlashCommandResult erase_write_page_from_ram(uintptr_t page_address,
70 const uint32_t* source_words)
71{
72 FlashCommandResult result = { 0U, 0U };
73 const uint32_t previous_primask = __get_PRIMASK();
74 __disable_irq();
75
76 const uint32_t previous_mode = EFC1->EEFC_FMR;
77 EFC1->EEFC_FMR = (previous_mode & ~EEFC_FMR_FWS_Msk) |
78 EEFC_FMR_FWS(PROGRAMMING_FWS);
79
80 uint32_t wait_count = READY_WAIT_LIMIT;
81
82 while (((EFC1->EEFC_FSR & EEFC_FSR_FRDY) == 0U) &&
83 (wait_count != 0U))
84 {
85 --wait_count;
86 }
87
88 if (wait_count == 0U)
89 {
90 result.timed_out = 1U;
91 EFC1->EEFC_FMR = previous_mode;
92
93 if (previous_primask == 0U)
94 {
95 __enable_irq();
96 }
97
98 return result;
99 }
100
101 volatile uint32_t* destination =
102 reinterpret_cast<volatile uint32_t*>(page_address);
103
104 for (uint32_t word = 0U;
105 word < (SAM3X_INTERNAL_FLASH_PAGE_SIZE / sizeof(uint32_t));
106 ++word)
107 {
108 destination[word] = source_words[word];
109 }
110
111 const uint32_t page_number = static_cast<uint32_t>(
112 (page_address - FLASH1_ORIGIN) / SAM3X_INTERNAL_FLASH_PAGE_SIZE);
113
114 __DSB();
115 EFC1->EEFC_FCR = EEFC_FCR_FKEY_PASSWD |
116 EEFC_FCR_FARG(page_number) |
117 EEFC_FCR_FCMD(EEFC_COMMAND_EWP);
118
119 wait_count = READY_WAIT_LIMIT;
120
121 while (((EFC1->EEFC_FSR & EEFC_FSR_FRDY) == 0U) &&
122 (wait_count != 0U))
123 {
124 --wait_count;
125 }
126
127 if (wait_count == 0U)
128 {
129 result.timed_out = 1U;
130 }
131 else
132 {
133 result.status = EFC1->EEFC_FSR;
134 }
135
136 EFC1->EEFC_FMR = previous_mode;
137 __DSB();
138 __ISB();
139
140 if (previous_primask == 0U)
141 {
142 __enable_irq();
143 }
144
145 return result;
146}

207Sam3xInternalFlashResult sam3x_internal_flash_erase_write_page(
208 uintptr_t page_address,
209 const uint8_t page[SAM3X_INTERNAL_FLASH_PAGE_SIZE])
210{
211 if (page == 0)
212 {
213 return SAM3X_INTERNAL_FLASH_INVALID_ARGUMENT;
214 }
215
216 if (((page_address & (SAM3X_INTERNAL_FLASH_PAGE_SIZE - 1U)) != 0U) ||
217 (page_address < FLASH1_ORIGIN) ||
218 (page_address >= FLASH1_END) ||
219 (range_valid(page_address, SAM3X_INTERNAL_FLASH_PAGE_SIZE) == 0U))
220 {
221 return SAM3X_INTERNAL_FLASH_OUT_OF_RANGE;
222 }
223
224 const uint32_t page_number = static_cast<uint32_t>(
225 (page_address - FLASH1_ORIGIN) / SAM3X_INTERNAL_FLASH_PAGE_SIZE);
226
227 if (page_number >= FLASH1_PAGE_COUNT)
228 {
229 return SAM3X_INTERNAL_FLASH_OUT_OF_RANGE;
230 }
231
232 uint32_t aligned_page[SAM3X_INTERNAL_FLASH_PAGE_SIZE / sizeof(uint32_t)];
233 memcpy(aligned_page, page, SAM3X_INTERNAL_FLASH_PAGE_SIZE);
234
235 const FlashCommandResult command =
236 erase_write_page_from_ram(page_address, aligned_page);
237
238 if (command.timed_out != 0U)
239 {
240 return SAM3X_INTERNAL_FLASH_NOT_READY;
241 }
242
243 if ((command.status & EEFC_FSR_FLOCKE) != 0U)
244 {
245 return SAM3X_INTERNAL_FLASH_LOCK_ERROR;
246 }
247
248 if ((command.status & EEFC_FSR_FCMDE) != 0U)
249 {
250 return SAM3X_INTERNAL_FLASH_COMMAND_ERROR;
251 }
252
253 if (memcmp(reinterpret_cast<const void*>(page_address),
254 aligned_page,
255 SAM3X_INTERNAL_FLASH_PAGE_SIZE) != 0)
256 {
257 return SAM3X_INTERNAL_FLASH_VERIFY_ERROR;
258 }
259
260 return SAM3X_INTERNAL_FLASH_OK;
261}
