28/**
29 * @brief Полное описание одного поддерживаемого типа X2X.
30 *
31 * Каталог является единой точкой регистрации: валидатор конфигурации, реестр,
32 * диспетчер опроса и диагностика используют одну запись. Добавление типа не
33 * требует нового `switch` в X2X service или service console.
34 */
35struct X2XModuleDescriptor
36{
37 X2XDeviceType type; /**< Стабильный числовой ID X2X.TXT. */
38 const char* name; /**< Краткое неизменяемое имя типа. */
39 size_t object_size; /**< sizeof конкретной runtime-структуры. */
40 uint16_t sp_count; /**< Число single-point значений. */
41 uint16_t me_count; /**< Число integer measured values. */
42 uint16_t tf_count; /**< Число float measured values. */
43 uint8_t allowed_in_x2x_config; /**< Разрешён ли ID во внешней цепочке. */
44 X2XModuleConstructFunction construct; /**< Конструктор экземпляра в слоте. */
45 X2XModulePollFunction poll; /**< Неблокирующий драйвер типа. */
46 X2XModulePrintFunction print; /**< Специфическая диагностика типа. */
47};
