39void update_holding_map(void)
40{
41 memset(g_holding, 0, sizeof(g_holding));
42 const uint8_t service_paused = field_sensor_service_paused();
43
44 for (uint8_t port_index = 0U;
45 port_index < field_sensor_service_port_count();
46 ++port_index)
47 {
48 const FieldSensorPortState& port = field_sensor_service_port(
49 static_cast<LcpFieldPortId>(port_index));
50 const uint16_t base = static_cast<uint16_t>(
51 port_index * LCP_MODBUS_TCP_REGISTERS_PER_FIELD_PORT);
52 uint16_t quality = 0U;
53
54 g_holding[base] = port.registers[0];
55 g_holding[base + 1U] = port.registers[1];
56
57 if (port.valid != 0U)
58 {
59 quality |= FIELD_SENSOR_TCP_QUALITY_VALID;
60 }
61
62 if (port.connection_lost != 0U)
63 {
64 quality |= FIELD_SENSOR_TCP_QUALITY_CONNECTION_LOST;
65 }
66
67 if (port.port_present != 0U)
68 {
69 quality |= FIELD_SENSOR_TCP_QUALITY_PORT_PRESENT;
70 }
71
72 if (port.request_active != 0U)
73 {
74 quality |= FIELD_SENSOR_TCP_QUALITY_REQUEST_ACTIVE;
75 }
76
77 if (service_paused != 0U)
78 {
79 quality |= FIELD_SENSOR_TCP_QUALITY_SERVICE_PAUSED;
80 }
81
82 quality |= static_cast<uint16_t>(
83 static_cast<uint16_t>(port.last_result) << 8U);
84 g_holding[base + 2U] = quality;
85 }
86}

88uint8_t read_holding(void*,
89 uint16_t start_address,
90 uint16_t register_count,
91 uint16_t* output)
92{
93 if ((output == 0) ||
94 (start_address >= LCP_MODBUS_TCP_HOLDING_COUNT) ||
95 (register_count > static_cast<uint16_t>(
96 LCP_MODBUS_TCP_HOLDING_COUNT - start_address)))
97 {
98 return MODBUS_EXCEPTION_ILLEGAL_ADDRESS;
99 }
100
101 for (uint16_t index = 0U; index < register_count; ++index)
102 {
103 output[index] = g_holding[start_address + index];
104 }
105
106 return 0U;
107}

120void apply_configuration(void)
121{
122 W5500NetworkConfig configs[LCP_ETHERNET_COUNT];
123 EthernetNetworkConfigReport reports[LCP_ETHERNET_COUNT];
124
125 lcp_config_bundle_ethernet(lcp_config_service_active(), configs);
126 prepare_internal_config_report(reports);
127 lcp_ethernet_init_pins();
128
129 for (uint8_t index = 0U; index < LCP_ETHERNET_COUNT; ++index)
130 {
131 EthernetModbusInterfaceState& interface_state = g_interfaces[index];
132 interface_state.config = configs[index];
133 interface_state.config_report = reports[index];
134 modbus_tcp_server_init(interface_state.server);
135 interface_state.transport_error_count = 0U;
136 interface_state.initialized = 1U;
137 interface_state.init_ok = w5500_lite_begin(
138 static_cast<LcpEthernetId>(index),
139 interface_state.config);
140 interface_state.last_socket_status = 0U;
141
142 if (interface_state.init_ok != 0U)
143 {
144 w5500_lite_tcp_server_begin(static_cast<LcpEthernetId>(index),
145 LCP_MODBUS_TCP_PORT);
146 }
147 }
148
149 g_applied_config_generation = lcp_config_service_generation();
150 g_reload_pending = 0U;
151}

153uint8_t flush_pending_response(LcpEthernetId ethernet_id,
154 EthernetModbusInterfaceState& interface_state)
155{
156 const uint16_t response_length =
157 modbus_tcp_server_response_length(interface_state.server);
158
159 if (response_length == 0U)
160 {
161 return 1U;
162 }
163
164 const uint16_t sent = w5500_lite_tcp_server_send(
165 ethernet_id,
166 modbus_tcp_server_response(interface_state.server),
167 response_length);
168
169 if (sent != response_length)
170 {
171 /* TX_FSR может быть временно мал; ответ сохраняется до следующего poll. */
172 return 0U;
173 }
174
175 modbus_tcp_server_response_sent(interface_state.server);
176 return 1U;
177}

185void poll_interface(LcpEthernetId ethernet_id)
186{
187 EthernetModbusInterfaceState& interface_state =
188 g_interfaces[ethernet_id];
189
190 if ((interface_state.initialized == 0U) ||
191 (interface_state.init_ok == 0U))
192 {
193 return;
194 }
195
196 interface_state.last_socket_status =
197 w5500_lite_tcp_server_status(ethernet_id);
198
199 if (interface_state.last_socket_status == W5500_SOCKET_ESTABLISHED)
200 {
201 if (flush_pending_response(ethernet_id, interface_state) == 0U)
202 {
203 return;
204 }
205 }
206
207 const uint16_t received = w5500_lite_tcp_server_receive(
208 ethernet_id,
209 LCP_MODBUS_TCP_PORT,
210 g_rx_chunk[ethernet_id],
211 RX_CHUNK_CAPACITY);
212
213 interface_state.last_socket_status =
214 w5500_lite_tcp_server_status(ethernet_id);
215
216 if (interface_state.last_socket_status != W5500_SOCKET_ESTABLISHED)
217 {
218 reset_stream_if_disconnected(interface_state);
219 return;
220 }
221
222 if (received != 0U)
223 {
224 if (modbus_tcp_server_append(interface_state.server,
225 g_rx_chunk[ethernet_id],
226 received) == 0U)
227 {
228 ++interface_state.transport_error_count;
229 }
230 }
231
232 for (uint8_t response_index = 0U;
233 response_index < MAX_RESPONSES_PER_POLL;
234 ++response_index)
235 {
236 if (modbus_tcp_server_process(interface_state.server,
237 read_holding,
238 0) == 0U)
239 {
240 break;
241 }
242
243 if (flush_pending_response(ethernet_id, interface_state) == 0U)
244 {
245 break;
246 }
247 }
248}
