59LcpConfigSdResult load_baud(LcpConfigBundle& bundle,
60 LcpConfigSdReport& report)
61{
62 uint32_t values[LCP_FIELD_PORT_COUNT + 1U];
63 uint8_t loaded_count = 0U;
64 const SdConfigResult result = sd_config_load_uint32(
65 BAUD_FILE,
66 values,
67 sizeof(values) / sizeof(values[0]),
68 &loaded_count);
69
70 if (result != SD_CONFIG_OK)
71 {
72 return fail_sd(report,
73 BAUD_FILE,
74 result,
75 (result == SD_CONFIG_CARD_NOT_READY)
76 ? LCP_CONFIG_SD_STORAGE_NOT_READY
77 : LCP_CONFIG_SD_FILE_ERROR);
78 }
79
80 if ((loaded_count != LCP_FIELD_PORT_COUNT) ||
81 (values[0] != LCP_FIELD_PORT_COUNT))
82 {
83 return fail_sd(report,
84 BAUD_FILE,
85 SD_CONFIG_INVALID_COUNT,
86 LCP_CONFIG_SD_INVALID_VALUE);
87 }
88
89 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
90 {
91 const uint32_t baudrate = values[index + 1U];
92
93 if ((baudrate < MINIMUM_BAUDRATE) ||
94 (baudrate > MAXIMUM_BAUDRATE))
95 {
96 return fail_sd(report,
97 BAUD_FILE,
98 SD_CONFIG_INVALID_VALUE,
99 LCP_CONFIG_SD_INVALID_VALUE);
100 }
101
102 bundle.field_baudrate[index] = baudrate;
103 }
104
105 return LCP_CONFIG_SD_OK;
106}

157LcpConfigSdResult load_octets(const char* file_name,

197 LCP_CONFIG_SD_INVALID_VALUE);

200 output[index] = static_cast<uint8_t>(value);

326uint8_t mac_valid(const uint8_t mac[6])
327{
328 uint8_t any_nonzero = 0U;
329
330 for (uint8_t index = 0U; index < 6U; ++index)
331 {
332 any_nonzero |= mac[index];
333 }
334
335 return ((any_nonzero != 0U) && ((mac[0] & 0x01U) == 0U)) ? 1U : 0U;
336}

338uint8_t ip_valid(const uint8_t ip[4])
339{
340 if (((ip[0] == 0U) && (ip[1] == 0U) &&
341 (ip[2] == 0U) && (ip[3] == 0U)) ||
342 ((ip[0] >= 224U) && (ip[0] <= 239U)) ||
343 ((ip[0] == 255U) && (ip[1] == 255U) &&
344 (ip[2] == 255U) && (ip[3] == 255U)))
345 {
346 return 0U;
347 }
348
349 return 1U;
350}

352uint8_t subnet_valid(const uint8_t subnet[4])
353{
354 const uint32_t mask =
355 (static_cast<uint32_t>(subnet[0]) << 24U) |
356 (static_cast<uint32_t>(subnet[1]) << 16U) |
357 (static_cast<uint32_t>(subnet[2]) << 8U) |
358 static_cast<uint32_t>(subnet[3]);
359
360 if (mask == 0U)
361 {
362 return 0U;
363 }
364
365 const uint32_t inverted = ~mask;
366 return ((inverted & (inverted + 1U)) == 0U) ? 1U : 0U;
367}

384void lcp_config_bundle_set_defaults(LcpConfigBundle& bundle)
385{
386 memset(&bundle, 0, sizeof(bundle));
387
388 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
389 {
390 bundle.field_baudrate[index] = 9600U;
391 bundle.field_parity[index] = 0U;
392 }
393
394 const uint8_t mac1[6] = { 0x02U, 0x4CU, 0x43U, 0x50U, 0x00U, 0x01U };
395 const uint8_t mac2[6] = { 0x02U, 0x4CU, 0x43U, 0x50U, 0x00U, 0x02U };
396 const uint8_t ip1[4] = { 192U, 168U, 1U, 1U };
397 const uint8_t ip2[4] = { 192U, 168U, 1U, 2U };
398 const uint8_t gateway[4] = { 192U, 168U, 1U, 254U };
399 const uint8_t subnet[4] = { 255U, 255U, 255U, 0U };
400
401 memcpy(bundle.ethernet_mac[0], mac1, sizeof(mac1));
402 memcpy(bundle.ethernet_mac[1], mac2, sizeof(mac2));
403 memcpy(bundle.ethernet_ip[0], ip1, sizeof(ip1));
404 memcpy(bundle.ethernet_ip[1], ip2, sizeof(ip2));
405 memcpy(bundle.ethernet_gateway[0], gateway, sizeof(gateway));
406 memcpy(bundle.ethernet_gateway[1], gateway, sizeof(gateway));
407 memcpy(bundle.ethernet_subnet[0], subnet, sizeof(subnet));
408 memcpy(bundle.ethernet_subnet[1], subnet, sizeof(subnet));
409
410 bundle.x2x_module_count = 0U;
411 bundle.eia_count = LCP_CONFIG_EIA_VALUE_COUNT;
412 bundle.eia_value[0] = 3;
413 bundle.eia_value[1] = 4;
414 bundle.eia_value[2] = 5;

418uint8_t lcp_config_bundle_validate(const LcpConfigBundle& bundle)
419{
420 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
421 {
422 if ((bundle.field_baudrate[index] < MINIMUM_BAUDRATE) ||
423 (bundle.field_baudrate[index] > MAXIMUM_BAUDRATE) ||
424 (bundle.field_parity[index] > 2U))
425 {
426 return 0U;
427 }
428 }
429
430 for (uint8_t index = 0U; index < LCP_ETHERNET_COUNT; ++index)
431 {
432 if ((mac_valid(bundle.ethernet_mac[index]) == 0U) ||
433 (ip_valid(bundle.ethernet_ip[index]) == 0U) ||
434 (subnet_valid(bundle.ethernet_subnet[index]) == 0U))
435 {
436 return 0U;
437 }
438 }
439
440 if (bundle.x2x_module_count > X2X_MAX_MODULES)
441 {
442 return 0U;
443 }
444
445 for (uint8_t index = 0U; index < X2X_MAX_MODULES; ++index)
446 {
447 if (index < bundle.x2x_module_count)
448 {
449 const X2XModuleDescriptor* descriptor =
450 x2x_catalog_find_by_id(bundle.x2x_module_id[index]);
451
452 if ((descriptor == 0) ||
453 (descriptor->allowed_in_x2x_config == 0U))
454 {
455 return 0U;
456 }
457 }
458 else if (bundle.x2x_module_id[index] != 0U)
459 {
460 return 0U;
461 }
462 }
463
464 if (bundle.eia_count != LCP_CONFIG_EIA_VALUE_COUNT)
465 {
466 return 0U;
467 }
468
469 for (uint8_t index = 0U; index < LCP_CONFIG_EIA_VALUE_COUNT; ++index)
470 {
471 if ((bundle.eia_value[index] < 0) ||
472 (bundle.eia_value[index] > 255))
473 {
474 return 0U;
475 }
476 }
477
478 for (uint8_t index = 0U; index < LCP_CONFIG_RESERVED_BYTES; ++index)
479 {
480 if (bundle.reserved[index] != 0U)
481 {
482 return 0U;
483 }
484 }
485
486 return 1U;
487}

489LcpConfigSdResult lcp_config_bundle_load_sd(LcpConfigBundle& output,
490 LcpConfigSdReport& report)
491{
492 prepare_report(report);
493
494 if (lcp_sd_storage_ready() == 0U)

497 report.sd_result = SD_CONFIG_CARD_NOT_READY;
498 return report.result;
499 }
500
501 LcpConfigBundle candidate;
502 lcp_config_bundle_set_defaults(candidate);
503
504 if (load_baud(candidate, report) != LCP_CONFIG_SD_OK)
505 {
506 return report.result;
507 }
508
509 if (load_parity(candidate, report) != LCP_CONFIG_SD_OK)
510 {
511 return report.result;
512 }
513
514 if (load_ethernet(candidate, report) != LCP_CONFIG_SD_OK)
515 {
516 return report.result;
517 }
518
519 if (load_x2x(candidate, report) != LCP_CONFIG_SD_OK)
520 {
521 return report.result;
522 }
523
524 if (load_eia(candidate, report) != LCP_CONFIG_SD_OK)
525 {
526 return report.result;
527 }
528
529 if (lcp_config_bundle_validate(candidate) == 0U)
530 {
531 report.result = LCP_CONFIG_SD_INVALID_VALUE;
532 report.failed_file = "bundle";
533 report.sd_result = SD_CONFIG_INVALID_VALUE;
534 return report.result;
535 }
536
537 output = candidate;
538 report.result = LCP_CONFIG_SD_OK;
539 report.bundle_crc32 = lcp_config_bundle_crc32(candidate);
540 return report.result;
