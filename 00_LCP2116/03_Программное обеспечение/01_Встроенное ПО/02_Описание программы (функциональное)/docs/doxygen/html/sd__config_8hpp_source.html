15/**
16 * @brief Результат операции с числовым конфигурационным файлом.
17 */
18enum SdConfigResult
19{
20 SD_CONFIG_OK = 0, /**< Файл корректно прочитан или записан. */
21 SD_CONFIG_CARD_NOT_READY = 1, /**< microSD ещё не смонтирована. */
22 SD_CONFIG_FILE_NOT_FOUND = 2, /**< Файл с указанным именем отсутствует. */
23 SD_CONFIG_FILE_OPEN_FAILED = 3, /**< Файл найден, но не открыт. */
24 SD_CONFIG_EMPTY_FILE = 4, /**< Не найдена строка с количеством. */
25 SD_CONFIG_INVALID_COUNT = 5, /**< Количество отсутствует или не помещается в буфер. */
26 SD_CONFIG_INVALID_VALUE = 6, /**< Обнаружено значение вне числового формата или диапазона типа. */
27 SD_CONFIG_INCOMPLETE_FILE = 7, /**< Значений меньше, чем объявлено первой строкой. */
28 SD_CONFIG_WRITE_FAILED = 8, /**< Выходной текст не помещается во внутренний буфер. */
29 SD_CONFIG_LINE_TOO_LONG = 9, /**< Физическая строка превышает внутренний предел парсера. */
30 SD_CONFIG_STORAGE_ERROR = 10, /**< Низкоуровневая операция хранилища завершилась ошибкой. */
31 SD_CONFIG_NOT_ATTEMPTED = 11 /**< Загрузка ещё не запускалась. */
32};

34/**
35 * @brief Загружает знаковые 16-битные значения из числового TXT-файла.
36 *
37 * Формат файла:
38 *
39 * @code
40 * N
41 * value_1
42 * ...
43 * value_N
44 * fin
45 * @endcode
46 *
47 * Маркер `fin` необязателен. Пустые строки, UTF-8 BOM в начале файла и
48 * комментарии после `//` или `#` игнорируются.
49 *
50 * @param[in] file_name Каноническое имя файла в формате FAT 8.3.
51 * @param[out] output Буфер результата; `output[0]` получает количество N,
52 * значения размещаются в `output[1..N]`.
53 * @param[in] capacity Количество элементов в `output`, включая `output[0]`.
54 * @param[out] loaded_count Фактическое количество значений без `output[0]`.
55 * @return Код результата операции.
56 */
57SdConfigResult sd_config_load_int16(const char* file_name,
58 int16_t* output,
59 size_t capacity,
60 uint8_t* loaded_count);
61
62/**
63 * @brief Загружает беззнаковые 32-битные значения из числового TXT-файла.
64 *
65 * Использует тот же формат, правила комментариев и необязательный маркер `fin`,
66 * что и sd_config_load_int16(). Функция предназначена, в частности, для
67 * реальных значений baudrate, которые не обязаны помещаться в int16_t.
68 *
69 * @param[in] file_name Каноническое имя файла в формате FAT 8.3.
70 * @param[out] output Буфер результата; `output[0]` получает количество N,
71 * значения размещаются в `output[1..N]`.
72 * @param[in] capacity Количество элементов в `output`, включая `output[0]`.
73 * @param[out] loaded_count Фактическое количество значений без `output[0]`.
74 * @return Код результата операции.
75 */
76SdConfigResult sd_config_load_uint32(const char* file_name,
77 uint32_t* output,
78 size_t capacity,
79 uint8_t* loaded_count);

81/**
82 * @brief Сохраняет int16-конфигурацию в числовой TXT-файл.
83 *
84 * @param[in] file_name Имя файла в формате FAT 8.3.
85 * @param[in] input Буфер; значения берутся из `input[1..size]`.
86 * @param[in] size Количество сохраняемых значений.
87 * @return Код результата операции.
88 *
89 * @note Функция записывает количество и значения. Маркер `fin` не обязателен

91 */
92SdConfigResult sd_config_save_int16(const char* file_name,

94 uint8_t size);
