98static uint8_t days_in_month(uint16_t year, uint8_t month)
99{
100 static const uint8_t days[] =
101 {
102 31U, 28U, 31U, 30U, 31U, 30U,
103 31U, 31U, 30U, 31U, 30U, 31U
104 };
105
106 if ((month == 2U) && (is_leap_year(year) != 0U))
107 {
108 return 29U;
109 }
110
111 if ((month < 1U) || (month > 12U))
112 {
113 return 0U;
114 }
115
116 return days[month - 1U];
117}

149static uint32_t build_time_register(const Sam3xRtcDateTime* date_time)
150{
151 return (u8_to_bcd(date_time->second) << 0U) |
152 (u8_to_bcd(date_time->minute) << 8U) |
153 (u8_to_bcd(date_time->hour) << 16U);
154}

156static uint32_t build_date_register(const Sam3xRtcDateTime* date_time)
157{
158 const uint16_t normalized_year = sam3x_rtc_normalize_year(date_time->year);
159 const uint8_t century = static_cast<uint8_t>(normalized_year / 100U);
160 const uint8_t year_2_digits = static_cast<uint8_t>(normalized_year % 100U);
161
162 return (u8_to_bcd(century) << 0U) |
163 (u8_to_bcd(year_2_digits) << 8U) |
164 (u8_to_bcd(date_time->month) << 16U) |
165 (static_cast<uint32_t>(date_time->day_of_week & 0x07U) << 21U) |
166 (u8_to_bcd(date_time->day) << 24U);
167}

169static Sam3xRtcResult prepare_datetime_registers(const Sam3xRtcDateTime* date_time)
170{
171 if (date_time == 0)
172 {
173 return SAM3X_RTC_INVALID_ARGUMENT;
174 }
175
176 Sam3xRtcDateTime normalized = *date_time;
177 normalized.year = sam3x_rtc_normalize_year(normalized.year);
178
179 if (sam3x_rtc_datetime_range_valid(&normalized) == 0U)
180 {
181 return SAM3X_RTC_INVALID_DATE_TIME;
182 }
183
184 normalized.day_of_week = sam3x_rtc_calculate_day_of_week(normalized.year,
185 normalized.month,
186 normalized.day);
187
188 g_pending_time_register = build_time_register(&normalized);
189 g_pending_calendar_register = build_date_register(&normalized);
190
191 return SAM3X_RTC_OK;
192}

194static void complete_datetime_update(void)
195{
196 RTC->RTC_SCCR = RTC_SCCR_ACKCLR;
197 RTC->RTC_TIMR = g_pending_time_register;
198 RTC->RTC_CALR = g_pending_calendar_register;
199 RTC->RTC_CR &= ~(RTC_CR_UPDTIM | RTC_CR_UPDCAL);
200 RTC->RTC_SCCR = RTC_SCCR_SECCLR | RTC_SCCR_TIMCLR | RTC_SCCR_CALCLR;
201
202 if (sam3x_rtc_registers_valid() != 0U)
203 {
204 g_update_result = SAM3X_RTC_OK;
205 g_update_state = SAM3X_RTC_UPDATE_DONE;
206 }
207 else
208 {
209 g_update_result = SAM3X_RTC_VERIFY_FAILED;
210 g_update_state = SAM3X_RTC_UPDATE_FAILED;
211 }

239Sam3xRtcResult sam3x_rtc_get_datetime(Sam3xRtcDateTime* date_time)
240{
241 if (date_time == 0)
242 {
243 return SAM3X_RTC_INVALID_ARGUMENT;
244 }
245
246 const uint32_t time_register = read_stable_time_register();
247 const uint32_t date_register = read_stable_date_register();
248
249 date_time->second = bcd_to_u8((time_register >> 0U) & 0x7FU);
250 date_time->minute = bcd_to_u8((time_register >> 8U) & 0x7FU);
251 date_time->hour = bcd_to_u8((time_register >> 16U) & 0x3FU);
252
253 const uint8_t century = bcd_to_u8((date_register >> 0U) & 0x7FU);
254 const uint8_t year_2_digits = bcd_to_u8((date_register >> 8U) & 0xFFU);
255
256 if (century == 0U)
257 {
258 date_time->year = static_cast<uint16_t>(2000U + year_2_digits);
259 }
260 else
261 {
262 date_time->year = static_cast<uint16_t>((static_cast<uint16_t>(century) * 100U) + year_2_digits);
263 }
264
265 date_time->month = bcd_to_u8((date_register >> 16U) & 0x1FU);
266 date_time->day_of_week = static_cast<uint8_t>((date_register >> 21U) & 0x07U);
267 date_time->day = bcd_to_u8((date_register >> 24U) & 0x3FU);
268
269 return SAM3X_RTC_OK;
270}

272Sam3xRtcResult sam3x_rtc_set_datetime(const Sam3xRtcDateTime* date_time)
273{
274 Sam3xRtcResult result = sam3x_rtc_start_datetime_update(date_time);
275
276 if (result != SAM3X_RTC_OK)
277 {
278 return result;
279 }
280
281 for (uint32_t guard = 0U; guard < RTC_BLOCKING_TIMEOUT_COUNT; ++guard)
282 {
283 sam3x_rtc_poll_datetime_update();
284
285 if (g_update_state == SAM3X_RTC_UPDATE_DONE)
286 {
287 return g_update_result;
288 }
289
290 if (g_update_state == SAM3X_RTC_UPDATE_FAILED)
291 {
292 return g_update_result;
293 }
294 }
295
296 sam3x_rtc_cancel_datetime_update(SAM3X_RTC_ACK_TIMEOUT);
297 return SAM3X_RTC_ACK_TIMEOUT;
298}

300Sam3xRtcResult sam3x_rtc_start_datetime_update(const Sam3xRtcDateTime* date_time)
301{
302 if (g_update_state == SAM3X_RTC_UPDATE_WAIT_ACK)
303 {
304 return SAM3X_RTC_BUSY;
305 }
306
307 const Sam3xRtcResult result = prepare_datetime_registers(date_time);
308
309 if (result != SAM3X_RTC_OK)
310 {
311 g_update_result = result;
312 g_update_state = SAM3X_RTC_UPDATE_FAILED;
313 return result;
314 }
315
316 RTC->RTC_IDR = RTC_IDR_ACKDIS;
317 RTC->RTC_CR |= (RTC_CR_UPDTIM | RTC_CR_UPDCAL);
318
319 g_update_result = SAM3X_RTC_BUSY;
320 g_update_state = SAM3X_RTC_UPDATE_WAIT_ACK;
321
322 return SAM3X_RTC_OK;
323}

325void sam3x_rtc_poll_datetime_update(void)
326{
327 if (g_update_state != SAM3X_RTC_UPDATE_WAIT_ACK)
328 {
329 return;
330 }
331
332 if ((RTC->RTC_SR & RTC_SR_ACKUPD) == RTC_SR_ACKUPD)
333 {
334 complete_datetime_update();

387uint8_t sam3x_rtc_datetime_range_valid(const Sam3xRtcDateTime* date_time)
388{
389 if (date_time == 0)
390 {
391 return 0U;
392 }
393
394 const uint16_t normalized_year = sam3x_rtc_normalize_year(date_time->year);
395
396 if ((normalized_year < 2000U) || (normalized_year > 2099U))
397 {
398 return 0U;
399 }
400
401 if ((date_time->month < 1U) || (date_time->month > 12U))
402 {
403 return 0U;
404 }
405
406 const uint8_t max_day = days_in_month(normalized_year, date_time->month);
407
408 if ((date_time->day < 1U) || (date_time->day > max_day))
409 {
410 return 0U;
411 }
412
413 if (date_time->hour > 23U)
414 {
415 return 0U;
416 }
417
418 if (date_time->minute > 59U)
419 {
420 return 0U;
421 }
422
423 if (date_time->second > 59U)
424 {
425 return 0U;
426 }
427
428 return 1U;
429}

441uint8_t sam3x_rtc_calculate_day_of_week(uint16_t year, uint8_t month, uint8_t day)
442{
443 uint16_t normalized_year = sam3x_rtc_normalize_year(year);
444 uint16_t y = normalized_year;
445 uint8_t m = month;
446
447 if ((m == 1U) || (m == 2U))
448 {
449 m = static_cast<uint8_t>(m + 12U);
450 --y;
451 }
452
453 uint8_t week = static_cast<uint8_t>((day + (2U * m) + ((3U * (m + 1U)) / 5U) + y + (y / 4U) - (y / 100U) + (y / 400U)) % 7U);
454 ++week;
455
456 return week;
457}
