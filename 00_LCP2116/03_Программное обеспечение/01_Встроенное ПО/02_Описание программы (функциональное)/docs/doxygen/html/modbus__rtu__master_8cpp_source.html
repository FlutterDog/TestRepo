55void append_crc(ModbusRtuMaster& master, uint16_t payload_length)
56{
57 const uint16_t crc = modbus_rtu_crc16(master.tx_buffer, payload_length);
58 master.tx_buffer[payload_length] = static_cast<uint8_t>(crc & 0x00FFU);
59 master.tx_buffer[payload_length + 1U] = static_cast<uint8_t>(crc >> 8U);
60 master.tx_length = payload_length + MODBUS_CRC_LENGTH;
61}

63uint8_t start_transmission(ModbusRtuMaster& master, uint32_t timeout_ms)
64{
65 master.transport->clear_rx();
66 master.rx_length = 0U;
67 master.exception_code = 0U;
68 master.timeout_ms = timeout_ms;
69 master.deadline_ms = millis() + timeout_ms;
70
71 const size_t written = master.transport->write(master.tx_buffer,
72 master.tx_length);
73
74 if (written != master.tx_length)
75 {
76 finish_transaction(master, MODBUS_RTU_RESULT_TRANSPORT_ERROR);
77 return 0U;
78 }
79
80 master.result = MODBUS_RTU_RESULT_BUSY;
81 master.state = MODBUS_RTU_MASTER_STATE_WAIT_TX;
82 return 1U;
83}

85uint8_t response_crc_valid(const ModbusRtuMaster& master,
86 uint16_t response_length)
87{
88 if (response_length < MODBUS_CRC_LENGTH)
89 {
90 return 0U;
91 }
92
93 const uint16_t calculated = modbus_rtu_crc16(
94 master.rx_buffer,
95 response_length - MODBUS_CRC_LENGTH);
96 const uint16_t received =
97 static_cast<uint16_t>(master.rx_buffer[response_length - 2U]) |
98 static_cast<uint16_t>(master.rx_buffer[response_length - 1U] << 8U);
99
100 return (calculated == received) ? 1U : 0U;
101}

103void validate_response(ModbusRtuMaster& master)
104{
105 uint16_t response_length = master.expected_response_length;
106
107 if ((master.rx_length >= 2U) &&
108 (master.rx_buffer[1] ==
109 static_cast<uint8_t>(master.function_code | 0x80U)))
110 {
111 response_length = MODBUS_EXCEPTION_RESPONSE_LENGTH;
112 }
113
114 if (master.rx_length < response_length)
115 {
116 return;
117 }
118
119 if (response_crc_valid(master, response_length) == 0U)
120 {
121 finish_transaction(master, MODBUS_RTU_RESULT_CRC_ERROR);
122 return;
123 }
124
125 if (master.rx_buffer[0] != master.slave_address)
126 {
127 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
128 return;
129 }
130
131 if (master.rx_buffer[1] ==
132 static_cast<uint8_t>(master.function_code | 0x80U))
133 {
134 master.exception_code = master.rx_buffer[2];
135 finish_transaction(master, MODBUS_RTU_RESULT_EXCEPTION);
136 return;
137 }
138
139 if (master.rx_buffer[1] != master.function_code)
140 {
141 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
142 return;
143 }
144
145 if (master.transaction_type == MODBUS_RTU_TRANSACTION_READ_HOLDING)
146 {
147 const uint16_t expected_byte_count = master.register_count * 2U;
148
149 if ((master.rx_buffer[2] != expected_byte_count) ||
150 (master.read_output == 0))
151 {
152 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
153 return;
154 }
155
156 for (uint16_t register_index = 0U;
157 register_index < master.register_count;
158 ++register_index)
159 {
160 const uint16_t byte_index = 3U + (register_index * 2U);
161 master.read_output[register_index] =
162 static_cast<uint16_t>(master.rx_buffer[byte_index] << 8U) |
163 static_cast<uint16_t>(master.rx_buffer[byte_index + 1U]);
164 }
165
166 finish_transaction(master, MODBUS_RTU_RESULT_OK);
167 return;
168 }
169
170 if (master.transaction_type == MODBUS_RTU_TRANSACTION_WRITE_MULTIPLE)
171 {
172 const uint16_t response_address =
173 static_cast<uint16_t>(master.rx_buffer[2] << 8U) |
174 static_cast<uint16_t>(master.rx_buffer[3]);
175 const uint16_t response_count =
176 static_cast<uint16_t>(master.rx_buffer[4] << 8U) |
177 static_cast<uint16_t>(master.rx_buffer[5]);
178
179 if ((response_address != master.start_address) ||
180 (response_count != master.register_count))
181 {
182 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
183 return;
184 }
185
186 finish_transaction(master, MODBUS_RTU_RESULT_OK);
187 return;
188 }
189
190 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
191}

193void receive_available_bytes(ModbusRtuMaster& master)
194{
195 /*
196 * Цикл ограничен фактическим RX count и статической ёмкостью 256 байт.
197 * Он не ожидает появления новых данных.
198 */
199 while (master.transport->available() > 0U)
200 {
201 if (master.rx_length >= MODBUS_RTU_MASTER_RX_CAPACITY)
202 {
203 finish_transaction(master, MODBUS_RTU_RESULT_INVALID_RESPONSE);
204 return;
205 }
206
207 const int value = master.transport->read();
208
209 if (value < 0)
210 {
211 break;
212 }
213
214 master.rx_buffer[master.rx_length] = static_cast<uint8_t>(value);
215 ++master.rx_length;
216
217 uint16_t required_length = master.expected_response_length;
218
219 if ((master.rx_length >= 2U) &&
220 (master.rx_buffer[1] ==
221 static_cast<uint8_t>(master.function_code | 0x80U)))
222 {
223 required_length = MODBUS_EXCEPTION_RESPONSE_LENGTH;
224 }
225
226 if (master.rx_length >= required_length)
227 {
228 break;
229 }
230 }
231}

246void modbus_rtu_master_reset(ModbusRtuMaster& master)
247{
248 const ModbusRtuTransport* transport = master.transport;
249 const uint32_t interframe_gap_ms = master.interframe_gap_ms;
250 const uint32_t next_request_ms = master.next_request_ms;
251
252 memset(&master, 0, sizeof(master));
253 master.transport = transport;
254 master.state = MODBUS_RTU_MASTER_STATE_IDLE;
255 master.result = MODBUS_RTU_RESULT_IDLE;
256 master.interframe_gap_ms = interframe_gap_ms;
257 master.next_request_ms = next_request_ms;
258
259 if (transport_valid(transport) != 0U)
260 {
261 transport->clear_rx();
262 }

265uint8_t modbus_rtu_master_ready(const ModbusRtuMaster& master)
266{
267 if ((master.state != MODBUS_RTU_MASTER_STATE_IDLE) ||
268 (master.result == MODBUS_RTU_RESULT_BUSY))
269 {
270 return 0U;

276uint8_t modbus_rtu_master_start_read_holding(ModbusRtuMaster& master,
277 uint8_t slave_address,
278 uint16_t start_address,
279 uint16_t register_count,
280 uint16_t* output,
281 uint32_t timeout_ms)
282{
283 if ((transport_valid(master.transport) == 0U) ||
284 (slave_address_valid(slave_address) == 0U) ||
285 (register_count == 0U) ||
286 (register_count > MODBUS_RTU_MASTER_MAX_READ_REGISTERS) ||
287 (output == 0) ||
288 (timeout_ms == 0U))
289 {
290 master.result = MODBUS_RTU_RESULT_INVALID_ARGUMENT;
291 return 0U;
292 }
293
294 if (modbus_rtu_master_ready(master) == 0U)
295 {
296 return 0U;
297 }
298
299 master.transaction_type = MODBUS_RTU_TRANSACTION_READ_HOLDING;
300 master.slave_address = slave_address;
301 master.function_code = MODBUS_FUNCTION_READ_HOLDING;
302 master.start_address = start_address;
303 master.register_count = register_count;
304 master.read_output = output;
305 master.expected_response_length = 5U + (register_count * 2U);
306
307 master.tx_buffer[0] = slave_address;
308 master.tx_buffer[1] = MODBUS_FUNCTION_READ_HOLDING;
309 master.tx_buffer[2] = static_cast<uint8_t>(start_address >> 8U);
310 master.tx_buffer[3] = static_cast<uint8_t>(start_address & 0x00FFU);
311 master.tx_buffer[4] = static_cast<uint8_t>(register_count >> 8U);
312 master.tx_buffer[5] = static_cast<uint8_t>(register_count & 0x00FFU);
313 append_crc(master, MODBUS_REQUEST_BASE_LENGTH);
314
315 return start_transmission(master, timeout_ms);
316}

318uint8_t modbus_rtu_master_start_write_multiple(ModbusRtuMaster& master,
319 uint8_t slave_address,
320 uint16_t start_address,
321 const uint16_t* values,
322 uint16_t register_count,
323 uint32_t timeout_ms)
324{
325 const uint16_t payload_length = 7U + (register_count * 2U);
326
327 if ((transport_valid(master.transport) == 0U) ||
328 (slave_address_valid(slave_address) == 0U) ||
329 (values == 0) ||
330 (register_count == 0U) ||
331 (register_count > MODBUS_RTU_MASTER_MAX_WRITE_REGISTERS) ||
332 ((payload_length + MODBUS_CRC_LENGTH) >
333 MODBUS_RTU_MASTER_TX_CAPACITY) ||
334 (timeout_ms == 0U))
335 {
336 master.result = MODBUS_RTU_RESULT_INVALID_ARGUMENT;
337 return 0U;
338 }
339
340 if (modbus_rtu_master_ready(master) == 0U)
341 {
342 return 0U;
343 }
344
345 master.transaction_type = MODBUS_RTU_TRANSACTION_WRITE_MULTIPLE;
346 master.slave_address = slave_address;
347 master.function_code = MODBUS_FUNCTION_WRITE_MULTIPLE;
348 master.start_address = start_address;
349 master.register_count = register_count;
350 master.read_output = 0;
351 master.expected_response_length = MODBUS_WRITE_RESPONSE_LENGTH;
352
353 master.tx_buffer[0] = slave_address;
354 master.tx_buffer[1] = MODBUS_FUNCTION_WRITE_MULTIPLE;
355 master.tx_buffer[2] = static_cast<uint8_t>(start_address >> 8U);
356 master.tx_buffer[3] = static_cast<uint8_t>(start_address & 0x00FFU);
357 master.tx_buffer[4] = static_cast<uint8_t>(register_count >> 8U);
358 master.tx_buffer[5] = static_cast<uint8_t>(register_count & 0x00FFU);
359 master.tx_buffer[6] = static_cast<uint8_t>(register_count * 2U);
360
361 for (uint16_t register_index = 0U;
362 register_index < register_count;
363 ++register_index)
364 {
365 const uint16_t byte_index = 7U + (register_index * 2U);
366 master.tx_buffer[byte_index] =
367 static_cast<uint8_t>(values[register_index] >> 8U);
368 master.tx_buffer[byte_index + 1U] =
369 static_cast<uint8_t>(values[register_index] & 0x00FFU);
370 }
371
372 append_crc(master, payload_length);
373 return start_transmission(master, timeout_ms);
374}

376void modbus_rtu_master_poll(ModbusRtuMaster& master)
377{
378 if ((master.state == MODBUS_RTU_MASTER_STATE_IDLE) ||
379 (master.result != MODBUS_RTU_RESULT_BUSY))
380 {
381 return;
382 }
383
384 const uint32_t now_ms = millis();
385
386 if (master.state == MODBUS_RTU_MASTER_STATE_WAIT_TX)
387 {
388 if (master.transport->tx_idle() != 0U)
389 {
390 master.state = MODBUS_RTU_MASTER_STATE_WAIT_RESPONSE;
391 master.deadline_ms = now_ms + master.timeout_ms;
392 return;
393 }
394
395 if (deadline_reached(now_ms, master.deadline_ms) != 0U)
396 {
397 finish_transaction(master,
398 MODBUS_RTU_RESULT_TRANSPORT_ERROR);
399 }
400
401 return;
402 }
403
404 if (master.state == MODBUS_RTU_MASTER_STATE_WAIT_RESPONSE)
405 {
406 receive_available_bytes(master);
407
408 if (master.state == MODBUS_RTU_MASTER_STATE_IDLE)
409 {
410 return;
411 }
412
413 validate_response(master);
414
415 if (master.state == MODBUS_RTU_MASTER_STATE_IDLE)
416 {
417 return;
418 }
419
420 if (deadline_reached(now_ms, master.deadline_ms) != 0U)
421 {
422 finish_transaction(master, MODBUS_RTU_RESULT_TIMEOUT);
423 }
424 }
425}
