16constexpr uint32_t CONFIG_STORAGE_MAGIC = 0x4C434346UL; /* LCCF */
17constexpr uint32_t CONFIG_COMMIT_MAGIC = 0x4C434343UL; /* LCCC */
18constexpr uint16_t CONFIG_STORAGE_VERSION = 1U;
19constexpr uint16_t CONFIG_SCHEMA_VERSION = 1U;
20constexpr uint32_t CONFIG_RESERVED_TOTAL_BYTES = 32U * 1024U;
21constexpr uint32_t CONFIG_SLOT_SIZE = CONFIG_RESERVED_TOTAL_BYTES / 2U;
22constexpr uint32_t CONFIG_SLOT_COUNT = 2U;

24struct __attribute__((packed)) ConfigRecordHeader
25{
26 uint32_t magic;
27 uint16_t storage_version;
28 uint16_t schema_version;
29 uint32_t sequence;
30 uint32_t payload_size;
31 uint32_t payload_crc32;
32 uint32_t header_crc32;
33 uint32_t reserved[2];
34};
35
36struct __attribute__((packed)) ConfigCommitRecord
37{
38 uint32_t magic;
39 uint16_t storage_version;
40 uint16_t schema_version;
41 uint32_t sequence;
42 uint32_t payload_crc32;
43 uint32_t header_crc32;
44 uint32_t commit_crc32;
45 uint32_t reserved[2];
46};
47
48static_assert(sizeof(ConfigRecordHeader) == 32U,
49 "ConfigRecordHeader must remain 32 bytes");
50static_assert(sizeof(ConfigCommitRecord) == 32U,
51 "ConfigCommitRecord must remain 32 bytes");
52static_assert((sizeof(ConfigRecordHeader) + sizeof(LcpConfigBundle)) <=
53 SAM3X_INTERNAL_FLASH_PAGE_SIZE,
54 "Configuration record must fit one Flash page");

146uint8_t read_slot(LcpConfigSlotId slot_id)
147{
148 const uint8_t index = slot_index(slot_id);
149 LcpConfigSlotStatus& status = g_slot_status[index];
150 reset_slot_status(status);
151
152 ConfigRecordHeader header;
153 g_last_flash_result = sam3x_internal_flash_read(slot_base(slot_id),
154 &header,
155 sizeof(header));
156 status.flash_result = g_last_flash_result;
157
158 if (g_last_flash_result != SAM3X_INTERNAL_FLASH_OK)
159 {
160 return 0U;
161 }
162
163 if ((header.magic != CONFIG_STORAGE_MAGIC) ||
164 (header.storage_version != CONFIG_STORAGE_VERSION) ||
165 (header.schema_version != CONFIG_SCHEMA_VERSION) ||
166 (header.sequence == 0U) ||
167 (header.payload_size != sizeof(LcpConfigBundle)) ||
168 (header.header_crc32 != record_header_crc32(header)))
169 {
170 return 0U;
171 }
172
173 LcpConfigBundle bundle;
174 g_last_flash_result = sam3x_internal_flash_read(
175 slot_base(slot_id) + sizeof(ConfigRecordHeader),
176 &bundle,
177 sizeof(bundle));
178 status.flash_result = g_last_flash_result;
179
180 if (g_last_flash_result != SAM3X_INTERNAL_FLASH_OK)
181 {
182 return 0U;
183 }
184
185 if ((lcp_config_bundle_validate(bundle) == 0U) ||
186 (header.payload_crc32 != lcp_config_bundle_crc32(bundle)))
187 {
188 return 0U;
189 }
190
191 ConfigCommitRecord commit;
192 g_last_flash_result = sam3x_internal_flash_read(slot_commit_page(slot_id),
193 &commit,
194 sizeof(commit));
195 status.flash_result = g_last_flash_result;
196
197 if (g_last_flash_result != SAM3X_INTERNAL_FLASH_OK)
198 {
199 return 0U;
200 }
201
202 if ((commit.magic != CONFIG_COMMIT_MAGIC) ||
203 (commit.storage_version != CONFIG_STORAGE_VERSION) ||
204 (commit.schema_version != CONFIG_SCHEMA_VERSION) ||
205 (commit.sequence != header.sequence) ||
206 (commit.payload_crc32 != header.payload_crc32) ||
207 (commit.header_crc32 != header.header_crc32) ||
208 (commit.commit_crc32 != commit_record_crc32(commit)))
209 {
210 return 0U;
211 }
212
213 g_slot_bundle[index] = bundle;
214 status.valid = 1U;
215 status.sequence = header.sequence;
216 status.payload_crc32 = header.payload_crc32;
217 return 1U;
218}

256uint32_t next_sequence(void)
257{
258 uint32_t newest = 0U;
259
260 for (uint8_t index = 0U; index < CONFIG_SLOT_COUNT; ++index)
261 {
262 if ((g_slot_status[index].valid != 0U) &&
263 ((newest == 0U) ||
264 (sequence_newer(g_slot_status[index].sequence, newest) != 0U)))
265 {
266 newest = g_slot_status[index].sequence;
267 }
268 }
269
270 ++newest;
271 return (newest == 0U) ? 1U : newest;
272}

274void build_write_pages(const LcpConfigBundle& bundle, uint32_t sequence)
275{
276 memset(g_record_page, 0xFF, sizeof(g_record_page));
277 memset(g_commit_page, 0xFF, sizeof(g_commit_page));
278
279 ConfigRecordHeader header;
280 memset(&header, 0, sizeof(header));
281 header.magic = CONFIG_STORAGE_MAGIC;
282 header.storage_version = CONFIG_STORAGE_VERSION;
283 header.schema_version = CONFIG_SCHEMA_VERSION;
284 header.sequence = sequence;
285 header.payload_size = sizeof(LcpConfigBundle);
286 header.payload_crc32 = lcp_config_bundle_crc32(bundle);
287 header.header_crc32 = record_header_crc32(header);
288
289 memcpy(g_record_page, &header, sizeof(header));
290 memcpy(g_record_page + sizeof(header), &bundle, sizeof(bundle));
291
292 ConfigCommitRecord commit;
293 memset(&commit, 0, sizeof(commit));
294 commit.magic = CONFIG_COMMIT_MAGIC;
295 commit.storage_version = CONFIG_STORAGE_VERSION;
296 commit.schema_version = CONFIG_SCHEMA_VERSION;
297 commit.sequence = sequence;
298 commit.payload_crc32 = header.payload_crc32;
299 commit.header_crc32 = header.header_crc32;
300 commit.commit_crc32 = commit_record_crc32(commit);
301 memcpy(g_commit_page, &commit, sizeof(commit));
302}

334void poll_sd_first(void)
335{
336 LcpConfigSdReport report;
337 const LcpConfigSdResult result =
338 lcp_config_bundle_load_sd(g_sd_first_bundle, report);
339
340 if (result != LCP_CONFIG_SD_OK)
341 {
342 g_candidate_status.report = report;
343 finish_operation(LCP_CONFIG_SERVICE_SD_ERROR);
344 return;
345 }
346
347 g_step = CONFIG_STEP_SD_READ_SECOND;
348}
349
350void poll_sd_second(void)
351{
352 LcpConfigBundle second;
353 LcpConfigSdReport report;
354 const LcpConfigSdResult result = lcp_config_bundle_load_sd(second, report);
355
356 if (result != LCP_CONFIG_SD_OK)
357 {
358 g_candidate_status.report = report;
359 finish_operation(LCP_CONFIG_SERVICE_SD_ERROR);
360 return;
361 }
362
363 const uint32_t first_crc = lcp_config_bundle_crc32(g_sd_first_bundle);
364 const uint32_t second_crc = lcp_config_bundle_crc32(second);
365
366 if ((first_crc != second_crc) ||
367 (lcp_config_bundle_equal(g_sd_first_bundle, second) == 0U))
368 {
369 g_candidate_status.report = report;
370 g_candidate_status.available = 1U;
371 g_candidate_status.stable = 0U;
372 g_candidate_status.crc32 = second_crc;
373 finish_operation(LCP_CONFIG_SERVICE_SD_UNSTABLE);
374 return;
375 }
376
377 g_candidate_bundle = second;
378 g_candidate_status.available = 1U;
379 g_candidate_status.stable = 1U;
380 g_candidate_status.crc32 = second_crc;
381 g_candidate_status.equal_to_active =
382 lcp_config_bundle_equal(g_candidate_bundle, g_active_bundle);
383 g_candidate_status.report = report;
384
385 if ((g_operation == LCP_CONFIG_OPERATION_SCAN_SD) ||
386 (g_operation == LCP_CONFIG_OPERATION_COMPARE_SD))
387 {
388 finish_operation(LCP_CONFIG_SERVICE_OK);
389 return;
390 }
391
392 if (((g_operation == LCP_CONFIG_OPERATION_IMPORT_SD) ||
393 (g_operation == LCP_CONFIG_OPERATION_AUTO_IMPORT)) &&
394 (g_source == LCP_CONFIG_SOURCE_INTERNAL_FLASH) &&
395 (g_candidate_status.equal_to_active != 0U))
396 {
397 finish_operation(LCP_CONFIG_SERVICE_NO_CHANGE);
398 return;
399 }
400
401 begin_write(g_candidate_bundle);
402}

481void lcp_config_service_init(void)
482{
483 if (sam3x_internal_flash_config_size() != CONFIG_RESERVED_TOTAL_BYTES)
484 {
485 memset(g_slot_status, 0, sizeof(g_slot_status));
486 activate_defaults();
487 g_operation = LCP_CONFIG_OPERATION_NONE;
488 g_result = LCP_CONFIG_SERVICE_INVALID_STATE;
489 g_step = CONFIG_STEP_IDLE;
490 g_auto_import_pending = 0U;
491 return;
492 }
493
494 memset(g_slot_bundle, 0, sizeof(g_slot_bundle));
495 memset(g_erased_page, 0xFF, sizeof(g_erased_page));
496 reset_candidate_status();
497 (void)read_slot(LCP_CONFIG_SLOT_A);
498 (void)read_slot(LCP_CONFIG_SLOT_B);
499
500 const uint8_t a_valid = g_slot_status[0].valid;
501 const uint8_t b_valid = g_slot_status[1].valid;
502
503 if ((a_valid != 0U) && (b_valid != 0U))
504 {
505 activate_slot(
506 (sequence_newer(g_slot_status[1].sequence,
507 g_slot_status[0].sequence) != 0U)
508 ? LCP_CONFIG_SLOT_B
509 : LCP_CONFIG_SLOT_A);
510 }
511 else if (a_valid != 0U)
512 {
513 activate_slot(LCP_CONFIG_SLOT_A);
514 }
515 else if (b_valid != 0U)
516 {
517 activate_slot(LCP_CONFIG_SLOT_B);
518 }
519 else
520 {
521 activate_defaults();
522 }
523
524 g_operation = LCP_CONFIG_OPERATION_NONE;
525 g_result = LCP_CONFIG_SERVICE_IDLE;
526 g_step = CONFIG_STEP_IDLE;
527 g_last_flash_result = SAM3X_INTERNAL_FLASH_OK;
528 g_auto_import_pending =
529 (g_source == LCP_CONFIG_SOURCE_DEFAULTS) ? 1U : 0U;
530}

532void lcp_config_service_poll(void)
533{
534 start_auto_import_if_ready();
535
536 switch (g_step)
537 {
538 case CONFIG_STEP_SD_READ_FIRST:
539 poll_sd_first();
540 break;
541 case CONFIG_STEP_SD_READ_SECOND:
542 poll_sd_second();
543 break;
544 case CONFIG_STEP_WRITE_RECORD:
545 poll_write_record();
546 break;
547 case CONFIG_STEP_WRITE_COMMIT:
548 poll_write_commit();
549 break;
550 case CONFIG_STEP_VERIFY_SLOT:
551 poll_verify_slot();
552 break;
553 case CONFIG_STEP_ERASE_FIRST:
554 poll_erase_slot(g_erase_first, CONFIG_STEP_ERASE_SECOND);
555 break;
556 case CONFIG_STEP_ERASE_SECOND:
557 poll_erase_slot(g_erase_second, CONFIG_STEP_IDLE);
558
559 if (g_step == CONFIG_STEP_IDLE)
560 {
561 activate_defaults();
562 g_auto_import_pending = 0U;
563 g_result = LCP_CONFIG_SERVICE_OK;
564 }
565 break;
566 case CONFIG_STEP_IDLE:
567 default:
568 break;
569 }
570}

662LcpConfigServiceResult lcp_config_service_request_rollback(void)
663{
664 if (g_step != CONFIG_STEP_IDLE)
665 {
666 return LCP_CONFIG_SERVICE_BUSY;
667 }
668
669 if (g_active_slot == LCP_CONFIG_SLOT_NONE)
670 {
671 g_operation = LCP_CONFIG_OPERATION_ROLLBACK;
672 g_result = LCP_CONFIG_SERVICE_NO_ROLLBACK;
673 return g_result;
674 }
675
676 const LcpConfigSlotId previous = other_slot(g_active_slot);
677
678 if (g_slot_status[slot_index(previous)].valid == 0U)
679 {
680 g_operation = LCP_CONFIG_OPERATION_ROLLBACK;
681 g_result = LCP_CONFIG_SERVICE_NO_ROLLBACK;
682 return g_result;
683 }
684
685 g_operation = LCP_CONFIG_OPERATION_ROLLBACK;
686 g_result = LCP_CONFIG_SERVICE_IN_PROGRESS;
687 begin_write(g_slot_bundle[slot_index(previous)]);
688 return g_result;
689}

691LcpConfigServiceResult lcp_config_service_request_erase(void)
692{
693 if (g_step != CONFIG_STEP_IDLE)
694 {
695 return LCP_CONFIG_SERVICE_BUSY;
696 }
697
698 g_operation = LCP_CONFIG_OPERATION_ERASE;
699 g_result = LCP_CONFIG_SERVICE_IN_PROGRESS;
