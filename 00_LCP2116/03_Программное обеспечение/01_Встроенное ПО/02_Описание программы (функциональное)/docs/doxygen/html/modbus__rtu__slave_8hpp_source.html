1/**
2 * @file modbus_rtu_slave.hpp
3 * @brief Неблокирующий минимальный Modbus RTU slave для примеров Lorentz.
4 *
5 * Реализация предназначена как ясный reusable-пример альтернативной роли
6 * одного S1..S4. Один физический half-duplex UART нельзя одновременно передать
7 * этому slave и ModbusRtuMaster. Baseline поддерживает только FC03.
8 */
9

16/** Максимальный размер входного RTU-кадра минимального slave. */
17static const uint16_t MODBUS_RTU_SLAVE_RX_CAPACITY = 64U;
18
19/** Максимальный размер выходного RTU-кадра минимального slave. */
20static const uint16_t MODBUS_RTU_SLAVE_TX_CAPACITY = 64U;
21
22/** Максимальное ожидание физического окончания ответа. */
23static const uint32_t MODBUS_RTU_SLAVE_TX_TIMEOUT_MS = 1000U;
24
25/** @brief Внутреннее состояние неблокирующего slave. */
26enum ModbusRtuSlaveState : uint8_t
27{
28 MODBUS_RTU_SLAVE_RECEIVE = 0U, /**< Приём запроса и ожидание межкадровой паузы. */
29 MODBUS_RTU_SLAVE_WAIT_TX = 1U /**< Ответ принят transport и физически передаётся. */
30};

32/**
33 * @brief Runtime минимального Modbus RTU slave.
34 *
35 * `transport` и `holding_registers` не принадлежат объекту и должны жить всё
36 * время его использования. Карта регистров читается непосредственно при
37 * формировании ответа; отдельная копия не создаётся.
38 */
39struct ModbusRtuSlave
40{
41 const ModbusRtuTransport* transport; /**< Единственный физический UART transport. */
42 uint8_t slave_address; /**< Обычный адрес 1..247. */
43 uint16_t* holding_registers; /**< Внешняя read-only для протокола карта. */
44 uint16_t holding_register_count; /**< Размер карты в 16-битных регистрах. */
45 ModbusRtuSlaveState state; /**< Текущее состояние автомата. */
46 uint16_t rx_length; /**< Накопленная длина запроса. */
47 uint16_t tx_length; /**< Длина сформированного ответа. */
48 uint32_t last_rx_ms; /**< Время последнего принятого байта. */
49 uint32_t tx_deadline_ms; /**< Крайний срок физического TX idle. */
50 uint32_t interframe_gap_ms; /**< Пауза завершения RTU-кадра. */
51 uint32_t request_count; /**< Полные кадры, переданные parser. */
52 uint32_t response_count; /**< Ответы, принятые transport. */
53 uint32_t error_count; /**< CRC, формат, адрес карты или TX error. */
54 uint8_t rx_buffer[MODBUS_RTU_SLAVE_RX_CAPACITY]; /**< Внутренний запрос. */
55 uint8_t tx_buffer[MODBUS_RTU_SLAVE_TX_CAPACITY]; /**< Внутренний ответ. */
56};

80/**
81 * @brief Выполняет один неблокирующий шаг приёма или передачи.
82 *
83 * Цикл RX ограничен доступными байтами и статической ёмкостью. Функция не ждёт
84 * новый байт и должна вызываться регулярно из прикладного service poll.
85 * Состояние WAIT_TX автоматически завершается ошибкой через
86 * MODBUS_RTU_SLAVE_TX_TIMEOUT_MS, если transport не сообщает tx_idle.
87 */
88void modbus_rtu_slave_poll(ModbusRtuSlave& slave);
