169 return static_cast<uint16_t>((value + 1U) % HAL_UART_BUFFER_SIZE);

172static uint8_t hal_uart_ring_push(HalUartRingBuffer& buffer, uint8_t value)
173{
174 const uint16_t next = hal_uart_ring_next(buffer.head);
175
176 if (next == buffer.tail)
177 {
178 return 0U;
179 }
180
181 buffer.data[buffer.head] = value;
182 buffer.head = next;
183
184 return 1U;
185}
186
187static int hal_uart_ring_pop(HalUartRingBuffer& buffer)
188{
189 if (buffer.head == buffer.tail)
190 {
191 return -1;
192 }
193
194 const uint8_t value = buffer.data[buffer.tail];
195 buffer.tail = hal_uart_ring_next(buffer.tail);
196
197 return value;
198}

289static void hal_uart_handle_uart_irq(HalUartPort& port)
290{
291 const uint32_t status = port.uart->UART_SR;
292 const uint32_t interrupt_mask = port.uart->UART_IMR;
293
294 if ((status & UART_SR_RXRDY) != 0U)
295 {
296 (void)hal_uart_ring_push(port.rx, static_cast<uint8_t>(port.uart->UART_RHR));
297 }
298
299 if (((status & UART_SR_TXRDY) != 0U) && ((interrupt_mask & UART_SR_TXRDY) != 0U))
300 {
301 const int value = hal_uart_ring_pop(port.tx);
302
303 if (value >= 0)
304 {
305 port.uart->UART_THR = static_cast<uint8_t>(value);
306 }
307 else
308 {
309 port.uart->UART_IDR = UART_IDR_TXRDY;
310 port.uart->UART_IER = UART_IER_TXEMPTY;
311 }
312 }
313
314 if (((status & UART_SR_TXEMPTY) != 0U) && ((interrupt_mask & UART_SR_TXEMPTY) != 0U))
315 {
316 port.uart->UART_IDR = UART_IDR_TXEMPTY;
317 hal_uart_set_receive(port);
318 }
319
320 if ((status & (UART_SR_OVRE | UART_SR_FRAME)) != 0U)
321 {
322 ++port.error_count;
323 port.uart->UART_CR = UART_CR_RSTSTA;
324 }
325}

327static void hal_uart_handle_usart_irq(HalUartPort& port)
328{
329 const uint32_t status = port.usart->US_CSR;
330 const uint32_t interrupt_mask = port.usart->US_IMR;
331
332 if ((status & US_CSR_RXRDY) != 0U)
333 {
334 (void)hal_uart_ring_push(port.rx, static_cast<uint8_t>(port.usart->US_RHR));
335 }
336
337 if (((status & US_CSR_TXRDY) != 0U) && ((interrupt_mask & US_CSR_TXRDY) != 0U))
338 {
339 const int value = hal_uart_ring_pop(port.tx);
340
341 if (value >= 0)
342 {
343 port.usart->US_THR = static_cast<uint8_t>(value);
344 }
345 else
346 {
347 port.usart->US_IDR = US_IDR_TXRDY;
348 port.usart->US_IER = US_IER_TXEMPTY;
349 }
350 }
351
352 if (((status & US_CSR_TXEMPTY) != 0U) && ((interrupt_mask & US_CSR_TXEMPTY) != 0U))
353 {
354 port.usart->US_IDR = US_IDR_TXEMPTY;
355 hal_uart_set_receive(port);
356 }
357
358 if ((status & (US_CSR_OVRE | US_CSR_FRAME | US_CSR_PARE)) != 0U)
359 {
360 ++port.error_count;
361 port.usart->US_CR = US_CR_RSTSTA;
362 }

394void hal_uart_begin(HalUartId port_id, uint32_t baudrate, HalUartFrame frame)
395{
396 HalUartPort* port = hal_uart_get_port(port_id);
397
398 if ((port == 0) || (baudrate == 0U))
399 {
400 return;
401 }
402
403 hal_uart_configure_pins(port_id);
404 hal_uart_enable_peripheral_clock(port->peripheral_id);
405
406 hal_uart_ring_reset(port->rx);
407 hal_uart_ring_reset(port->tx);
408 port->error_count = 0U;
409
410 if (port->is_usart == 0U)
411 {
412 port->uart->UART_PTCR = UART_PTCR_RXTDIS | UART_PTCR_TXTDIS;
413 port->uart->UART_CR = UART_CR_RSTRX | UART_CR_RSTTX | UART_CR_RXDIS | UART_CR_TXDIS;
414 port->uart->UART_IDR = 0xFFFFFFFFU;
415 port->uart->UART_MR = hal_uart_mode_from_frame(frame) | UART_MR_CHMODE_NORMAL;
416 port->uart->UART_BRGR = (SystemCoreClock / baudrate) >> 4U;
417 port->uart->UART_IER = UART_IER_RXRDY | UART_IER_OVRE | UART_IER_FRAME;
418 port->uart->UART_CR = UART_CR_RXEN | UART_CR_TXEN;
419 }
420 else
421 {
422 port->usart->US_PTCR = US_PTCR_RXTDIS | US_PTCR_TXTDIS;
423 port->usart->US_CR = US_CR_RSTRX | US_CR_RSTTX | US_CR_RXDIS | US_CR_TXDIS;
424 port->usart->US_IDR = 0xFFFFFFFFU;
425 port->usart->US_MR = hal_usart_mode_from_frame(frame);
426 port->usart->US_BRGR = (SystemCoreClock / baudrate) >> 4U;
427 port->usart->US_IER = US_IER_RXRDY | US_IER_OVRE | US_IER_FRAME | US_IER_PARE;
428 port->usart->US_CR = US_CR_RXEN | US_CR_TXEN;
429 }
430
431 NVIC_SetPriority(port->irq, 1U);
432 NVIC_EnableIRQ(port->irq);
433
434 port->initialized = 1U;
435 hal_uart_set_receive(*port);
436}

501size_t hal_uart_write(HalUartId port_id, uint8_t value)
502{
503 HalUartPort* port = hal_uart_get_port(port_id);
504
505 if ((port == 0) || (port->initialized == 0U))
506 {
507 return 0U;
508 }
509
510 __disable_irq();
511
512 const uint8_t pushed = hal_uart_ring_push(port->tx, value);
513
514 if (pushed != 0U)
515 {
516 hal_uart_set_transmit(*port);
517
518 if (port->is_usart == 0U)
519 {
520 port->uart->UART_IDR = UART_IDR_TXEMPTY;
521 port->uart->UART_IER = UART_IER_TXRDY;
522 }
523 else
524 {
525 port->usart->US_IDR = US_IDR_TXEMPTY;
526 port->usart->US_IER = US_IER_TXRDY;
527 }
528 }
529
530 __enable_irq();
531
532 return (pushed != 0U) ? 1U : 0U;
533}

535void hal_uart_flush(HalUartId port_id)
536{
537 HalUartPort* port = hal_uart_get_port(port_id);
538
539 if (port == 0)
540 {
541 return;
542 }
543
544 for (;;)
545 {
546 const uint8_t tx_buffer_empty = (port->tx.head == port->tx.tail) ? 1U : 0U;
547 uint8_t hardware_empty;
548
549 if (port->is_usart == 0U)
550 {
551 hardware_empty = ((port->uart->UART_SR & UART_SR_TXEMPTY) != 0U) ? 1U : 0U;
552 }
553 else
554 {
555 hardware_empty = ((port->usart->US_CSR & US_CSR_TXEMPTY) != 0U) ? 1U : 0U;
556 }
557
558 if ((tx_buffer_empty != 0U) && (hardware_empty != 0U))
559 {
560 break;
561 }
562
563 hal_uart_cpu_relax();
564 }
