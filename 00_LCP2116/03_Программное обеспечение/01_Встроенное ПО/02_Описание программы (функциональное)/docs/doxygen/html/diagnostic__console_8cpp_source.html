114char ascii_to_lower(char value)
115{
116 if ((value >= 'A') && (value <= 'Z'))
117 {
118 return static_cast<char>(value + ('a' - 'A'));
119 }
120
121 return value;
122}

124void command_to_lower(char* command)
125{
126 if (command == 0)
127 {
128 return;
129 }
130
131 for (uint16_t index = 0U; command[index] != '\0'; ++index)
132 {
133 command[index] = ascii_to_lower(command[index]);
134 }
135}

137void trim_command(char* command)
138{
139 if (command == 0)
140 {
141 return;
142 }
143
144 uint16_t start = 0U;
145
146 while ((command[start] == ' ') || (command[start] == '\t'))
147 {
148 ++start;
149 }
150
151 if (start != 0U)
152 {
153 uint16_t destination = 0U;
154
155 while (command[start] != '\0')
156 {
157 command[destination++] = command[start++];
158 }
159
160 command[destination] = '\0';
161 }
162
163 uint16_t length = 0U;
164
165 while (command[length] != '\0')
166 {
167 ++length;
168 }
169
170 while ((length > 0U) &&
171 ((command[length - 1U] == ' ') ||
172 (command[length - 1U] == '\t')))
173 {
174 command[--length] = '\0';
175 }
176}

178uint8_t command_equals(const char* command, const char* expected)
179{
180 return ((command != 0) && (expected != 0) &&
181 (strcmp(command, expected) == 0)) ? 1U : 0U;
182}

298void print_rtos_summary(void)
299{
300 const uint32_t heap_total = app_rtos_heap_total_bytes();
301 const uint32_t heap_free = app_rtos_free_heap_bytes();
302 const uint32_t heap_used = (heap_total >= heap_free) ?
303 (heap_total - heap_free) : 0U;
304 const uint32_t stack_total = app_rtos_lcp_stack_total_bytes();
305 const uint32_t stack_total_words = app_rtos_lcp_stack_total_words();
306 const uint32_t stack_free_words = app_rtos_lcp_stack_free_words();
307 const uint32_t stack_word_bytes = (stack_total_words != 0U) ?
308 (stack_total / stack_total_words) : 0U;
309 const uint32_t stack_free = stack_free_words * stack_word_bytes;
310 const uint32_t stack_peak_used = (stack_total >= stack_free) ?
311 (stack_total - stack_free) : 0U;
312
313 diagnostic_print_section("RTOS SUMMARY");
314 SerialUSB.print("scheduler = ");
315 SerialUSB.print((app_rtos_scheduler_running() != 0U) ?
316 "running" : "not running");
317 SerialUSB.print(", tick_count = ");
318 SerialUSB.print(static_cast<unsigned long>(app_rtos_tick_count()));
319 SerialUSB.print("\r\nheap_used = ");
320 SerialUSB.print(static_cast<unsigned long>(heap_used));
321 SerialUSB.print("/");
322 SerialUSB.print(static_cast<unsigned long>(heap_total));
323 SerialUSB.print(" bytes (");
324 diagnostic_print_percent(heap_used, heap_total);
325 SerialUSB.print(")\r\ntask_stack_peak_used = ");
326 SerialUSB.print(static_cast<unsigned long>(stack_peak_used));
327 SerialUSB.print("/");
328 SerialUSB.print(static_cast<unsigned long>(stack_total));
329 SerialUSB.print(" bytes (");
330 diagnostic_print_percent(stack_peak_used, stack_total);
331 SerialUSB.print(")\r\n");

460uint8_t execute_local_command(const char* command)
461{
462 if ((command_equals(command, "help") != 0U) ||
463 (command_equals(command, "?") != 0U))
464 {
465 print_help();
466 return 1U;
467 }
468
469 if ((command_equals(command, "version") != 0U) ||
470 (command_equals(command, "ver") != 0U))
471 {
472 print_version_report();
473 return 1U;
474 }
475
476 if (command_equals(command, "status") != 0U)
477 {
478 print_status();
479 return 1U;
480 }
481
482 if (command_equals(command, "rtos") != 0U)
483 {
484 print_rtos_status();
485 return 1U;
486 }
487
488 if (command_equals(command, "sc16is") != 0U)
489 {
490 lcp_sc16is_print_probe_report();
491 return 1U;
492 }
493
494 if (command_equals(command, "rs485") != 0U)
495 {
496 print_rs485_status();
497 return 1U;
498 }
499
500 if (command_equals(command, "sd") != 0U)
501 {
502 sd_card_test_print_report();
503 return 1U;
504 }
505
506 if (command_equals(command, "sd test") != 0U)
507 {
508 sd_card_test_run_file_test();
509 return 1U;
510 }
511
512 if (command_equals(command, "battery") != 0U)
513 {
514 battery_status_print_report();
515 return 1U;
516 }
517
518 if (command_equals(command, "uptime") != 0U)
519 {
520 print_uptime_report();

526 g_periodic_report_enabled = 1U;

528 SerialUSB.print("periodic status = enabled\r\ninterval_ms = ");

535 if (command_equals(command, "periodic off") != 0U)
536 {
537 g_periodic_report_enabled = 0U;
538 SerialUSB.print("periodic status = disabled\r\n");
539 return 1U;
540 }
541
542 return 0U;
543}

545void execute_command(char* command)
546{
547 trim_command(command);
548 command_to_lower(command);
549
550 if ((command == 0) || (command[0] == '\0'))
551 {
552 return;
553 }
554
555 if ((execute_local_command(command) != 0U) ||
556 (config_status_handle_command(command) != 0U) ||
557 (field_status_handle_command(command) != 0U) ||
558 (x2x_status_handle_command(command) != 0U) ||
559 (ethernet_status_handle_command(command) != 0U) ||
560 (rtc_status_handle_command(command) != 0U) ||
561 (watchdog_status_handle_command(command) != 0U))
562 {
563 return;
564 }
565
566 SerialUSB.print("unknown command: ");
567 SerialUSB.print(command);
568 SerialUSB.print("\r\ntype 'help' or '?' for the command list\r\n");
569}

571void process_rx_byte(uint8_t value)
572{
573 if ((value == '\r') || (value == '\n'))
574 {
575 if (g_command_length != 0U)
576 {
577 g_command_buffer[g_command_length] = '\0';
578 execute_command(g_command_buffer);
579 g_command_length = 0U;
580 }
581
582 return;
583 }
584
585 if (g_command_length < (DIAGNOSTIC_COMMAND_BUFFER_SIZE - 1U))
586 {
587 g_command_buffer[g_command_length++] = static_cast<char>(value);
588 }
589 else
590 {
591 g_command_length = 0U;
592 SerialUSB.print("command rejected: input exceeds 63 characters\r\n");
593 }
594}

596void poll_usb_open_state(void)
597{
598 const uint8_t usb_open = SerialUSB ? 1U : 0U;
599
600 if ((usb_open != 0U) && (g_usb_was_open == 0U))
601 {
602 SerialUSB.print("\r\nLCP firmware ready. Type 'help' or '?'.\r\n");
603 g_command_length = 0U;
604 }
605
606 if (usb_open == 0U)
607 {
608 g_command_length = 0U;
609 }
610
611 g_usb_was_open = usb_open;
612}

614void poll_periodic_report(void)
615{
616 if (g_periodic_report_enabled == 0U)
617 {
618 return;
619 }
620
621 const uint32_t now_ms = millis();
622
623 if ((uint32_t)(now_ms - g_last_periodic_report_ms) >=
624 DIAGNOSTIC_PERIODIC_REPORT_MS)
625 {
626 g_last_periodic_report_ms = now_ms;
627 print_status();
628 }
629}

640void diagnostic_console_poll(void)
641{
642 poll_usb_open_state();
643
644 if (!SerialUSB)
645 {
646 return;
647 }
648
649 while (SerialUSB.available() > 0)
650 {
651 const int value = SerialUSB.read();
652
653 if (value >= 0)
654 {
655 process_rx_byte(static_cast<uint8_t>(value));
656 }
657 }
658
659 poll_periodic_report();
660}
