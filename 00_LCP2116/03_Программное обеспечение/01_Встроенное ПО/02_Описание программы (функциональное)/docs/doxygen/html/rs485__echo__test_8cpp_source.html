16constexpr uint32_t RS485_ECHO_BAUDRATE = 9600U;
17constexpr uint32_t RS485_INTERFRAME_GAP_MS = 5U;
18constexpr uint32_t RS485_RESPONSE_DELAY_MS = 10U;
19constexpr uint32_t RS485_TX_TIMEOUT_MS = 1000U;
20constexpr uint8_t RS485_ECHO_MAX_BYTES_PER_POLL = 16U;
21constexpr uint16_t RS485_ECHO_BUFFER_SIZE = 128U;

23struct Rs485EchoState
24{
25 uint8_t buffer[RS485_ECHO_BUFFER_SIZE]; /**< Полный принятый кадр. */
26 uint16_t length; /**< Длина кадра. */
27 uint16_t tx_offset; /**< Первый ещё не переданный байт. */
28 uint32_t last_rx_ms; /**< Время последнего RX-байта. */
29 uint32_t response_due_ms; /**< Момент разрешения ответа. */
30 uint32_t tx_deadline_ms; /**< Крайний срок постановки всего ответа. */
31 uint8_t response_pending; /**< Кадр завершён и ожидает/выполняет TX. */
32};

34Rs485EchoState g_echo = { {0U}, 0U, 0U, 0U, 0U, 0U, 0U };
35uint8_t g_x2x_echo_enabled = 1U;

37uint8_t deadline_reached(uint32_t now_ms, uint32_t deadline_ms)
38{
39 return (static_cast<int32_t>(now_ms - deadline_ms) >= 0) ? 1U : 0U;
40}

42void drop_frame(uint32_t now_ms)
43{
44 g_echo.length = 0U;
45 g_echo.tx_offset = 0U;
46 g_echo.last_rx_ms = now_ms;
47 g_echo.response_due_ms = 0U;
48 g_echo.tx_deadline_ms = 0U;
49 g_echo.response_pending = 0U;
50}
51
52void accept_rx(uint32_t now_ms)
53{
54 if (g_echo.tx_offset != 0U)
55 {
56 return;
57 }
58
59 uint8_t processed = 0U;
60
61 while ((Serial.available() > 0) &&
62 (processed < RS485_ECHO_MAX_BYTES_PER_POLL))
63 {
64 const int value = Serial.read();
65
66 if (value < 0)
67 {
68 break;
69 }
70
71 if (g_echo.length >= RS485_ECHO_BUFFER_SIZE)
72 {
73 drop_frame(now_ms);
74 return;
75 }
76
77 g_echo.buffer[g_echo.length++] = static_cast<uint8_t>(value);
78 g_echo.last_rx_ms = now_ms;
79 g_echo.response_pending = 0U;
80 g_echo.response_due_ms = 0U;
81 g_echo.tx_deadline_ms = 0U;
82 ++processed;
83 }
84}
85
86void arm_response_if_frame_complete(uint32_t now_ms)
87{
88 if ((g_echo.length == 0U) || (g_echo.response_pending != 0U))
89 {
90 return;
91 }
92
93 if ((uint32_t)(now_ms - g_echo.last_rx_ms) < RS485_INTERFRAME_GAP_MS)
94 {
95 return;
96 }
97
98 g_echo.response_due_ms = now_ms + RS485_RESPONSE_DELAY_MS;
99 g_echo.tx_deadline_ms = now_ms + RS485_TX_TIMEOUT_MS;
100 g_echo.response_pending = 1U;
101 g_echo.tx_offset = 0U;
102}
103
104void send_if_response_due(uint32_t now_ms)
105{
106 if (g_echo.response_pending == 0U)
107 {
108 return;
109 }
110
111 if (deadline_reached(now_ms, g_echo.tx_deadline_ms) != 0U)
112 {
113 drop_frame(now_ms);
114 return;
115 }
116
117 if (static_cast<int32_t>(now_ms - g_echo.response_due_ms) < 0)
118 {
119 return;
120 }
121
122 const size_t remaining =
123 static_cast<size_t>(g_echo.length - g_echo.tx_offset);
124 const size_t written = Serial.write(
125 &g_echo.buffer[g_echo.tx_offset],
126 remaining);
127
128 g_echo.tx_offset = static_cast<uint16_t>(g_echo.tx_offset + written);
129
130 if (g_echo.tx_offset >= g_echo.length)
131 {
132 drop_frame(now_ms);
133 }
134}
135}

144void rs485_echo_test_poll(void)
145{
146 if (g_x2x_echo_enabled == 0U)
147 {
148 return;
149 }
150
151 const uint32_t now_ms = millis();
152 accept_rx(now_ms);
153 arm_response_if_frame_complete(now_ms);
154 send_if_response_due(now_ms);
155}
