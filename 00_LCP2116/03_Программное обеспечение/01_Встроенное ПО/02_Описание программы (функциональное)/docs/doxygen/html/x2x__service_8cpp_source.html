40void reset_runtime_state(void)
41{
42 modbus_rtu_master_reset(g_modbus_master);
43
44 for (uint8_t index = 0U; index <= X2X_MAX_MODULES; ++index)
45 {
46 x2x_module_runtime_reset(g_module_runtime[index]);
47 }
48
49 memset(g_register_buffer, 0, sizeof(g_register_buffer));
50 memset(&g_waveform, 0, sizeof(g_waveform));
51 g_cycle_in_progress = 0U;
52 g_current_slave =
53 (x2x_registry_module_count() > 0U) ? 1U : 0U;
54 g_next_cycle_ms = millis();
55}

57X2XConfigResult apply_config_now(void)
58{
59 X2XConfig candidate_config;
60 X2XConfigError candidate_error;
61 x2x_config_reset(candidate_config, candidate_error);
62
63 g_reload_requested = 0U;
64
65 if (lcp_config_bundle_x2x(lcp_config_service_active(),
66 candidate_config) == 0U)
67 {
68 g_last_config_result = X2X_CONFIG_INVALID_DEVICE_ID;
69 g_last_config_error.result = X2X_CONFIG_INVALID_DEVICE_ID;
70 g_last_config_error.physical_line = 0U;
71 g_last_config_error.value = 0L;
72 return g_last_config_result;
73 }
74
75 const X2XRegistryResult registry_result =
76 x2x_registry_build(candidate_config, X2X_DEFAULT_ASDU);
77
78 if (registry_result != X2X_REGISTRY_OK)
79 {
80 g_last_registry_result = registry_result;
81 g_last_config_result = X2X_CONFIG_REGISTRY_BUILD_FAILED;
82 g_last_config_error.result = X2X_CONFIG_REGISTRY_BUILD_FAILED;
83 g_last_config_error.physical_line = 0U;
84 g_last_config_error.value = static_cast<int32_t>(registry_result);
85 return g_last_config_result;
86 }
87
88 g_active_config = candidate_config;
89 g_last_config_error = candidate_error;
90 g_last_config_result = X2X_CONFIG_OK;
91 g_last_registry_result = X2X_REGISTRY_OK;
92 g_config_loaded = 1U;
93 g_applied_config_generation = lcp_config_service_generation();
94 reset_runtime_state();
95 return X2X_CONFIG_OK;
96}

112void advance_slave(uint32_t now_ms)
113{
114 const uint8_t module_count = x2x_registry_module_count();
115
116 if (module_count == 0U)
117 {
118 g_current_slave = 0U;
119 }
120 else
121 {
122 ++g_current_slave;
123
124 if ((g_current_slave == 0U) || (g_current_slave > module_count))
125 {
126 g_current_slave = 1U;
127 }
128 }
129
130 g_cycle_in_progress = 0U;
131 g_next_cycle_ms = now_ms + X2X_MODULE_CYCLE_GAP_MS;
132}

134void poll_active_module(uint32_t now_ms)
135{
136 X2XDeviceHeader* device =
137 x2x_registry_get_by_slave(g_current_slave);
138
139 if (device == 0)
140 {
141 advance_slave(now_ms);
142 return;
143 }
144
145 const X2XModuleDescriptor* descriptor =
146 x2x_catalog_find(device->type);
147
148 if ((descriptor == 0) || (descriptor->poll == 0))
149 {
150 x2x_module_mark_failure(*device,
151 MODBUS_RTU_RESULT_INVALID_ARGUMENT,
152 0U);
153 advance_slave(now_ms);
154 return;
155 }
156
157 X2XModuleContext context =
158 {
159 &g_modbus_master,
160 &g_module_runtime[g_current_slave],
161 &g_waveform,
162 g_register_buffer,
163 MODBUS_RTU_MASTER_MAX_READ_REGISTERS,
164 now_ms
165 };
166
167 const X2XModulePollResult result =
168 descriptor->poll(device, context);
169
170 if (result == X2X_MODULE_POLL_CYCLE_COMPLETE)
171 {
172 advance_slave(now_ms);
173 }
174}

200void x2x_service_poll(void)
201{
202 const uint32_t now_ms = millis();
203
204 if (g_applied_config_generation != lcp_config_service_generation())
205 {
206 g_reload_requested = 1U;
207 }
208
209 if (g_cycle_in_progress != 0U)
210 {
211 modbus_rtu_master_poll(g_modbus_master);
212 poll_active_module(now_ms);
213 }
214
215 if ((g_reload_requested != 0U) &&
216 (g_cycle_in_progress == 0U) &&
217 (modbus_rtu_master_busy(g_modbus_master) == 0U))
218 {
219 (void)apply_config_now();
220 complete_pause_if_safe();
221 return;
222 }
223
224 complete_pause_if_safe();
225
226 if ((g_config_loaded == 0U) ||
227 (g_paused != 0U) ||
228 (g_pause_requested != 0U) ||
229 (x2x_registry_module_count() == 0U))
230 {
231 return;
232 }
233
234 if (g_cycle_in_progress == 0U)
235 {
236 if (static_cast<int32_t>(now_ms - g_next_cycle_ms) < 0)
237 {
238 return;
239 }
240
241 g_cycle_in_progress = 1U;
242 poll_active_module(now_ms);
243 }
244}
