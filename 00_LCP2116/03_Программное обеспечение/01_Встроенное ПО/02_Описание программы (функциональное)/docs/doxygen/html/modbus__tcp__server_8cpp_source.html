32void remove_request(ModbusTcpServer& server, uint16_t request_length)
33{
34 if (request_length >= server.rx_length)
35 {
36 server.rx_length = 0U;
37 return;
38 }
39
40 const uint16_t remaining =
41 static_cast<uint16_t>(server.rx_length - request_length);
42 memmove(server.rx_buffer,
43 &server.rx_buffer[request_length],
44 remaining);
45 server.rx_length = remaining;
46}

48void copy_mbap_prefix(ModbusTcpServer& server, uint16_t length_field)
49{
50 server.tx_buffer[0] = server.rx_buffer[0];

76uint8_t modbus_tcp_server_append(ModbusTcpServer& server,

98uint8_t modbus_tcp_server_process(ModbusTcpServer& server,
99 ModbusTcpReadHoldingHandler read_handler,
100 void* context)
101{
102 if ((server.tx_length != 0U) || (server.rx_length < MODBUS_TCP_HEADER_WITH_UNIT))
103 {
104 return 0U;
105 }
106
107 const uint16_t protocol_id = read_u16_be(&server.rx_buffer[2]);
108 const uint16_t length_field = read_u16_be(&server.rx_buffer[4]);
109 const uint32_t expected_length = 6UL + static_cast<uint32_t>(length_field);
110
111 if ((protocol_id != 0U) ||
112 (length_field < 2U) ||
113 (expected_length > MODBUS_TCP_ADU_CAPACITY))
114 {
115 server.rx_length = 0U;
116 ++server.malformed_count;
117 return 0U;
118 }
119
120 if (server.rx_length < expected_length)
121 {
122 return 0U;
123 }
124
125 const uint16_t request_length = static_cast<uint16_t>(expected_length);
126 const uint8_t function_code = server.rx_buffer[7];
127 ++server.request_count;
128
129 if (function_code != MODBUS_FUNCTION_READ_HOLDING)
130 {
131 build_exception(server,
132 function_code,
133 MODBUS_EXCEPTION_ILLEGAL_FUNCTION);
134 remove_request(server, request_length);
135 return 1U;
136 }
137
138 if ((request_length != MODBUS_TCP_READ_REQUEST_LENGTH) ||
139 (length_field != 6U))
140 {
141 build_exception(server,
142 function_code,
143 MODBUS_EXCEPTION_ILLEGAL_VALUE);
144 remove_request(server, request_length);
145 return 1U;
146 }
147
148 const uint16_t start_address = read_u16_be(&server.rx_buffer[8]);
149 const uint16_t register_count = read_u16_be(&server.rx_buffer[10]);
150
151 if ((register_count == 0U) ||
152 (register_count > MODBUS_TCP_MAX_READ_REGISTERS))
153 {
154 build_exception(server,
155 function_code,
156 MODBUS_EXCEPTION_ILLEGAL_VALUE);
157 remove_request(server, request_length);
158 return 1U;
159 }
160
161 uint16_t values[MODBUS_TCP_MAX_READ_REGISTERS];
162 uint8_t exception_code = MODBUS_EXCEPTION_ILLEGAL_ADDRESS;
163
164 if (read_handler != 0)
165 {
166 exception_code = read_handler(context,
167 start_address,
168 register_count,
169 values);
170 }
171
172 if (exception_code != 0U)
173 {
174 build_exception(server, function_code, exception_code);
175 remove_request(server, request_length);
176 return 1U;
177 }
178
179 const uint16_t response_length_field =
180 static_cast<uint16_t>(3U + (register_count * 2U));
181 const uint16_t response_length =
182 static_cast<uint16_t>(6U + response_length_field);
183
184 if (response_length > MODBUS_TCP_ADU_CAPACITY)
185 {
186 build_exception(server,
187 function_code,
188 MODBUS_EXCEPTION_ILLEGAL_VALUE);
189 remove_request(server, request_length);
190 return 1U;
191 }
192
193 copy_mbap_prefix(server, response_length_field);
194 server.tx_buffer[7] = MODBUS_FUNCTION_READ_HOLDING;
195 server.tx_buffer[8] = static_cast<uint8_t>(register_count * 2U);
196
197 for (uint16_t index = 0U; index < register_count; ++index)
198 {
199 write_u16_be(&server.tx_buffer[9U + (index * 2U)], values[index]);
200 }
201
202 server.tx_length = response_length;
203 ++server.response_count;
204 remove_request(server, request_length);
205 return 1U;
206}
