116Sc16isProbeResult sc16is_probe_channel(uint32_t chip_select,
117 Sc16isChannel channel)
118{
119 static const uint8_t first = 0x55U;
120 static const uint8_t second = 0xAAU;
121
122 sc16is_write_register(chip_select, channel, SC16IS_REG_SPR, first);
123 const uint8_t first_read =
124 sc16is_read_register(chip_select, channel, SC16IS_REG_SPR);
125
126 sc16is_write_register(chip_select, channel, SC16IS_REG_SPR, second);
127 const uint8_t second_read =
128 sc16is_read_register(chip_select, channel, SC16IS_REG_SPR);
129
130 return ((first_read == first) && (second_read == second)) ?
131 SC16IS_PROBE_PRESENT : SC16IS_PROBE_ABSENT;
132}

134uint8_t sc16is_probe_dual_channel(uint32_t chip_select)
135{
136 static const uint8_t value_a = 0x11U;
137 static const uint8_t value_b = 0x22U;
138
139 if (sc16is_probe_channel(chip_select, SC16IS_CHANNEL_A) !=
140 SC16IS_PROBE_PRESENT)
141 {
142 return 0U;
143 }
144
145 sc16is_write_register(chip_select,
146 SC16IS_CHANNEL_A,
147 SC16IS_REG_SPR,
148 value_a);
149 sc16is_write_register(chip_select,
150 SC16IS_CHANNEL_B,
151 SC16IS_REG_SPR,
152 value_b);
153
154 const uint8_t read_a = sc16is_read_register(chip_select,
155 SC16IS_CHANNEL_A,
156 SC16IS_REG_SPR);
157 const uint8_t read_b = sc16is_read_register(chip_select,
158 SC16IS_CHANNEL_B,
159 SC16IS_REG_SPR);
160
161 return ((read_a == value_a) && (read_b == value_b)) ? 1U : 0U;
162}

164void sc16is_begin(uint32_t chip_select,
165 Sc16isChannel channel,
166 uint32_t baudrate,
167 HalUartFrame frame)
168{
169 if (baudrate == 0U)
170 {
171 return;
172 }
173
174 const uint32_t divisor = SC16IS_XTAL_HZ / (baudrate * 16UL);
175
176 sc16is_write_register(chip_select, channel, SC16IS_REG_IER, 0x00U);
177 sc16is_write_register(chip_select,
178 channel,
179 SC16IS_REG_LCR,
180 SC16IS_LCR_ENABLE_DIVISOR);
181 sc16is_write_register(chip_select,
182 channel,
183 SC16IS_REG_DLL,
184 static_cast<uint8_t>(divisor & 0xFFU));
185 sc16is_write_register(chip_select,
186 channel,
187 SC16IS_REG_DLH,
188 static_cast<uint8_t>((divisor >> 8U) & 0xFFU));
189 sc16is_write_register(chip_select,
190 channel,
191 SC16IS_REG_LCR,
192 sc16is_lcr_for_frame(frame));
193 sc16is_write_register(chip_select,
194 channel,
195 SC16IS_REG_FCR,
196 SC16IS_FCR_ENABLE_FIFO |
197 SC16IS_FCR_RX_FIFO_RESET |
198 SC16IS_FCR_TX_FIFO_RESET);
199 sc16is_write_register(chip_select, channel, SC16IS_REG_MCR, 0x00U);
200 sc16is_write_register(chip_select,
201 channel,
202 SC16IS_REG_EFCR,
203 SC16IS_EFCR_RS485_ENABLE_RTS |
204 SC16IS_EFCR_RTS_INVERT |
205 SC16IS_EFCR_RS485_MODE);
206}

229size_t sc16is_write(uint32_t chip_select,
230 Sc16isChannel channel,
231 uint8_t value)

243size_t sc16is_write_buffer(uint32_t chip_select,
244 Sc16isChannel channel,
245 const uint8_t* buffer,
246 size_t length)
247{
248 if (buffer == 0)
249 {
250 return 0U;
251 }
252
253 size_t written = 0U;
254
255 while (written < length)
256 {
257 uint8_t available = sc16is_tx_available(chip_select, channel);
258
259 if (available == 0U)
260 {
261 break;
262 }
263
264 while ((available > 0U) && (written < length))
265 {
266 sc16is_write_register(chip_select,
267 channel,
268 SC16IS_REG_THR,
269 buffer[written]);
270 ++written;
271 --available;
272 }
273 }
274
275 return written;
276}
