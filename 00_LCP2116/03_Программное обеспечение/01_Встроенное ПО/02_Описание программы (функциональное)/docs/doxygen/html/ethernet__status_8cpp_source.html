20void print_hex8(uint8_t value)
21{
22 static const char hex[] = "0123456789ABCDEF";
23 char text[5];
24 text[0] = '0';
25 text[1] = 'x';
26 text[2] = hex[(value >> 4U) & 0x0FU];
27 text[3] = hex[value & 0x0FU];
28 text[4] = '\0';
29 SerialUSB.print(text);
30}

80void print_interface(LcpEthernetId ethernet_id)
81{
82 const EthernetModbusInterfaceState& state =
83 ethernet_modbus_service_interface(ethernet_id);
84 const uint8_t initialized = state.initialized;
85 const uint8_t version = (initialized != 0U) ?
86 w5500_lite_version(ethernet_id) : 0U;
87 const uint8_t link_up = ((initialized != 0U) &&
88 (state.init_ok != 0U)) ?
89 w5500_lite_link_up(ethernet_id) : 0U;
90
91 diagnostic_print_group(lcp_ethernet_name(ethernet_id));
92
93 diagnostic_print_assignment("initialized");
94 SerialUSB.print((initialized != 0U) ? "yes" : "no");
95 SerialUSB.print(", ");
96 diagnostic_print_assignment("init");
97 SerialUSB.print((state.init_ok != 0U) ? "ok" : "failed");
98 SerialUSB.print(", ");
99 diagnostic_print_assignment("link");
100 SerialUSB.print((link_up != 0U) ? "up" : "down");
101 SerialUSB.print("\r\n");
102
103 diagnostic_print_assignment("VERSIONR");
104 print_hex8(version);
105 SerialUSB.print(", ");
106 diagnostic_print_assignment("socket_status");
107 print_hex8(state.last_socket_status);
108 SerialUSB.print(", ");
109 diagnostic_print_assignment("tcp_port");
110 SerialUSB.print(static_cast<int>(LCP_MODBUS_TCP_PORT));
111 SerialUSB.print("\r\n");
112
113 diagnostic_print_assignment("mac");
114 print_mac(state.config.mac);
115 SerialUSB.print(", ");
116 diagnostic_print_assignment("ip");
117 print_ip(state.config.ip);
118 SerialUSB.print("\r\n");
119
120 diagnostic_print_assignment("subnet");
121 print_ip(state.config.subnet);
122 SerialUSB.print(", ");
123 diagnostic_print_assignment("gateway");
124 print_ip(state.config.gateway);
125 SerialUSB.print("\r\n");
126
127 diagnostic_print_assignment("requests");
128 SerialUSB.print(static_cast<unsigned long>(state.server.request_count));
129 SerialUSB.print(", ");
130 diagnostic_print_assignment("responses");
131 SerialUSB.print(static_cast<unsigned long>(state.server.response_count));
132 SerialUSB.print("\r\n");
133
134 diagnostic_print_assignment("exceptions");
135 SerialUSB.print(static_cast<unsigned long>(state.server.exception_count));
136 SerialUSB.print(", ");

139 SerialUSB.print(", ");
140 diagnostic_print_assignment("transport_errors");

170void print_register_map(void)
171{
172 diagnostic_print_group("Published holding registers");
173 diagnostic_print_assignment("layout");
174 SerialUSB.print("value0, value1, quality\r\n");
175 diagnostic_print_assignment("registers_per_FieldSensor_port");
176 SerialUSB.print("3\r\n");
177
178 for (uint8_t port_index = 0U;
179 port_index < LCP_FIELD_PORT_COUNT;
180 ++port_index)
181 {
182 const uint16_t base = static_cast<uint16_t>(
183 port_index * LCP_MODBUS_TCP_REGISTERS_PER_FIELD_PORT);
184 const uint16_t value0 = ethernet_modbus_service_holding(base);
185 const uint16_t value1 = ethernet_modbus_service_holding(base + 1U);
186 const uint16_t quality = ethernet_modbus_service_holding(base + 2U);
187
188 SerialUSB.print("S");
189 SerialUSB.print(static_cast<int>(port_index + 1U));
190 SerialUSB.print(": ");
191 diagnostic_print_assignment("address");
192 SerialUSB.print(static_cast<unsigned long>(base));
193 SerialUSB.print("..");
194 SerialUSB.print(static_cast<unsigned long>(base + 2U));
195 SerialUSB.print(", ");
196 diagnostic_print_assignment("quality");
197 print_hex16(quality);
198 SerialUSB.print("\r\n ");
199 diagnostic_print_assignment("value0");
200 SerialUSB.print(static_cast<unsigned long>(value0));
201 SerialUSB.print(", ");
202 diagnostic_print_assignment("value1");
203 SerialUSB.print(static_cast<unsigned long>(value1));
204 SerialUSB.print("\r\n ");
205 print_quality_flags(quality);
206 SerialUSB.print("\r\n");
207 }
208}

211void ethernet_status_print_report(void)
212{
213 diagnostic_print_section("ETHERNET MODBUS TCP");
214 diagnostic_print_assignment("service_reload");
215 SerialUSB.print((ethernet_modbus_service_reload_pending() != 0U) ?

220 SerialUSB.print("\r\n");
221 diagnostic_print_assignment("config_generation");
222 SerialUSB.print(static_cast<unsigned long>(
223 lcp_config_service_generation()));
224 SerialUSB.print(", ");
225 diagnostic_print_assignment("function");
226 SerialUSB.print("0x03 read-only\r\n");
227 diagnostic_print_assignment("holding_address");
228 SerialUSB.print("0..11\r\n");
229
230 print_interface(LCP_ETHERNET_1);
231 print_interface(LCP_ETHERNET_2);
232 print_register_map();
233}

235uint8_t ethernet_status_handle_command(const char* command)
236{
237 if (command == 0)
238 {
239 return 0U;
240 }
241
242 if ((strcmp(command, "eth") == 0) ||
243 (strcmp(command, "eth status") == 0))
244 {
245 ethernet_status_print_report();
246 return 1U;
247 }
248
249 if (strcmp(command, "eth reload") == 0)
250 {
251 ethernet_modbus_service_request_reload();
252 SerialUSB.print("Ethernet active configuration reload = pending\r\n");
253 SerialUSB.print("both W5500 interfaces will be reset\r\n");
254 return 1U;
255 }
256
257 return 0U;
