32uint32_t frame_duration_ms(LcpFieldPortId port_id, size_t byte_count)
33{
34 const uint32_t baudrate = g_configs[port_id].baudrate;
35
36 if ((baudrate == 0U) || (byte_count == 0U))
37 {
38 return 0U;
39 }
40
41 return static_cast<uint32_t>(
42 ((byte_count * BITS_PER_CHARACTER_WITH_MARGIN * 1000U) +
43 (baudrate - 1U)) /
44 baudrate);
45}

47void remember_tx_duration(LcpFieldPortId port_id, size_t byte_count)
48{
49 g_tx_not_before_ms[port_id] =
50 millis() + frame_duration_ms(port_id, byte_count) + 1U;
51}
52
53uint8_t transmission_time_elapsed(LcpFieldPortId port_id)
54{
55 return (static_cast<int32_t>(millis() - g_tx_not_before_ms[port_id]) >= 0) ?
56 1U : 0U;
57}

59size_t embedded_write(HalUartId uart_id,
60 LcpFieldPortId port_id,
61 const uint8_t* data,
62 size_t length)
63{
64 if (data == 0)
65 {
66 return 0U;
67 }
68
69 size_t written = 0U;
70
71 while (written < length)
72 {
73 if (hal_uart_write(uart_id, data[written]) == 0U)
74 {
75 break;
76 }
77
78 ++written;
79 }
80
81 remember_tx_duration(port_id, written);
82 return written;
83}

85size_t s1_write(const uint8_t* data, size_t length)
86{
87 if ((g_s1_port.present == 0U) || (data == 0))
88 {
89 return 0U;
90 }
91
92 const size_t written = sc16is_write_buffer(g_s1_port.chip_select,
93 g_s1_port.channel,
94 data,
95 length);
96 remember_tx_duration(LCP_FIELD_PORT_S1, written);
97 return written;
98}

112uint8_t s1_tx_idle(void)
113{
114 if (g_s1_port.present == 0U)
115 {
116 return 1U;
117 }
118
119 if (transmission_time_elapsed(LCP_FIELD_PORT_S1) == 0U)
120 {
121 return 0U;
122 }
123
124 return ((sc16is_line_status(g_s1_port.chip_select, g_s1_port.channel) &
125 SC16IS_LSR_TRANSMITTER_EMPTY) != 0U) ? 1U : 0U;
126}

194void lcp_field_ports_init(const LcpFieldPortConfig configs[LCP_FIELD_PORT_COUNT])
195{
196 if (configs != 0)
197 {
198 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
199 {
200 g_configs[index] = configs[index];
201
202 if (g_configs[index].baudrate == 0U)
203 {
204 g_configs[index].baudrate = DEFAULT_BAUDRATE;
205 }
206 }
207 }
208
209 lcp_sc16is_init_pins();
210 lcp_sc16is_probe();
211 g_s1_port = lcp_sc16is_get_map().s1;
212
213 if (g_s1_port.present != 0U)
214 {
215 sc16is_begin(g_s1_port.chip_select,
216 g_s1_port.channel,
217 g_configs[LCP_FIELD_PORT_S1].baudrate,
218 g_configs[LCP_FIELD_PORT_S1].frame);
219 s1_clear_rx();
220 }
221
222 hal_uart_begin(HAL_UART_PORT_1,
223 g_configs[LCP_FIELD_PORT_S2].baudrate,
224 g_configs[LCP_FIELD_PORT_S2].frame);
225 hal_uart_begin(HAL_UART_PORT_3,
226 g_configs[LCP_FIELD_PORT_S3].baudrate,
227 g_configs[LCP_FIELD_PORT_S3].frame);
228 hal_uart_begin(HAL_UART_PORT_2,
229 g_configs[LCP_FIELD_PORT_S4].baudrate,
230 g_configs[LCP_FIELD_PORT_S4].frame);
231
232 s2_clear_rx();
233 s3_clear_rx();
234 s4_clear_rx();
235
236 const uint32_t now_ms = millis();
237
238 for (uint8_t index = 0U; index < LCP_FIELD_PORT_COUNT; ++index)
239 {
240 g_tx_not_before_ms[index] = now_ms;
241 }
