123 void initIo(void)
124 {
125 /*
126 * PD2 — управление направлением RS-485.
127 * PD3 — выбор SPI-регистра светодиодов.
128 * PD4 — выбор SPI-регистра режимов аналоговых входов.
129 * PD5 — разрешение сдвиговых регистров.
130 */
131 DDRD |= static_cast<uint8_t>(
132 RS485_DE_MASK |
133 LED_SS_MASK |
134 MODE_SS_MASK |
135 SHIFT_ENABLE_MASK
136 );
137
138 clearPortBits(PORTD, RS485_DE_MASK);
139 setPortBits(PORTD, static_cast<uint8_t>(LED_SS_MASK | MODE_SS_MASK | SHIFT_ENABLE_MASK));
140
141 /*
142 * PD6, PD7 — адресные входы с внутренней подтяжкой.
143 */
144 DDRD &= static_cast<uint8_t>(~ADDR_D_MASK);
145 setPortBits(PORTD, ADDR_D_MASK);
146
147 /*
148 * PB0, PB1, PB2 — адресные входы с внутренней подтяжкой.
149 * PB2 после чтения аппаратного адреса используется как линия SPI SS.
150 */
151 DDRB &= static_cast<uint8_t>(~ADDR_B_MASK);
152 setPortBits(PORTB, ADDR_B_MASK);
153
154 /*
155 * PC0 — светодиод состояния связи.
156 */
157 DDRC |= STATUS_LED_MASK;
158 clearPortBits(PORTC, STATUS_LED_MASK);
159
160 s_rs485TransmitMode = 0U;
161 }

163 uint8_t readSlaveAddress(void)
164 {
165 uint8_t address = 0U;

180 }

185 }
186
187 if (isPinLow(PINB, _BV(PB2)))
188 {
189 address |= static_cast<uint8_t>(1U << 4);
190 }
