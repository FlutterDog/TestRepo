10 * @details
11 * Модуль реализует:

14 * - функцию hal::delay();

26/**
27 * @brief Счётчик миллисекунд с момента запуска системного тикера.
28 *
29 * @details
30 * Значение увеличивается в обработчике прерывания TIMER0_COMPA_vect.
31 *
32 * @note
33 * Переменная имеет размер 32 бита. На восьмибитных AVR чтение такого значения
34 * должно выполняться атомарно относительно прерываний.

36static volatile uint32_t s_millis = 0;

40 /**
41 * @brief Атомарно читает счётчик миллисекунд.
42 *
43 * @return Текущее значение счётчика миллисекунд.
44 *
45 * @details
46 * На восьмибитных AVR чтение 32-битной переменной может быть прервано
47 * обработчиком ISR. Функция кратковременно запрещает прерывания,
48 * считывает значение и восстанавливает прежнее состояние регистра SREG.
49 */
50 static inline uint32_t atomicReadMillis(void)
51 {
52 const uint8_t sreg = SREG;
53
54 cli();
55
56 const uint32_t value = s_millis;
57
58 SREG = sreg;
59
60 return value;
61 }

66 /**
67 * @brief Инициализирует системный миллисекундный тикер.
68 *
69 * @details
70 * Настраивает Timer0:
71 * - режим CTC;
72 * - предделитель 64;
73 * - сравнение по OCR0A;
74 * - прерывание TIMER0_COMPA_vect.
75 *
76 * Для F_CPU = 16000000 Гц:
77 * 16000000 / 64 / (249 + 1) = 1000 Гц.
78 *
79 * @note
80 * Функция сбрасывает счётчик Timer0, флаг сравнения и программный
81 * счётчик миллисекунд.
82 */
83 void tick_init(void)
84 {
85 const uint8_t sreg = SREG;
86
87 cli();
88
89 s_millis = 0u;
90
91 TCCR0A = 0u;
92 TCCR0B = 0u;
93 TCNT0 = 0u;
94
95 TCCR0A = static_cast<uint8_t>(_BV(WGM01));
96 OCR0A = 249u;
97
98 /*
99 * Сброс возможного ранее установленного флага совпадения.
100 * На AVR флаг очищается записью логической единицы.
101 */
102 TIFR0 = static_cast<uint8_t>(_BV(OCF0A));
103
104 TIMSK0 = static_cast<uint8_t>(_BV(OCIE0A));
105 TCCR0B = static_cast<uint8_t>(_BV(CS01) | _BV(CS00));
106
107 SREG = sreg;
108 }

145 /**
146 * @brief Выполняет блокирующую задержку в микросекундах.
147 *
148 * @param us Длительность задержки в микросекундах.
149 *
150 * @details
151 * Задержка построена на _delay_loop_2(). Один цикл _delay_loop_2(N)
152 * занимает приблизительно 4 * N тактов CPU.
153 *
154 * Для F_CPU = 16000000 Гц используется 4 итерации на 1 мкс.
155 *
156 * @warning
157 * Точность функции зависит от F_CPU, оптимизации компилятора и накладных
158 * расходов на вызов функции и цикл.
159 */
160 void delayMicroseconds(uint16_t us)
161 {
162 const uint16_t iters_per_us =
163 static_cast<uint16_t>(F_CPU / 4000000UL);
164
165 if (iters_per_us == 0u)
166 {
167 return;
168 }
169
170 while (us-- != 0u)
171 {
172 _delay_loop_2(iters_per_us);
173 }
174 }
175}
