426 /**
427 * @brief Читает значение float из EEPROM.
428 *
429 * @param address Адрес первого байта значения.
430 * @param defaultValue Значение по умолчанию при пустой ячейке EEPROM.
431 *
432 * @return Прочитанное значение или значение по умолчанию.
433 */
434 static float eepromReadFloat(uint16_t address, float defaultValue)
435 {
436 union
437 {
438 float f;
439 uint8_t b[4];
440 } u;
441
442 uint8_t empty = 1U;
443
444 for (uint8_t i = 0U; i < 4U; ++i)
445 {
446 u.b[i] = eepromReadByte(static_cast<uint16_t>(address + i));
447
448 if (u.b[i] != 0xFFU)
449 {
450 empty = 0U;
451 }
452 }
453
454 if (empty != 0U)
455 {
456 return defaultValue;
457 }
458
459 return u.f;
460 }

481 /**
482 * @brief Возвращает один Modbus-регистр из IEEE-754 значения float.
483 *
484 * @param value Значение float.
485 * @param order Индекс шестнадцатибитного слова: 0 или 1.
486 *
487 * @return Шестнадцатибитное слово для передачи через Modbus.
488 */
489 static int16_t convertFloat(float value, uint8_t order)
490 {
491 union
492 {
493 float flt;
494 int32_t i32;
495 uint8_t bytes[4];
496 } u;
497
498 u.flt = value;
499
500 return static_cast<int16_t>(
501 (static_cast<uint16_t>(u.bytes[1U + 2U * order]) << 8) |
502 static_cast<uint16_t>(u.bytes[2U * order])
503 );
504 }

553 /**
554 * @brief Рассчитывает коэффициенты линейной калибровки канала.
555 *
556 * @param channel Номер канала в диапазоне 1...8.
557 */
558 static void calculateCalibration(uint8_t channel)
559 {
560 const int16_t x1 =
561 sensorAnalogX[static_cast<uint8_t>(2U * channel - 1U)];
562
563 const int16_t x2 =
564 sensorAnalogX[static_cast<uint8_t>(2U * channel)];
565
566 const int16_t y1 =
567 gasConcentrationY[static_cast<uint8_t>(2U * channel - 1U)];
568
569 const int16_t y2 =
570 gasConcentrationY[static_cast<uint8_t>(2U * channel)];
571
572 if (x2 == x1)
573 {
574 k[channel] = 1.0F;
575 b[channel] = 0.0F;
576 return;
577 }
578
579 k[channel] =
580 1.0F * static_cast<float>(y2 - y1) / static_cast<float>(x2 - x1);
581
582 b[channel] =
583 static_cast<float>(x1) - static_cast<float>(y1) / k[channel];
584 }
