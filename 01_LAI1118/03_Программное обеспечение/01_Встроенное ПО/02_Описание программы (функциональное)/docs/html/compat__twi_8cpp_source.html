21 *
22 * @details
23 * Значение задано в итерациях цикла ожидания. Таймаут защищает основной

29 /**
30 * @brief Ожидает завершения текущей операции TWI.
31 *
32 * @return true — флаг TWINT установлен; false — истёк таймаут ожидания.
33 */
34 static bool waitTwint(void)
35 {
36 uint16_t timeout = TWI_TIMEOUT;
37
38 while ((TWCR & _BV(TWINT)) == 0U)
39 {
40 if (timeout == 0U)
41 {
42 return false;
43 }
44
45 --timeout;
46 }
47
48 return true;
49 }

51 /**
52 * @brief Формирует START или repeated START на шине TWI.
53 *
54 * @return true — условие START сформировано успешно; false — ошибка или таймаут.
55 */
56 static bool startCondition(void)
57 {
58 TWCR = static_cast<uint8_t>(_BV(TWINT) | _BV(TWSTA) | _BV(TWEN));

207 /*
208 * Prescaler = 1.

211 TWSR = 0U;

213 uint32_t twbrValue = 0UL;
