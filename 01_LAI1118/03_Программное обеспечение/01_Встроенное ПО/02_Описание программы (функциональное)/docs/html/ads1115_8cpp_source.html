175uint16_t ADS1115::buildSingleShotConfig(uint8_t channel) const
176{
177 const uint16_t mux = static_cast<uint16_t>(
178 static_cast<uint16_t>(ADS1115_SINGLE_ENDED_MUX_BASE + channel) << ADS1115_MUX_SHIFT
179 );
180
181 const uint16_t pga = static_cast<uint16_t>(
182 static_cast<uint16_t>(gain_ & 0x07U) << ADS1115_PGA_SHIFT
183 );
184
185 const uint16_t mode = (mode_ != 0U)
186 ? ADS1115_MODE_SINGLE_SHOT
187 : ADS1115_MODE_CONTINUOUS;
188
189 const uint16_t dataRate = static_cast<uint16_t>(
190 static_cast<uint16_t>(dataRate_ & 0x07U) << ADS1115_DATA_RATE_SHIFT
191 );
192
193 return static_cast<uint16_t>(
194 ADS1115_OS_START |
195 mux |
196 pga |
197 mode |
198 dataRate |
199 ADS1115_COMP_DISABLE
200 );
201}
