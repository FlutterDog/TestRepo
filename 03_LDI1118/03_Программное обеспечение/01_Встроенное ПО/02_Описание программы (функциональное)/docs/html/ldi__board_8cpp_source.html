3/**
4 * @file ldi_board.cpp
5 * @brief Реализация аппаратного слоя платы LDI.
6 *
7 * @details
8 * В этом модуле сосредоточена привязка сигналов LDI к портам ATmega328P:
9 * - адресный переключатель PC0...PC4;
10 * - RS-485 direction на PD2;
11 * - 74HC165: latch PD3, clock PD4, data PD5;
12 * - 74HC595: latch PD3, clock PD4, data PB1;
13 * - светодиод состояния связи PB0.
14 *
15 * Линии latch и clock используются совместно для 74HC165 и 74HC595.
16 * Поэтому при чтении входов аппаратный слой сохраняет корректное состояние
17 * LED-панели и восстанавливает его после тактирования 74HC165.
18 */

178 /**
179 * @brief Выполняет захват состояния параллельных входов 74HC165.
180 *
181 * @details
182 * Для 74HC165 линия SH/LD активна низким уровнем.
183 * Последовательность:
184 * - перевод линии в высокий уровень;
185 * - кратковременный низкий уровень для загрузки параллельных входов;
186 * - возврат в высокий уровень для последовательного сдвига.
187 */

190 PORTD |= SR_LATCH_MASK;
191 ioNop();
192
193 PORTD &= static_cast<uint8_t>(~SR_LATCH_MASK);
194 ioNop();
195
196 PORTD |= SR_LATCH_MASK;
197 ioNop();
198 }

211 /**
212 * @brief Зеркально разворачивает младшие пять бит.
213 *
214 * @param value Входное значение.
215 *
216 * @return Значение с развёрнутыми пятью младшими битами.
217 *
218 * @details
219 * Используется для преобразования порядка битов адресного переключателя.
220 */
221 uint8_t mirrorFiveBits(uint8_t value)
222 {
223 uint8_t result = 0u;
224
225 if ((value & 0x01u) != 0u) { result |= 0x10u; } // PC0 -> bit 4
226 if ((value & 0x02u) != 0u) { result |= 0x08u; } // PC1 -> bit 3
227 if ((value & 0x04u) != 0u) { result |= 0x04u; } // PC2 -> bit 2
228 if ((value & 0x08u) != 0u) { result |= 0x02u; } // PC3 -> bit 1
229 if ((value & 0x10u) != 0u) { result |= 0x01u; } // PC4 -> bit 0
230
231 return result;
232 }
233
234 /**
235 * @brief Возвращает состояние бита в маске.
236 *
237 * @param value Битовая маска.
238 * @param bit Номер бита.
239 *
240 * @return true — бит установлен; false — бит сброшен.
241 */
242 bool isBitSet(uint16_t value, uint8_t bit)
243 {
244 return ((value & static_cast<uint16_t>(1u << bit)) != 0u);
245 }
246
247 /**
248 * @brief Передаёт маску в 74HC595 без изменения сохранённого состояния.
249 *
250 * @param state Битовая маска светодиодов.
251 *
252 * @details
253 * Сначала передаются исходные bit 7...bit 0, затем bit 15...bit 8.
254 * Такой порядок соответствует разводке LED-панели.
255 */
256 void transferPanelMask(uint16_t state)
257 {
258 for (int8_t bit = 7; bit >= 0; --bit)
259 {
260 ledShiftDataWrite(isBitSet(state, static_cast<uint8_t>(bit)));
261 shiftClockPulse();
262 }
263
264 for (int8_t bit = 15; bit >= 8; --bit)
265 {
266 ledShiftDataWrite(isBitSet(state, static_cast<uint8_t>(bit)));
267 shiftClockPulse();
268 }
269
270 ledShiftDataWrite(false);
271 latch595Pulse();
272 }
273
274 /**
275 * @brief Восстанавливает состояние LED-панели после чтения входов.
276 *
277 * @details
278 * При чтении 74HC165 линия clock также тактирует 74HC595. Поэтому после
279 * чтения входов необходимо повторно загрузить последнюю LED-маску.
280 */
281 void restorePanelMask(void)
282 {
283 transferPanelMask(s_lastPanelMask);
284 }

289 /**
290 * @brief Инициализирует аппаратные линии платы LDI.
291 */
292 void initIo(void)
293 {
294 /*
295 * Адресный переключатель: PC0...PC4, входы с внутренней подтяжкой.
296 * Переключатели активны низким уровнем.
297 */
298 DDRC &= static_cast<uint8_t>(~ADDRESS_INPUTS_MASK);
299 PORTC |= ADDRESS_INPUTS_MASK;
300
301 /*
302 * UART/RS-485:
303 * - PD0 RXD, вход без внутренней подтяжки;
304 * - PD1 TXD, выход;
305 * - PD2 DE/RE, выход, начально режим приёма.
306 */
307 DDRD &= static_cast<uint8_t>(~_BV(PD0));
308 DDRD |= static_cast<uint8_t>(_BV(PD1) | RS485_DE_MASK);
309
310 PORTD &= static_cast<uint8_t>(~_BV(PD0));
311 PORTD &= static_cast<uint8_t>(~_BV(PD1));
312 rs485ReceiveMode();
313
314 /*
315 * Сдвиговые регистры:
316 * - PD3 latch, выход;
317 * - PD4 clock, выход;
318 * - PD5 data from 74HC165, вход без внутренней подтяжки;
319 * - PB1 data to 74HC595, выход.
320 */
321 DDRD |= static_cast<uint8_t>(SR_LATCH_MASK | SR_CLOCK_MASK);
322 DDRD &= static_cast<uint8_t>(~SR_INPUT_DATA_MASK);
323
324 PORTD &= static_cast<uint8_t>(~SR_LATCH_MASK);
325 PORTD &= static_cast<uint8_t>(~SR_CLOCK_MASK);
326 PORTD &= static_cast<uint8_t>(~SR_INPUT_DATA_MASK);
327
328 DDRB |= SR_LED_DATA_MASK;
329 PORTB &= static_cast<uint8_t>(~SR_LED_DATA_MASK);
330
331 /*
332 * Светодиод состояния связи: PB0, выход, начально выключен.
333 */
334 DDRB |= STATUS_LED_MASK;
335 statusLedWrite(false);
336
337 /*
338 * Начальное состояние панели индикации.
339 */
340 writePanelMask(0x0000u);
341 }

343 /**
344 * @brief Считывает Modbus-адрес с аппаратного адресного переключателя.
345 *
346 * @return Modbus-адрес устройства.
347 */
348 uint8_t readSlaveAddress(void)
349 {
350 /*
351 * Входы подтянуты к Vcc, активное положение переключателя даёт ноль.
352 * Поэтому состояние входов инвертируется.
353 */
354 const uint8_t raw =
355 static_cast<uint8_t>((~PINC) & ADDRESS_INPUTS_MASK);
356
357 uint8_t addr = mirrorFiveBits(raw);
358
359 if (addr == 0u)
360 {
361 addr = 1u;
362 }
363
364 return addr;
365 }

429 state |= 0x0001u;

431

433 }

436 restorePanelMask();
437
438 return state;
439 }
