473 /**
474 * @brief Обрабатывает обращение к Holding-регистрам Modbus.
475 *
476 * @param regs Массив регистров ответа.
477 * @param startAddr Начальный адрес регистра.
478 * @param regCount Количество обрабатываемых регистров.
479 * @param value Значение регистра при операции записи.
480 *
481 * @details
482 * Карта регистров LDI:
483 * - HR 0: состояние шестнадцати дискретных входов;
484 * - HR 39: серийный номер устройства;
485 * - HR 40: версия встроенного ПО;
486 * - HR 100: Modbus-адрес устройства в RAM;
487 * - HR 591: запись и чтение серийного номера;
488 * - HR 594: программная перезагрузка;
489 * - HR 900: отложенный переход в bootloader;
490 * - HR 1000: запись и чтение аппаратной версии устройства;
491 * - HR 1001: идентификатор модуля.
492 *
493 * При каждом корректном обращении обновляется таймер контроля связи.
494 */
495 void modbusProceed(
496 int16_t* regs,
497 int16_t startAddr,
498 int16_t regCount,
499 int16_t value
500 )
501 {
502 int16_t registerAddr = startAddr;
503 bool writeReg = false;
504
505 lastModbusRequestTime = currentTime;
506
507 if ((IGAS_mb_X.func == FC_WRITE_REGS) || (IGAS_mb_X.func == FC_WRITE_REG))
508 {
509 writeReg = true;
510 }
511
512 if (communicationState == 0x00U)
513 {
514 communicationState = 0x01U;
515 statusLedWrite(true);
516 }
517
518 for (int16_t i = 0; i < regCount; ++i)
519 {
520 switch (registerAddr)
521 {
522 case 0:
523 /*
524 * HR 0.
525 * Read: текущее состояние шестнадцати дискретных входов.
526 */
527 regs[i] = static_cast<int16_t>(inputState);
528 break;
529
530 case 39:
531 /*
532 * HR 39.
533 * Read: серийный номер устройства.
534 */
535 regs[i] = static_cast<int16_t>(
536 eepromReadWord(EEPROM_DEVICE_SERIAL_NUMBER_ADDR)
537 );
538 break;
539
540 case 40:
541 /*
542 * HR 40.
543 * Read: версия встроенного программного обеспечения.
544 */
545 regs[i] = static_cast<int16_t>(DEVICE_SOFTWARE_VERSION);
546 break;
547
548 case 100:
549 /*
550 * HR 100.
551 * Write: временное изменение Modbus-адреса устройства.
552 * Read: текущий Modbus-адрес устройства.
553 *
554 * Адрес, записанный через этот регистр, действует до перезапуска.
555 * При старте адрес снова считывается с аппаратного переключателя.
556 */
557 if (writeReg)
558 {
559 if ((value > 0) && (value < 248))
560 {
561 slaveAddr = static_cast<uint8_t>(value);
562 IGAS_mb_X.slave = slaveAddr;
563 }
564 }
565
566 regs[i] = static_cast<int16_t>(slaveAddr);
567 break;
568
569 case 591:
570 /*
571 * HR 591.
572 * Write: запись серийного номера устройства.
573 * Read: текущий серийный номер устройства.
574 */
575 if (writeReg)
576 {
577 eepromUpdateWord(
578 EEPROM_DEVICE_SERIAL_NUMBER_ADDR,
579 static_cast<uint16_t>(value)
580 );
581 }
582
583 regs[i] = static_cast<int16_t>(
584 eepromReadWord(EEPROM_DEVICE_SERIAL_NUMBER_ADDR)
585 );
586 break;
587
588 case 594:
589 /*
590 * HR 594.
591 * Write 1: программная перезагрузка через watchdog.
592 * Read: состояние запроса перезагрузки.
593 */
594 if (writeReg && (value == 1))
595 {
596 rebootRequested = 0x01U;
597 }
598
599 regs[i] = static_cast<int16_t>(rebootRequested);
600 break;
601
602 case 900:
603 /*
604 * HR 900.
605 * Write: отложенный переход в bootloader.
606 * Read: состояние запроса перехода.
607 */
608 if (writeReg)
609 {
610 bootloaderRequestTime = currentTime;
611 }
612
613 regs[i] = (bootloaderRequestTime != 0UL) ? 1 : 0;
614 break;
615
616 case 1000:
617 /*
618 * HR 1000.
619 * Write: запись аппаратной версии устройства.
620 * Read: текущая аппаратная версия устройства.
621 */
622 if (writeReg)
623 {
624 eepromUpdateWord(
625 EEPROM_DEVICE_HARDWARE_VERSION_ADDR,
626 static_cast<uint16_t>(value)
627 );
628 }
629
630 regs[i] = static_cast<int16_t>(
631 eepromReadWord(EEPROM_DEVICE_HARDWARE_VERSION_ADDR)
632 );
633 break;
634
635 case 1001:
636 /*
637 * HR 1001.
638 * Read: идентификатор модуля.
639 */
640 regs[i] = static_cast<int16_t>(DEVICE_ID);
641 break;
642
643 default:
644 /*
645 * Неподдерживаемый регистр.
646 */
647 regs[i] = -1;
648 break;
649 }
650
651 ++registerAddr;
652 }
653 }

659 /**
660 * @brief Передаёт управление в bootloader.
661 */
662 void jumpToBootloader(void)
663 {
664 cli();
665
666 wdt_disable();
667
668#ifdef UCSR0B
669 UCSR0B = 0U;
670#endif
671
672#ifdef SPCR
673 SPCR = 0U;
674#endif
675
676#ifdef TWCR
677 TWCR = 0U;
678#endif
679
680 /*
681 * Восстановление нулевого регистра ABI AVR-GCC.
682 */
683 asm volatile ("clr __zero_reg__");
684
685 /*
686 * Установка стека в конец SRAM.
687 */
688 SP = RAMEND;
689
690 /*
691 * Перенос таблицы векторов прерываний в boot-секцию.
692 * Последовательность IVCE -> IVSEL должна выполняться без задержки
693 * между двумя записями.
694 */
695 MCUCR = static_cast<uint8_t>(_BV(IVCE));
696 MCUCR = static_cast<uint8_t>(_BV(IVSEL));
697
698 /*
699 * Прямой переход в bootloader.
700 * Адрес задаётся в словах flash.
701 */
702 reinterpret_cast<BootEntryPtr>(BOOT_START_ADDR_WORDS)();
703 }
704}

710/**
711 * @brief Выполняет однократную инициализацию приложения LDI.
712 */
713void setup(void)
714{
715 /*
716 * После сброса по watchdog флаг WDRF может удерживать watchdog включённым.
717 * Поэтому сначала выполняется сброс watchdog, очистка WDRF, затем настройка
718 * требуемого таймаута.
719 */
720 wdt_reset();
721 MCUSR &= static_cast<uint8_t>(~_BV(WDRF));
722
723 wdt_disable();
724 wdt_enable(WDTO_1S);
725
726 initBoardIo();
727
728 slaveAddr = readSlaveAddress();
729
730 currentTime = millis();
731 lastModbusRequestTime = currentTime;
732 loop100StartTime = currentTime;
733 loop500StartTime = currentTime;
734
